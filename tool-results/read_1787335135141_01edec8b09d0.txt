   815→// SM-2 Revision Review Helper
   816→// ═════════════════════════════════════════════════════════════════════
   817→/**
   818→ * Review a revision item using the SM-2 algorithm.
   819→ * @param store - the zustand store
   820→ * @param itemId - the revision item id
   821→ * @param quality - quality of recall (0-5). >= 3 means pass.
   822→ * @returns void (mutates the store)
   823→ */
   824→export function reviewRevisionItem(
   825→  store: ReturnType<typeof useStore>,
   826→  itemId: string,
   827→  quality: number
   828→) {
   829→  const item = store.getState().revisionItems.find((r) => r.id === itemId);
   830→  if (!item) return;
   831→
   832→  const q = Math.max(0, Math.min(5, quality));
   833→  let { easeFactor, interval, repetitions } = item;
   834→
   835→  if (q >= 3) {
   836→    // Good response
   837→    if (repetitions === 0) {
   838→      interval = 1;
   839→    } else if (repetitions === 1) {
   840→      interval = 6;
   841→    } else {
   842→      interval = Math.round(interval * easeFactor);
   843→    }
   844→    repetitions += 1;
   845→    easeFactor =
   846→      Math.max(1.3, easeFactor + 0.1 - (5 - q) * (0.08 + (5 - q) * 0.02));
   847→  } else {
   848→    // Bad response - reset
   849→    repetitions = 0;
   850→    interval = 1;
   851→  }
   852→
   853→  const nextReviewDate = new Date();
   854→  nextReviewDate.setDate(nextReviewDate.getDate() + interval);
   855→  const nextReview = nextReviewDate.toISOString().split('T')[0];
   856→
   857→  store.getState().updateRevisionItem(itemId, {
   858→    easeFactor,
   859→    interval,
   860→    repetitions,
   861→    nextReview,
   862→    lastReview: todayStr(),
   863→  });
   864→}

   870→/** Syllabus completion % for a subject */
   871→export function getSubjectProgress(
   872→  state: Pick<AppState, 'syllabusUnits'>,
   873→  subjectId: string
   874→): number {
   875→  const units = state.syllabusUnits.filter((u) => u.subjectId === subjectId);

   877→  if (allTopics.length === 0) return 0;

   879→  return Math.round((completed / allTopics.length) * 100);

   883→export function getSubjectAttendance(
   884→  state: Pick<AppState, 'attendance'>,
   885→  subjectId: string
   886→): { present: number; total: number; percentage: number } {
   887→  const records = state.attendance.filter((a) => a.subjectId === subjectId);

   891→  return { present, total, percentage };

   915→/** Current semester SGPA */
   916→export function calculateSGPA(state: Pick<AppState, 'subjects' | 'assessments'>): number {
   917→  const activeSubjects = state.subjects.filter((s) => !s.archived);
   918→  if (activeSubjects.length === 0) return 0;
   919→
   920→  let totalWeightedPoints = 0;
   921→  let totalCredits = 0;
   922→
   923→  for (const subject of activeSubjects) {
   924→    const grade = getSubjectGrade(state, subject.id);
   925→    const gradePoint = GRADE_POINTS[grade] ?? 0;
   926→    totalWeightedPoints += gradePoint * subject.credits;
   927→    totalCredits += subject.credits;
   928→  }
   929→
   930→  return totalCredits > 0
   931→    ? Math.round((totalWeightedPoints / totalCredits) * 100) / 100
   932→    : 0;
   933→}

   935→/** Overall CGPA (currently same as SGPA for single semester data) */
   936→export function calculateCGPA(state: Pick<AppState, 'subjects' | 'assessments'>): number {

   968→/** Health signal for a subject */
   969→export function getSubjectSignal(
   970→  state: AppState,
   971→  subjectId: string
   972→): SignalStatus {
   973→  const att = getSubjectAttendance(state, subjectId);
   974→  const progress = getSubjectProgress(state, subjectId);
   975→  const marks = getSubjectMarks(state, subjectId);
   976→  const dueRevisions = getDueRevisionItems(state).filter(
   977→    (r) => r.subjectId === subjectId
   978→  );
   979→
   980→  // No data at all
   981→  if (att.total === 0 && marks.max === 0 && progress === 0) {
   982→    return 'nodata';
   983→  }
   984→
   985→  // Critical: attendance below threshold or very low marks
   986→  if (
   987→    att.total > 0 && att.percentage < 60
   988→  ) {
   989→    return 'critical';
   990→  }
   991→
   992→  if (
   993→    marks.max > 0 && marks.percentage < 40
   994→  ) {
   995→    return 'critical';
   996→  }
   997→
   998→  // Attention: attendance near threshold, many overdue revisions, or low marks
   999→  if (
  1000→    (att.total > 0 && att.percentage < state.profile.attendanceThreshold) ||
  1001→    dueRevisions.length >= 3 ||
  1002→    (marks.max > 0 && marks.percentage < 55)
  1003→  ) {
  1004→    return 'attention';
  1005→  }
  1006→
  1007→  // Improving: some progress but not great
  1008→  if (
  1009→    progress > 0 && progress < 50 &&
  1010→    (marks.max === 0 || marks.percentage < 70)
  1011→  ) {
  1012→    return 'improving';
  1013→  }
  1014→
  1015→  // Healthy: good attendance, decent progress
  1016→  if (
  1017→    (att.total === 0 || att.percentage >= state.profile.attendanceThreshold) &&
  1018→    progress >= 50
  1019→  ) {
  1020→    return 'healthy';
  1021→  }
  1022→
  1023→  return 'improving';
  1024→}

  1026→/** Overall semester health score 0-100 */
  1027→export function getSemesterHealth(state: AppState): number {
  1028→  const activeSubjects = state.subjects.filter((s) => !s.archived);
  1029→  if (activeSubjects.length === 0) return 0;
  1030→
  1031→  let totalScore = 0;
  1032→  let maxScore = 0;
  1033→
  1034→  for (const subject of activeSubjects) {
  1035→    // Attendance score (30 points max)
  1036→    const att = getSubjectAttendance(state, subject.id);
  1037→    const attScore = att.total > 0 ? Math.min(30, (att.percentage / 100) * 30) : 10;
  1038→
  1039→    // Marks score (30 points max)
  1040→    const marks = getSubjectMarks(state, subject.id);
  1041→    const marksScore = marks.max > 0 ? (marks.percentage / 100) * 30 : 10;
  1042→
  1043→    // Syllabus progress (20 points max)
  1044→    const progress = getSubjectProgress(state, subject.id);
  1045→    const progressScore = (progress / 100) * 20;
  1046→
  1047→    // Revision health (20 points max)
  1048→    const subjectRevisions = state.revisionItems.filter(
  1049→      (r) => r.subjectId === subject.id
  1050→    );
  1051→    const overdueRevisions = subjectRevisions.filter(
  1052→      (r) => r.nextReview <= todayStr()
  1053→    ).length;
  1054→    const revisionScore = subjectRevisions.length > 0
  1055→      ? Math.max(0, 20 - overdueRevisions * 5)
  1056→      : 10;
  1057→
  1058→    totalScore += attScore + marksScore + progressScore + revisionScore;
  1059→    maxScore += 100;
  1060→  }
  1061→
  1062→  return Math.round((totalScore / maxScore) * 100);
  1063→}

  1065→/** Total study time this week (in seconds) */
  1066→export function getStudyTimeThisWeek(
  1067→  state: Pick<AppState, 'studySessions'>
  1068→): number {
  1069→  const now = new Date();
  1070→  const dayOfWeek = now.getDay(); // 0=Sun
  1071→   // Start of week: Monday (day 1) or Sunday (day 0) — we use Sunday as start
  1072→  const startOfWeek = new Date(now);
  1073→  startOfWeek.setDate(now.getDate() - dayOfWeek);
  1074→  startOfWeek.setHours(0, 0, 0, 0);
  1075→
  1076→  const startStr = startOfWeek.toISOString().split('T')[0];
  1077→
  1078→  return state.studySessions
  1079→    .filter((s) => s.date >= startStr)
  1080→    .reduce((sum, s) => sum + s.duration, 0);
  1081→}
