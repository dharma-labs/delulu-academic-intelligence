   814→   814→// ═════════════════════════════════════════════════════════════════════
   815→   815→// SM-2 Revision Review Helper
   816→   816→// ═════════════════════════════════════════════════════════════════════
   817→   817→/**
   818→   818→ * Review a revision item using the SM-2 algorithm.
   819→   819→ * @param store - the zustand store
   820→   820→ * @param itemId - the revision item id
   821→   821→ * @param quality - quality of recall (0-5). >= 3 means pass.
   822→   822→ * @returns void (mutates the store)
   823→   823→ */
   824→   824→export function reviewRevisionItem(
   825→   825→  store: ReturnType<typeof useStore>,
   826→   826→  itemId: string,
   827→   827→  quality: number
   828→   828→) {
   829→   829→  const item = store.getState().revisionItems.find((r) => r.id === itemId);
   830→   830→  if (!item) return;
   831→   831→
   832→   832→  const q = Math.max(0, Math.min(5, quality));
   833→   833→  let { easeFactor, interval, repetitions } = item;
   834→   834→
   835→   835→  if (q >= 3) {
   836→   836→    // Good response
   837→   837→    if (repetitions === 0) {
   838→   838→      interval = 1;
   839→   839→    } else if (repetitions === 1) {
   840→   840→      interval = 6;
   841→   841→    } else {
   842→   842→      interval = Math.round(interval * easeFactor);
   843→   843→    }
   844→   844→    repetitions += 1;
   845→   845→    easeFactor =
   846→   846→      Math.max(1.3, easeFactor + 0.1 - (5 - q) * (0.08 + (5 - q) * 0.02));
   847→   847→  } else {
   848→   848→    // Bad response - reset
   849→   849→    repetitions = 0;
   850→   850→    interval = 1;
   851→   851→  }
   852→   852→
   853→   853→  const nextReviewDate = new Date();
   854→   854→  nextReviewDate.setDate(nextReviewDate.getDate() + interval);
   855→   855→  const nextReview = nextReviewDate.toISOString().split('T')[0];
   856→   856→
   857→   857→  store.getState().updateRevisionItem(itemId, {
   858→   858→    easeFactor,
   859→   859→    interval,
   860→   860→    repetitions,
   861→   861→    nextReview,
   862→   862→    lastReview: todayStr(),
   863→   863→  });
   864→   864→}

   968→   968→/** Health signal for a subject */
   969→   969→export function getSubjectSignal(
   970→   970→  state: AppState,
   971→   971→  subjectId: string
   972→   972→): SignalStatus {
   973→   973→  const att = getSubjectAttendance(state, subjectId);
   974→   974→  const progress = getSubjectProgress(state, subjectId);
   975→   975→  const marks = getSubjectMarks(state, subjectId);
   976→   976→  const dueRevisions = getDueRevisionItems(state).filter(
   977→   977→    (r) => r.subjectId === subjectId
   978→   978→  );
   979→   979→
   980→   980→  // No data at all
   981→   981→  if (att.total === 0 && marks.max === 0 && progress === 0) {
   982→   982→    return 'nodata';
   983→   983→  }
   984→   984→
   985→   985→  // Critical: attendance below threshold or very low marks
   986→   986→  if (
   987→   987→    att.total > 0 && att.percentage < 60
   988→   988→  ) {
   989→   989→    return 'critical';
   990→   990→  }
   991→   991→
   992→   992→  if (
   993→   993→    marks.max > 0 && marks.percentage < 40
   994→   994→  ) {
   995→   995→    return 'critical';
   996→   996→  }
   997→   997→
   998→   998→  // Attention: attendance near threshold, many overdue revisions, or low marks
   999→   999→  if (
  1000→  1000→    (att.total > 0 && att.percentage < state.profile.attendanceThreshold) ||
  1001→  1001→    dueRevisions.length >= 3 ||
  1002→  1002→    (marks.max > 0 && marks.percentage < 55)
  1003→  1003→  ) {
  1004→  1004→    return 'attention';
  1005→  1005→  }
  1006→  1006→
  1007→  1007→  // Improving: some progress but not great
  1008→  1008→  if (
  1009→  1009→    progress > 0 && progress < 50 &&
  1010→  1010→    (marks.max === 0 || marks.percentage < 70)
  1011→  1011→  ) {
  1012→  1012→    return 'improving';
  1013→  1013→  }
  1014→  1014→
  1015→  1015→  // Healthy: good attendance, decent progress
  1016→  1016→  if (
  1017→  1017→    (att.total === 0 || att.percentage >= state.profile.attendanceThreshold) &&
  1018→  1018→    progress >= 50
  1019→  1019→  ) {
  1020→  1020→    return 'healthy';
  1021→  1021→  }
  1022→  1022→
  1023→  1023→  return 'improving';
  1024→  1024→}

  1026→  1026→/** Overall semester health score 0-100 */
  1027→  1027→export function getSemesterHealth(state: AppState): number {
  1028→  1028→  const activeSubjects = state.subjects.filter((s) => !s.archived);
  1029→  1029→  if (activeSubjects.length === 0) return 0;
  1030→  1030→
  1031→  1031→  let totalScore = 0;
  1032→  1032→  let maxScore = 0;
  1033→  1033→
  1034→  1034→  for (const subject of activeSubjects) {
  1035→  1035→    // Attendance score (30 points max)
  1036→  1036→    const att = getSubjectAttendance(state, subject.id);
  1037→  1037→    const attScore = att.total > 0 ? Math.min(30, (att.percentage / 100) * 30) : 10;
  1038→  1038→
  1039→  1039→    // Marks score (30 points max)
  1040→  1040→    const marks = getSubjectMarks(state, subject.id);
  1041→  1041→    const marksScore = marks.max > 0 ? (marks.percentage / 100) * 30 : 10;
  1042→  1042→
  1043→  1043→    // Syllabus progress (20 points max)
  1044→  1044→    const progress = getSubjectProgress(state, subject.id);
  1045→  1045→    const progressScore = (progress / 100) * 20;
  1046→  1046→
  1047→  1047→    // Revision health (20 points max)
  1048→  1048→    const subjectRevisions = state.revisionItems.filter(
  1049→  1049→      (r) => r.subjectId === subject.id
  1050→  1050→    );
  1051→  1051→    const overdueRevisions = subjectRevisions.filter(
  1052→  1052→      (r) => r.nextReview <= todayStr()
  1053→  1053→    ).length;
  1054→  1054→    const revisionScore = subjectRevisions.length > 0
  1055→  1055→      ? Math.max(0, 20 - overdueRevisions * 5)
  1056→  1056→      : 10;
  1057→  1057→
  1058→  1058→    totalScore += attScore + marksScore + progressScore + revisionScore;
  1059→  1059→    maxScore += 100;
  1060→  1060→  }
  1061→  1061→
  1062→  1062→  return Math.round((totalScore / maxScore) * 100);
  1063→  1063→}
