   123→function signalBadge(signal: SignalStatus) {
   124→  const map: Record<SignalStatus, { label: string; className: string }> = {
   125→    healthy: { label: 'Healthy', className: 'bg-[var(--delulu-success)]/15 text-[var(--delulu-success)] border-[var(--delulu-success)]/20' },
   126→    improving: { label: 'Improving', className: 'bg-[var(--delulu-info)]/15 text-[var(--delulu-info)] border-[var(--delulu-info)]/20' },
   127→    attention: { label: 'Needs Attention', className: 'bg-[var(--delulu-warning)]/15 text-[var(--delulu-warning)] border-[var(--delulu-warning)]/20' },
   128→    critical: { label: 'Critical', className: 'bg-[var(--delulu-danger)]/15 text-[var(--delulu-danger)] border-[var(--delulu-danger)]/20' },
   129→    upcoming: { label: 'Upcoming', className: 'bg-[var(--delulu-purple)]/15 text-[var(--delulu-purple)] border-[var(--delulu-purple)]/20' },
   130→    nodata: { label: 'No Data', className: 'bg-muted text-muted-foreground border-border' },
   131→  };
   132→  const s = map[signal];
   133→  return <Badge variant="outline" className={s.className}>{s.label}</Badge>;
   134→}
   135→
   136→function pctColor(pct: number): string {
   137→  if (pct >= 75) return 'text-[var(--delulu-success)]';
   138→  if (pct >= 50) return 'text-[var(--delulu-info)]';
   139→  if (pct >= 30) return 'text-[var(--delulu-warning)]';
   140→  return 'text-[var(--delulu-danger)]';
   141→}

   143→function formatDuration(seconds: number): string {
   144→  const h = Math.floor(seconds / 3600);
   145→  const m = Math.floor((seconds % 3600) / 60);
   146→  if (h > 0) return `${h}h ${m}m`;
   147→  return `${m}m`;
   148→}

   158→function OverviewTab({ subjectId }: { subjectId: string }) {
   159→  const subjects = useStore((s) => s.subjects);
   160→  const allSyllabusUnits = useStore((s) => s.syllabusUnits);
   161→  const allAssessments = useStore((s) => s.assessments);
   162→  const allAttendance = useStore((s) => s.attendance);
   163→  const allStudySessions = useStore((s) => s.studySessions);
   164→  const allExams = useStore((s) => s.exams);
   165→  const profile = useStore((s) => s.profile);
   166→  const subject = subjects.find((x) => x.id === subjectId);
   167→  const syllabusUnits = useMemo(() => allSyllabusUnits.filter((u) => u.subjectId === subjectId), [allSyllabusUnits, subjectId]);
   168→  const assessments = useMemo(() => allAssessments.filter((a) => a.subjectId === subjectId), [allAssessments, subjectId]);
   169→  const attendance = useMemo(() => allAttendance.filter((a) => a.subjectId === subjectId), [allAttendance, subjectId]);
   170→  const studySessions = useMemo(() => allStudySessions.filter((s) => s.subjectId === subjectId), [allStudySessions, subjectId]);
   171→  const exams = useMemo(() => allExams.filter((e) => e.subjectId === subjectId), [allExams, subjectId]);
   172→
   173→  const progress = getSubjectProgress({ syllabusUnits }, subjectId);
   174→  const att = getSubjectAttendance({ attendance }, subjectId);
   175→  const marks = getSubjectMarks({ assessments }, subjectId);
   176→  const grade = getSubjectGrade({ assessments }, subjectId);
   177→
   178→  // CA performance (average assessment %)
   179→  const caPct = assessments.length > 0
   180→    ? Math.round(assessments.reduce((sum, a) => sum + (a.maxMarks > 0 ? (a.obtainedMarks / a.maxMarks) * 100 : 0), 0) / assessments.length)
   181→    : 0;
   182→
   183→  // Next upcoming exam
   184→  const nextExam = exams
   185→    .filter((e) => e.status === 'upcoming')
   186→    .sort((a, b) => a.date.localeCompare(b.date))[0];
   187→
   188→  // Recommended action
   189→  const recommendation = useMemo(() => {
   190→    const recs: string[] = [];
   191→    if (att.total > 0 && att.percentage < profile.attendanceThreshold) {
   192→      recs.push(`Attendance is below ${profile.attendanceThreshold}% threshold. Focus on attending classes.`);
   193→    }
   194→    if (progress < 50 && progress > 0) {
   195→      recs.push('Syllabus is less than halfway done. Prioritize completing more topics.');
   196→    }
   197→    if (progress === 0) {
   198→      recs.push('No syllabus progress yet. Start by adding topics to your units.');
   199→    }
   200→    if (marks.percentage < 50 && marks.max > 0) {
   201→      recs.push('Assessment scores need improvement. Review weak areas before the next test.');
   202→    }
   203→    if (nextExam) {
   204→      const daysUntil = Math.ceil((parseISO(nextExam.date).getTime() - Date.now()) / 86400000);
   205→      if (daysUntil <= 7) {
   206→        recs.push(`${nextExam.name} is in ${daysUntil} day${daysUntil !== 1 ? 's' : ''}. Intensive revision recommended.`);
   207→      }
   208→    }
   209→    if (recs.length === 0) {
   210→      if (progress >= 80 && marks.percentage >= 70) {
   211→        recs.push('Great progress! Keep up the consistency and start revising for exams.');
   212→      } else {
   213→        recs.push('Add more data (attendance, marks, study sessions) to get personalized recommendations.');
   214→      }
   215→    }
   216→    return recs[0];
   217→  }, [att, progress, marks, nextExam, profile.attendanceThreshold]);
   218→
   219→  // Recent activity (last 5 items combined from sessions and assessments)
   220→  const recentActivity = useMemo(() => {
   221→    const items: { type: string; label: string; date: string }[] = [];
   222→    studySessions.forEach((s) =>
   223→      items.push({ type: 'session', label: `Studied for ${formatDuration(s.duration)}`, date: s.date })
   224→    );
   225→    assessments.forEach((a) =>
   226→      items.push({ type: 'assessment', label: `${a.name}: ${a.obtainedMarks}/${a.maxMarks}`, date: a.date })
   227→    );
   228→    return items.sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5);
   229→  }, [studySessions, assessments]);
   230→
   231→  if (!subject) return null;
   232→
   233→  const metrics = [
   234→    {
   235→      label: 'Current Grade',
   236→      value: marks.max > 0 ? grade : '--',
   237→      sub: marks.max > 0 ? `${marks.percentage}% average` : 'No assessments yet',
   238→      icon: GraduationCap,
   239→      color: marks.max > 0 ? pctColor(marks.percentage) : 'text-muted-foreground',
   240→    },
   241→    {
   242→      label: 'Attendance',
   243→      value: att.total > 0 ? `${att.percentage}%` : '--',
   244→      sub: att.total > 0 ? `${att.present}/${att.total} classes` : 'No records yet',
   245→      icon: ClipboardCheck,
   246→      color: att.total > 0
   247→        ? att.percentage >= profile.attendanceThreshold
   248→          ? 'text-[var(--delulu-success)]'
   249→          : 'text-[var(--delulu-danger)]'
   250→        : 'text-muted-foreground',
   251→    },
   252→    {
   253→      label: 'Syllabus',
   254→      value: `${progress}%`,
   255→      sub: progress > 0 ? 'topics completed' : 'No topics yet',
   256→      icon: BookOpen,
   257→      color: progress >= 75 ? 'text-[var(--delulu-success)]' : progress >= 40 ? 'text-[var(--delulu-info)]' : 'text-muted-foreground',
   258→    },
   259→    {
   260→      label: 'CA Performance',
   261→      value: caPct > 0 ? `${caPct}%` : '--',
   262→      sub: caPct > 0 ? `across ${assessments.length} assessment${assessments.length !== 1 ? 's' : ''}` : 'No assessments yet',
   263→      icon: BarChart3,
   264→      color: caPct > 0 ? pctColor(caPct) : 'text-muted-foreground',
   265→    },
   266→  ];
   267→
   268→  return (
   269→    <div className="space-y-6">
   270→      {/* 4 metric cards */}
   271→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   272→        {metrics.map((m) => (
   273→          <motion.div key={m.label} variants={fadeUp} initial="hidden" animate="show">
   274→            <Card className="p-4">
   275→              <div className="flex items-center gap-2 mb-2">
   276→                <m.icon className="h-4 w-4 text-muted-foreground" />
   277→                <span className="text-xs text-muted-foreground font-medium">{m.label}</span>
   278→              </div>
   279→              <p className={`text-2xl font-semibold ${m.color}`}>{m.value}</p>
   280→              <p className="text-xs text-muted-foreground mt-1">{m.sub}</p>
   281→            </Card>
   282→          </motion.div>
   283→        ))}
   284→      </div>
   285→
   286→      {/* Next Assessment + Recommendation */}
   287→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   288→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   289→          <Card className="p-4">
   290→            <div className="flex items-center gap-2 mb-3">
   291→              <CalendarDays className="h-4 w-4 text-muted-foreground" />
   292→              <span className="text-sm font-medium">Next Assessment</span>
   293→            </div>
   294→            {nextExam ? (
   295→              <div>
   296→                <p className="font-medium">{nextExam.name}</p>
   297→                <p className="text-sm text-muted-foreground mt-1">
   298→                  {format(parseISO(nextExam.date), 'dd MMM yyyy')}
   299→                </p>
   300→                <Badge variant="outline" className="mt-2 capitalize">{nextExam.type}</Badge>
   301→              </div>
   302→            ) : (
   303→              <p className="text-sm text-muted-foreground">No upcoming assessments</p>
   304→            )}
   305→          </Card>
   306→        </motion.div>
   307→
   308→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   309→          <Card className="p-4">
   310→            <div className="flex items-center gap-2 mb-3">
   311→              <Target className="h-4 w-4 text-muted-foreground" />
   312→              <span className="text-sm font-medium">Recommended Action</span>
   313→            </div>
   314→            <p className="text-sm leading-relaxed">{recommendation}</p>
   315→          </Card>
   316→        </motion.div>
   317→      </div>
   318→
   319→      {/* Recent Activity */}
   320→      <motion.div variants={fadeUp} initial="hidden" animate="show">
   321→        <Card className="p-4">
   322→          <h3 className="text-sm font-medium mb-3">Recent Activity</h3>
   323→          {recentActivity.length > 0 ? (
   324→            <div className="space-y-3 max-h-48 overflow-y-auto scrollbar-thin">
   325→              {recentActivity.map((item, i) => (
   326→                <div key={i} className="flex items-center gap-3">
   327→                  <div className="h-8 w-8 rounded-full bg-muted flex items-center justify-center shrink-0">
   328→                    {item.type === 'session' ? (
   329→                      <Timer className="h-3.5 w-3.5 text-muted-foreground" />
   330→                    ) : (
   331→                      <BarChart3 className="h-3.5 w-3.5 text-muted-foreground" />
   332→                    )}
   333→                  </div>
   334→                  <div className="min-w-0 flex-1">
   335→                    <p className="text-sm truncate">{item.label}</p>
   336→                    <p className="text-xs text-muted-foreground">{item.date}</p>
   337→                  </div>
   338→                </div>
   339→              ))}
   340→            </div>
   341→          ) : (
   342→            <p className="text-sm text-muted-foreground">No recent activity recorded</p>
   343→          )}
   344→        </Card>
   345→      </motion.div>
   346→    </div>
   347→  );
   348→}

   353→function SyllabusTab({ subjectId }: { subjectId: string }) {
   354→  const allUnits = useStore((s) => s.syllabusUnits);
   355→  const units = useMemo(() => allUnits.filter((u) => u.subjectId === subjectId).sort((a, b) => a.order - b.order), [allUnits, subjectId]);
   356→  const addSyllabusUnit = useStore((s) => s.addSyllabusUnit);
   357→  const updateSyllabusUnit = useStore((s) => s.updateSyllabusUnit);
   358→  const deleteSyllabusUnit = useStore((s) => s.deleteSyllabusUnit);
   359→  const addSyllabusTopic = useStore((s) => s.addSyllabusTopic);
   360→  const updateSyllabusTopic = useStore((s) => s.updateSyllabusTopic);
   361→  const deleteSyllabusTopic = useStore((s) => s.deleteSyllabusTopic);
   362→  const toggleTopicComplete = useStore((s) => s.toggleTopicComplete);
   363→
   364→  const [newTopicMap, setNewTopicMap] = useState<Record<string, string>>({});
   365→  const [editingUnitId, setEditingUnitId] = useState<string | null>(null);
   366→  const [editingTopicId, setEditingTopicId] = useState<string | null>(null);
   367→  const [editValue, setEditValue] = useState('');
   368→  const [showAddUnit, setShowAddUnit] = useState(false);
   369→  const [newUnitName, setNewUnitName] = useState('');
   370→
   371→  const totalTopics = units.reduce((sum, u) => sum + u.topics.length, 0);
   372→  const completedTopics = units.reduce((sum, u) => sum + u.topics.filter((t) => t.completed).length, 0);
   373→  const overallPct = totalTopics > 0 ? Math.round((completedTopics / totalTopics) * 100) : 0;
   374→
   375→  const handleAddTopic = (unitId: string) => {
   376→    const name = newTopicMap[unitId]?.trim();
   377→    if (!name) return;
   378→    const unit = units.find((u) => u.id === unitId);
   379→    addSyllabusTopic({
   380→      unitId,
   381→      subjectId,
   382→      name,
   383→      completed: false,
   384→      order: unit ? unit.topics.length + 1 : 1,
   385→    });
   386→    setNewTopicMap((prev) => ({ ...prev, [unitId]: '' }));
   387→  };
   388→
   389→  const handleAddUnit = () => {
   390→    const name = newUnitName.trim();
   391→    if (!name) return;
   392→    addSyllabusUnit({
   393→      subjectId,
   394→      name,
   395→      order: units.length + 1,
   396→    });
   397→    setNewUnitName('');
   398→    setShowAddUnit(false);
   399→  };
   400→
   401→  const startEditUnit = (unit: SyllabusUnit) => {
   402→    setEditingUnitId(unit.id);
   403→    setEditValue(unit.name);
   404→  };
   405→
   406→  const saveEditUnit = () => {
   407→    if (editingUnitId && editValue.trim()) {
   408→      updateSyllabusUnit(editingUnitId, { name: editValue.trim() });
   409→    }
   410→    setEditingUnitId(null);
   411→    setEditValue('');
   412→  };
   413→
   414→  const startEditTopic = (topic: SyllabusTopic) => {
   415→    setEditingTopicId(topic.id);
   416→    setEditValue(topic.name);
   417→  };
   418→
   419→  const saveEditTopic = () => {
   420→    if (editingTopicId && editValue.trim()) {
   421→      updateSyllabusTopic(editingTopicId, { name: editValue.trim() });
   422→    }
   423→    setEditingTopicId(null);
   424→    setEditValue('');
   425→  };
   426→
   427→  return (
   428→    <div className="space-y-4">
   429→      {/* Overall progress */}
   430→      <Card className="p-4">
   431→        <div className="flex items-center justify-between mb-2">
   432→          <span className="text-sm font-medium">Overall Progress</span>
   433→          <span className="text-sm font-semibold">{completedTopics}/{totalTopics} topics ({overallPct}%)</span>
   434→        </div>
   435→        <Progress value={overallPct} className="h-2" />
   436→      </Card>
   437→
   438→      {/* Units */}
   439→      {units.length === 0 && !showAddUnit ? (
   440→        <div className="text-center py-12">
   441→          <BookOpen className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   442→          <h3 className="text-sm font-medium mb-1">No syllabus units yet</h3>
   443→          <p className="text-xs text-muted-foreground mb-4">Add units to start tracking your syllabus progress</p>
   444→          <Button size="sm" onClick={() => setShowAddUnit(true)}>
   445→            <Plus className="h-4 w-4 mr-1" /> Add Unit
   446→          </Button>
   447→        </div>
   448→      ) : (
   449→        <Accordion type="multiple" className="space-y-2">
   450→          {units.map((unit) => {
   451→            const completed = unit.topics.filter((t) => t.completed).length;
   452→            const total = unit.topics.length;
   453→            const pct = total > 0 ? Math.round((completed / total) * 100) : 0;
   454→
   455→            return (
   456→              <AccordionItem key={unit.id} value={unit.id} className="border rounded-lg px-0">
   457→                <AccordionTrigger className="px-4 py-3 hover:no-underline hover:bg-muted/50 rounded-t-lg">
   458→                  <div className="flex items-center gap-3 flex-1 text-left">
   459→                    {/* Inline edit unit name */}
   460→                    {editingUnitId === unit.id ? (
   461→                      <div className="flex items-center gap-2 flex-1">
   462→                        <Input
   463→                          value={editValue}
   464→                          onChange={(e) => setEditValue(e.target.value)}
   465→                          onKeyDown={(e) => {
   466→                            if (e.key === 'Enter') saveEditUnit();
   467→                            if (e.key === 'Escape') setEditingUnitId(null);
   468→                          }}
   469→                          className="h-8 text-sm"
   470→                          autoFocus
   471→                          onClick={(e) => e.stopPropagation()}
   472→                        />
   473→                        <Button size="icon" variant="ghost" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); saveEditUnit(); }}>
   474→                          <Check className="h-3.5 w-3.5" />
   475→                        </Button>
   476→                      </div>
   477→                    ) : (
   478→                      <>
   479→                        <span className="font-medium text-sm flex-1" onDoubleClick={(e) => { e.stopPropagation(); startEditUnit(unit); }}>{unit.name}</span>
   480→                        <span className="text-xs text-muted-foreground">{completed}/{total}</span>
   481→                        <div className="w-20">
   482→                          <Progress value={pct} className="h-1.5" />
   483→                        </div>
   484→                      </>
   485→                    )}
   486→                  </div>
   487→                </AccordionTrigger>
   488→                <AccordionContent className="px-4 pb-3">
   489→                  <div className="space-y-1">
   490→                    {unit.topics.map((topic) => (
   491→                      <div
   492→                        key={topic.id}
   493→                        className="flex items-center gap-2 group py-1.5 px-2 rounded-md hover:bg-muted/50"
   494→                      >

   938→function AttendanceTab({ subjectId }: { subjectId: string }) {
   939→  const allRecords = useStore((s) => s.attendance);
   940→  const records = useMemo(() => allRecords.filter((a) => a.subjectId === subjectId), [allRecords, subjectId]);
   941→  const profile = useStore((s) => s.profile);
   942→  const addAttendance = useStore((s) => s.addAttendance);
   943→  const deleteAttendance = useStore((s) => s.deleteAttendance);
   944→
   945→  const att = getSubjectAttendance({ attendance: records }, subjectId);
   946→  const today = new Date().toISOString().split('T')[0];
   947→
   948→  const todayRecord = records.find((r) => r.date === today);
   949→
   950→  const status = att.total === 0
   951→    ? 'NO DATA'
   952→    : att.percentage >= profile.attendanceThreshold
   953→      ? 'SAFE'
   954→      : att.percentage >= profile.attendanceThreshold - 10
   955→        ? 'WATCH'
   956→        : 'RISK';
   957→
   958→  const statusColor =
   959→    status === 'SAFE' ? 'text-[var(--delulu-success)]' :
   960→    status === 'WATCH' ? 'text-[var(--delulu-warning)]' :
   961→    status === 'RISK' ? 'text-[var(--delulu-danger)]' :
   962→    'text-muted-foreground';
   963→
   964→  // How many more classes can be missed
   965→  const canMiss = useMemo(() => {
   966→    if (att.total === 0) return 'N/A';
   967→    const threshold = profile.attendanceThreshold;
   968→    // (present) / (total + x) >= threshold/100
   969→    // x <= (present * 100 / threshold) - total
   970→    const maxMissable = Math.floor((att.present * 100) / threshold) - att.total;
   971→    return Math.max(0, maxMissable);
   972→  }, [att, profile.attendanceThreshold]);
