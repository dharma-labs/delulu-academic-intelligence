   123→   123→function signalBadge(signal: SignalStatus) {
   124→   124→  const map: Record<SignalStatus, { label: string; className: string }> = {
   125→   125→    healthy: { label: 'Healthy', className: 'bg-[var(--delulu-success)]/15 text-[var(--delulu-success)] border-[var(--delulu-success)]/20' },
   126→   126→    improving: { label: 'Improving', className: 'bg-[var(--delulu-info)]/15 text-[var(--delulu-info)] border-[var(--delulu-info)]/20' },
   127→   127→    attention: { label: 'Needs Attention', className: 'bg-[var(--delulu-warning)]/15 text-[var(--delulu-warning)] border-[var(--delulu-warning)]/20' },
   128→   128→    critical: { label: 'Critical', className: 'bg-[var(--delulu-danger)]/15 text-[var(--delulu-danger)] border-[var(--delulu-danger)]/20' },
   129→   129→    upcoming: { label: 'Upcoming', className: 'bg-[var(--delulu-purple)]/15 text-[var(--delulu-purple)] border-[var(--delulu-purple)]/20' },
   130→   130→    nodata: { label: 'No Data', className: 'bg-muted text-muted-foreground border-border' },
   131→   131→  };
   132→   132→  const s = map[signal];
   133→   133→  return <Badge variant="outline" className={s.className}>{s.label}</Badge>;
   134→   134→}
   135→   135→
   136→   136→function pctColor(pct: number): string {
   137→   137→  if (pct >= 75) return 'text-[var(--delulu-success)]';
   138→   138→  if (pct >= 50) return 'text-[var(--delulu-info)]';
   139→   139→  if (pct >= 30) return 'text-[var(--delulu-warning)]';
   140→   140→  return 'text-[var(--delulu-danger)]';
   141→   141→}

   143→   143→function formatDuration(seconds: number): string {
   144→   144→  const h = Math.floor(seconds / 3600);
   145→   145→  const m = Math.floor((seconds % 3600) / 60);
   146→   146→  if (h > 0) return `${h}h ${m}m`;
   147→   147→  return `${m}m`;
   148→   148→}

   158→   158→function OverviewTab({ subjectId }: { subjectId: string }) {

   160→   160→  const allSyllabusUnits = useStore((s) => s.syllabusUnits);

   163→   163→  const allStudySessions = useStore((s) => s.studySessions);
   164→   164→  const allExams = useStore((s) => s.exams);
   165→   165→  const profile = useStore((s) => s.profile);
   166→   166→  const subject = subjects.find((x) => x.id === subjectId);
   167→   167→  const syllabusUnits = useMemo(() => allSyllabusUnits.filter((u) => u.subjectId === subjectId), [allSyllabusUnits, subjectId]);
   168→   168→  const assessments = useMemo(() => allAssessments.filter((a) => a.subjectId === subjectId), [allAssessments, subjectId]);
   169→   169→  const attendance = useMemo(() => allAttendance.filter((a) => a.subjectId === subjectId), [allAttendance, subjectId]);
   170→   170→  const studySessions = useMemo(() => allStudySessions.filter((s) => s.subjectId === subjectId), [allStudySessions, subjectId]);
   171→   171→  const exams = useMemo(() => allExams.filter((e) => e.subjectId === subjectId), [allExams, subjectId]);
   172→   172→
   173→   173→  const progress = getSubjectProgress({ syllabusUnits }, subjectId);
   174→   174→  const att = getSubjectAttendance({ attendance }, subjectId);
   175→   175→  const marks = getSubjectMarks({ assessments }, subjectId);
   176→   176→  const grade = getSubjectGrade({ assessments }, subjectId);
   177→   177→
   178→   178→  // CA performance (average assessment %)
   179→   179→  const caPct = assessments.length > 0
   180→   180→    ? Math.round(assessments.reduce((sum, a) => sum + (a.maxMarks > 0 ? (a.obtainedMarks / a.maxMarks) * 100 : 0), 0) / assessments.length)
   181→   181→    : 0;
   182→   182→
   183→   183→  // Next upcoming exam
   184→   184→  const nextExam = exams
   185→   185→    .filter((e) => e.status === 'upcoming')
   186→   186→    .sort((a, b) => a.date.localeCompare(b.date))[0];
   187→   187→
   188→   188→  // Recommended action
   189→   189→  const recommendation = useMemo(() => {
   190→   190→    const recs: string[] = [];
   191→   191→    if (att.total > 0 && att.percentage < profile.attendanceThreshold) {
   192→   192→      recs.push(`Attendance is below ${profile.attendanceThreshold}% threshold. Focus on attending classes.`);
   193→   193→    }
   194→   194→    if (progress < 50 && progress > 0) {
   195→   195→      recs.push('Syllabus is less than halfway done. Prioritize completing more topics.');
   196→   196→    }
   197→   197→    if (progress === 0) {
   198→   198→      recs.push('No syllabus progress yet. Start by adding topics to your units.');
   199→   199→    }
   200→   200→    if (marks.percentage < 50 && marks.max > 0) {
   201→   201→      recs.push('Assessment scores need improvement. Review weak areas before the next test.');
   202→   202→    }
   203→   203→    if (nextExam) {
   204→   204→      const daysUntil = Math.ceil((parseISO(nextExam.date).getTime() - Date.now()) / 86400000);
   205→   205→      if (daysUntil <= 7) {
   206→   206→        recs.push(`${nextExam.name} is in ${daysUntil} day${daysUntil !== 1 ? 's' : ''}. Intensive revision recommended.`);
   207→   207→      }
   208→   208→    }
   209→   209→    if (recs.length === 0) {
   210→   210→      if (progress >= 80 && marks.percentage >= 70) {
   211→   211→        recs.push('Great progress! Keep up the consistency and start revising for exams.');
   212→   212→      } else {
   213→   213→        recs.push('Add more data (attendance, marks, study sessions) to get personalized recommendations.');
   214→   214→      }
   215→   215→    }
   216→   216→    return recs[0];
   217→   217→  }, [att, progress, marks, nextExam, profile.attendanceThreshold]);
   218→   218→
   219→   219→  // Recent activity (last 5 items combined from sessions and assessments)
   220→   220→  const recentActivity = useMemo(() => {
   221→   221→    const items: { type: string; label: string; date: string }[] = [];
   222→   222→    studySessions.forEach((s) =>
   223→   223→      items.push({ type: 'session', label: `Studied for ${formatDuration(s.duration)}`, date: s.date })
   224→   224→    );
   225→   225→    assessments.forEach((a) =>
   226→   226→      items.push({ type: 'assessment', label: `${a.name}: ${a.obtainedMarks}/${a.maxMarks}`, date: a.date })
   227→   227→    );
   228→   228→    return items.sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5);
   229→   229→  }, [studySessions, assessments]);
   230→   230→
   231→   231→  if (!subject) return null;
   232→   232→
   233→   233→  const metrics = [
   234→   234→    {
   235→   235→      label: 'Current Grade',
   236→   236→      value: marks.max > 0 ? grade : '--',
   237→   237→      sub: marks.max > 0 ? `${marks.percentage}% average` : 'No assessments yet',
   238→   238→      icon: GraduationCap,
   239→   239→      color: marks.max > 0 ? pctColor(marks.percentage) : 'text-muted-foreground',
   240→   240→    },
   241→   241→    {
   242→   242→      label: 'Attendance',
   243→   243→      value: att.total > 0 ? `${att.percentage}%` : '--',
   244→   244→      sub: att.total > 0 ? `${att.present}/${att.total} classes` : 'No records yet',
   245→   245→      icon: ClipboardCheck,
   246→   246→      color: att.total > 0
   247→   247→        ? att.percentage >= profile.attendanceThreshold
   248→   248→          ? 'text-[var(--delulu-success)]'
   249→   249→          : 'text-[var(--delulu-danger)]'
   250→   250→        : 'text-muted-foreground',
   251→   251→    },
   252→   252→    {
   253→   253→      label: 'Syllabus',
   254→   254→      value: `${progress}%`,
   255→   255→      sub: progress > 0 ? 'topics completed' : 'No topics yet',
   256→   256→      icon: BookOpen,
   257→   257→      color: progress >= 75 ? 'text-[var(--delulu-success)]' : progress >= 40 ? 'text-[var(--delulu-info)]' : 'text-muted-foreground',
   258→   258→    },
   259→   259→    {
   260→   260→      label: 'CA Performance',
   261→   261→      value: caPct > 0 ? `${caPct}%` : '--',
   262→   262→      sub: caPct > 0 ? `across ${assessments.length} assessment${assessments.length !== 1 ? 's' : ''}` : 'No assessments yet',
   263→   263→      icon: BarChart3,
   264→   264→      color: caPct > 0 ? pctColor(caPct) : 'text-muted-foreground',
   265→   265→    },
   266→   266→  ];
   267→   267→
   268→   268→  return (
   269→   269→    <div className="space-y-6">
   270→   270→      {/* 4 metric cards */}
   271→   271→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   272→   272→        {metrics.map((m) => (
   273→   273→          <motion.div key={m.label} variants={fadeUp} initial="hidden" animate="show">
   274→   274→            <Card className="p-4">
   275→   275→              <div className="flex items-center gap-2 mb-2">
   276→   276→                <m.icon className="h-4 w-4 text-muted-foreground" />
   277→   277→                <span className="text-xs text-muted-foreground font-medium">{m.label}</span>
   278→   278→              </div>
   279→   279→              <p className={`text-2xl font-semibold ${m.color}`}>{m.value}</p>
   280→   280→              <p className="text-xs text-muted-foreground mt-1">{m.sub}</p>
   281→   281→            </Card>
   282→   282→          </motion.div>
   283→   283→        ))}
   284→   284→      </div>
   285→   285→
   286→   286→      {/* Next Assessment + Recommendation */}
   287→   287→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   288→   288→        <motion.div variants={fadeUp} initial="hidden" animate="show">

   353→   353→function SyllabusTab({ subjectId }: { subjectId: string }) {
   354→   354→  const allUnits = useStore((s) => s.syllabusUnits);
   355→   355→  const units = useMemo(() => allUnits.filter((u) => u.subjectId === subjectId).sort((a, b) => a.order - b.order), [allUnits, subjectId]);
   356→   356→  const addSyllabusUnit = useStore((s) => s.addSyllabusUnit);
   357→   357→  const updateSyllabusUnit = useStore((s) => s.updateSyllabusUnit);
   358→   358→  const deleteSyllabusUnit = useStore((s) => s.deleteSyllabusUnit);
   359→   359→  const addSyllabusTopic = useStore((s) => s.addSyllabusTopic);
   360→   360→  const updateSyllabusTopic = useStore((s) => s.updateSyllabusTopic);
   361→   361→  const deleteSyllabusTopic = useStore((s) => s.deleteSyllabusTopic);
   362→   362→  const toggleTopicComplete = useStore((s) => s.toggleTopicComplete);
   363→   363→
   364→   364→  const [newTopicMap, setNewTopicMap] = useState<Record<string, string>>({});
   365→   365→  const [editingUnitId, setEditingUnitId] = useState<string | null>(null);
   366→   366→  const [editingTopicId, setEditingTopicId] = useState<string | null>(null);
   367→   367→  const [editValue, setEditValue] = useState('');
   368→   368→  const [showAddUnit, setShowAddUnit] = useState(false);
   369→   369→  const [newUnitName, setNewUnitName] = useState('');
   370→   370→
   371→   371→  const totalTopics = units.reduce((sum, u) => sum + u.topics.length, 0);
   372→   372→  const completedTopics = units.reduce((sum, u) => sum + u.topics.filter((t) => t.completed).length, 0);
   373→   373→  const overallPct = totalTopics > 0 ? Math.round((completedTopics / totalTopics) * 100) : 0;
   374→   374→
   375→   375→  const handleAddTopic = (unitId: string) => {
   376→   376→    const name = newTopicMap[unitId]?.trim();
   377→   377→    if (!name) return;
   378→   378→    const unit = units.find((u) => u.id === unitId);
   379→   379→    addSyllabusTopic({
   380→   380→      unitId,
   381→   381→      subjectId,
   382→   382→      name,
   383→   383→      completed: false,
   384→   384→      order: unit ? unit.topics.length + 1 : 1,
   385→   385→    });
   386→   386→    setNewTopicMap((prev) => ({ ...prev, [unitId]: '' }));
   387→   387→  };
   388→   388→
   389→   389→  const handleAddUnit = () => {
   390→   390→    const name = newUnitName.trim();
   391→   391→    if (!name) return;
   392→   392→    addSyllabusUnit({
   393→   393→      subjectId,
   394→   394→      name,
   395→   395→      order: units.length + 1,
   396→   396→    });
   397→   397→    setNewUnitName('');
   398→   398→    setShowAddUnit(false);
   399→   399→  };
   400→   400→
   401→   401→  const startEditUnit = (unit: SyllabusUnit) => {
   402→   402→    setEditingUnitId(unit.id);
   403→   403→    setEditValue(unit.name);
   404→   404→  };
   405→   405→
   406→   406→  const saveEditUnit = () => {
   407→   407→    if (editingUnitId && editValue.trim()) {
   408→   408→      updateSyllabusUnit(editingUnitId, { name: editValue.trim() });
   409→   409→    }
   410→   410→    setEditingUnitId(null);
   411→   411→    setEditValue('');
   412→   412→  };
   413→   413→
   414→   414→  const startEditTopic = (topic: SyllabusTopic) => {
   415→   415→    setEditingTopicId(topic.id);
   416→   416→    setEditValue(topic.name);
   417→   417→  };
   418→   418→
   419→   419→  const saveEditTopic = () => {
   420→   420→    if (editingTopicId && editValue.trim()) {
   421→   421→      updateSyllabusTopic(editingTopicId, { name: editValue.trim() });
