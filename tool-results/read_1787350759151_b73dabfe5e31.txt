    59→    59→function timeToMinutes(time: string): number {
    60→    60→  const [h, m] = time.split(':').map(Number);
    61→    61→  return h * 60 + m;
    62→    62→}
    63→    63→
    64→    64→function minutesToTime(minutes: number): string {
    65→    65→  const h = Math.floor(minutes / 60);
    66→    66→  const m = minutes % 60;
    67→    67→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
    68→    68→}
    69→    69→
    70→    70→function formatTime(time: string): string {
    71→    71→  const [h, m] = time.split(':').map(Number);
    72→    72→  const ampm = h >= 12 ? 'PM' : 'AM';
    73→    73→  const h12 = h === 0 ? 12 : h > 12 ? h - 12 : h;
    74→    74→  return `${h12}:${String(m).padStart(2, '0')} ${ampm}`;
    75→    75→}
    76→    76→
    77→    77→function getSlotTop(time: string): number {
    78→    78→  const minutes = timeToMinutes(time);
    79→    79→  const offset = minutes - START_HOUR * 60;
    80→    80→  return (offset / 30) * SLOT_HEIGHT;
    81→    81→}
    82→    82→
    83→    83→function getSlotHeight(start: string, end: string): number {
    84→    84→  const diff = timeToMinutes(end) - timeToMinutes(start);
    85→    85→  return (diff / 30) * SLOT_HEIGHT;
    86→    86→}

    88→    88→function getNowIndicatorTop(): number {
    89→    89→  const now = new Date();

    91→    91→  const offset = minutes - START_HOUR * 60;
    92→    92→  return (offset / 30) * SLOT_HEIGHT;
    93→    93→}

   114→   114→function SlotDialog({
   115→   115→  open,
   116→   116→  onOpenChange,
   117→   117→  slot,
   118→   118→}: {
   119→   119→  open: boolean;
   120→   120→  onOpenChange: (v: boolean) => void;
   121→   121→  slot: TimetableSlot | null;
   122→   122→}) {
   123→   123→  const addTimetableSlot = useStore((s) => s.addTimetableSlot);
   124→   124→  const updateTimetableSlot = useStore((s) => s.updateTimetableSlot);
   125→   125→  const subjects = useStore((s) => s.subjects);
   126→   126→  const timetableSlots = useStore((s) => s.timetableSlots);
   127→   127→  const activeSubjects = subjects.filter((s) => !s.archived);
   128→   128→
   129→   129→  const isEditing = !!slot;
   130→   130→  const initialSubject = slot?.subjectId ?? (activeSubjects[0]?.id ?? '');
   131→   131→  const initialDay = slot ? String(slot.day) : String(new Date().getDay() === 0 ? 6 : new Date().getDay() - 1);
   132→   132→  const initialStart = slot?.startTime ?? '09:00';
   133→   133→  const initialEnd = slot?.endTime ?? '10:00';
   134→   134→  const initialRoom = slot?.room ?? '';
   135→   135→  const initialType = slot?.type ?? 'lecture';
   136→   136→
   137→   137→  const [subjectId, setSubjectId] = useState(initialSubject);
   138→   138→  const [day, setDay] = useState(initialDay);
   139→   139→  const [startTime, setStartTime] = useState(initialStart);
   140→   140→  const [endTime, setEndTime] = useState(initialEnd);
   141→   141→  const [room, setRoom] = useState(initialRoom);
   142→   142→  const [type, setType] = useState<TimetableSlot['type']>(initialType);
   143→   143→  const [error, setError] = useState('');
   144→   144→
   145→   145→  const handleSubmit = () => {
   146→   146→    setError('');
   147→   147→
   148→   148→    if (!subjectId) {
   149→   149→      setError('Please select a subject.');
   150→   150→      return;
   151→   151→    }
   152→   152→
   153→   153→    const startMin = timeToMinutes(startTime);
   154→   154→    const endMin = timeToMinutes(endTime);
   155→   155→
   156→   156→    if (endMin <= startMin) {
   157→   157→      setError('End time must be after start time.');
   158→   158→      return;
   159→   159→    }
   160→   160→
   161→   161→    if (startMin < START_HOUR * 60 || endMin > END_HOUR * 60) {
   162→   162→      setError(`Time must be between ${formatTime(`${START_HOUR}:00`)} and ${formatTime(`${END_HOUR}:00`)}.`);
   163→   163→      return;
   164→   164→    }
   165→   165→
   166→   166→    // Check for overlap with existing slots on the same day
   167→   167→    const dayNum = Number(day) as TimetableSlot['day'];
   168→   168→    const overlapping = timetableSlots.find((s) => {
   169→   169→      if (s.id === slot?.id) return false;
   170→   170→      if (s.day !== dayNum) return false;
   171→   171→      const sStart = timeToMinutes(s.startTime);
   172→   172→      const sEnd = timeToMinutes(s.endTime);
   173→   173→      return startMin < sEnd && endMin > sStart;
   174→   174→    });
   175→   175→
   176→   176→    if (overlapping) {
   177→   177→      const subject = subjects.find((s) => s.id === overlapping.subjectId);
   178→   178→      setError(`Overlaps with ${subject?.name ?? 'another class'} (${formatTime(overlapping.startTime)} - ${formatTime(overlapping.endTime)}).`);
   179→   179→      return;
   180→   180→    }
   181→   181→
   182→   182→    const data = {
   183→   183→      subjectId,
   184→   184→      day: dayNum,
   185→   185→      startTime,
   186→   186→      endTime,
   187→   187→      room: room.trim() || undefined,
   188→   188→      type,
   189→   189→    };
   190→   190→
   191→   191→    if (isEditing && slot) {
   192→   192→      updateTimetableSlot(slot.id, data);
   193→   193→    } else {
   194→   194→      addTimetableSlot(data);
   195→   195→    }
   196→   196→    onOpenChange(false);
   197→   197→  };
   198→   198→
   199→   199→  return (
   200→   200→    <Dialog open={open} onOpenChange={onOpenChange}>
   201→   201→      <DialogContent className='sm:max-w-md'>
   202→   202→        <DialogHeader>
   203→   203→          <DialogTitle>{isEditing ? 'Edit Slot' : 'Add Timetable Slot'}</DialogTitle>
   204→   204→        </DialogHeader>
   205→   205→
   206→   206→        <div className='space-y-4 py-2'>
   207→   207→          {error && (
   208→   208→            <div className='text-sm text-destructive bg-destructive/10 rounded-md px-3 py-2'>
   209→   209→              {error}
   210→   210→            </div>
   211→   211→          )}
   212→   212→
   213→   213→          <div className='space-y-2'>
   214→   214→            <Label>Subject *</Label>
   215→   215→            <Select value={subjectId} onValueChange={setSubjectId}>
   216→   216→              <SelectTrigger><SelectValue placeholder='Select subject' /></SelectTrigger>
   217→   217→              <SelectContent>
   218→   218→                {activeSubjects.map((s) => (
   219→   219→                  <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   220→   220→                ))}
   221→   221→              </SelectContent>
   222→   222→            </Select>
   223→   223→          </div>
   224→   224→
   225→   225→          <div className='grid grid-cols-2 gap-3'>
   226→   226→            <div className='space-y-2'>
   227→   227→              <Label>Day</Label>
   228→   228→              <Select value={day} onValueChange={setDay}>
   229→   229→                <SelectTrigger><SelectValue /></SelectTrigger>
   230→   230→                <SelectContent>
   231→   231→                  {DAYS.map((d, i) => (
   232→   232→                    <SelectItem key={i} value={String(i)}>{DAY_FULL[i]}</SelectItem>
   233→   233→                  ))}
   234→   234→                </SelectContent>
   235→   235→              </Select>
   236→   236→            </div>
   237→   237→
   238→   238→            <div className='space-y-2'>
   239→   239→              <Label>Type</Label>
   240→   240→              <Select value={type} onValueChange={(v) => setType(v as TimetableSlot['type'])}>
   241→   241→                <SelectTrigger><SelectValue /></SelectTrigger>
   242→   242→                <SelectContent>
   243→   243→                  <SelectItem value='lecture'>Lecture</SelectItem>
   244→   244→                  <SelectItem value='lab'>Lab</SelectItem>
   245→   245→                  <SelectItem value='tutorial'>Tutorial</SelectItem>
   246→   246→                </SelectContent>
   247→   247→              </Select>
   248→   248→            </div>
   249→   249→          </div>

   278→   278→function WeeklyGrid({
   279→   279→  weekStart,
   280→   280→  onSlotClick,
   281→   281→}: {
   282→   282→  weekStart: Date;
   283→   283→  onSlotClick: (slot: TimetableSlot) => void;
   284→   284→}) {
   285→   285→  const timetableSlots = useStore((s) => s.timetableSlots);
   286→   286→  const subjects = useStore((s) => s.subjects);
   287→   287→  const today = new Date();
   288→   288→  const gridRef = useRef<HTMLDivElement>(null);
   289→   289→  const [nowTop, setNowTop] = useState(getNowIndicatorTop);
   290→   290→
   291→   291→  // Map day number (0=Sun..6=Sat) to column index (0=Mon..6=Sun)
   292→   292→  const getColIndex = (day: number) => (day === 0 ? 6 : day - 1);
   293→   293→  const todayCol = getColIndex(today.getDay());
   294→   294→
   295→   295→  // Update now indicator every minute
   296→   296→  useEffect(() => {
   297→   297→    const interval = setInterval(() => {
   298→   298→      setNowTop(getNowIndicatorTop());
   299→   299→    }, 60000);
   300→   300→    return () => clearInterval(interval);
   301→   301→  }, []);
   302→   302→
   303→   303→  // Scroll to current time on mount
   304→   304→  useEffect(() => {
   305→   305→    if (gridRef.current) {
   306→   306→      const top = getNowIndicatorTop() - 200;
   307→   307→      gridRef.current.scrollTop = Math.max(0, top);
   308→   308→    }
   309→   309→  }, []);
   310→   310→
   311→   311→  const totalHeight = TOTAL_HOURS * 2 * SLOT_HEIGHT;
   312→   312→  const isTodayInView = todayCol >= 0 && todayCol <= 6;
   313→   313→  const nowInRange = nowTop >= 0 && nowTop <= totalHeight;
   314→   314→
   315→   315→  return (
   316→   316→    <div ref={gridRef} className='overflow-y-auto scrollbar-thin max-h-[calc(100vh-280px)] rounded-lg border border-border bg-card'>
   317→   317→      <div className='relative min-w-[700px]'>
   318→   318→        {/* Header row */}

   323→   323→              <div
   324→   324→                key={d}
   325→   325→                className={`text-center text-xs font-medium py-2 border-l border-border ${
   326→   326→                  i === todayCol ? 'bg-primary/5 text-primary' : 'text-muted-foreground'
   327→   327→                }`}
   328→   328→              >
   329→   329→                {d}
   330→   330→              </div>
   331→   331→            ))}
   332→   332→          </div>
   333→   333→        </div>
   334→   334→
   335→   335→        {/* Time grid */}
   336→   336→        <div className='relative' style={{ height: totalHeight }}>
   337→   337→          {/* Grid lines and labels */}
   338→   338→          {TIME_LABELS.map((label, i) => {
   339→   339→            const top = i * SLOT_HEIGHT;
   340→   340→            return (
   341→   341→              <div key={label}>
   342→   342→                {/* Label */}
   343→   343→                <div
   344→   344→                  className='absolute left-0 w-[56px] text-right pr-2 text-[10px] text-muted-foreground -translate-y-1/2'
   345→   345→                  style={{ top }}
   346→   346→                >
   347→   347→                  {label}
   348→   348→                </div>
   349→   349→                {/* Horizontal line */}
   350→   350→                <div
   351→   351→                  className='absolute left-[60px] right-0 border-t border-border/50'
   352→   352→                  style={{ top }}
   353→   353→                />
   354→   354→              </div>
   355→   355→            );
   356→   356→          })}

   358→   358→          {/* Vertical day lines */}

   425→   425→                  <div
   426→   426→                    onClick={(e) => {
   427→   427→                      e.stopPropagation();

   437→   437→
   438→   438→          {/* Now indicator */}

   440→   440→            <div
   441→   441→              className='absolute left-[60px] right-0 z-30 pointer-events-none'

   450→   450→        </div>
   451→   451→      </div>
   452→   452→    </div>
   453→   453→  );

   457→   457→function MobileTodayView({ onSlotClick }: { onSlotClick: (slot: TimetableSlot) => void }) {
   458→   458→  const timetableSlots = useStore((s) => s.timetableSlots);
   459→   459→  const subjects = useStore((s) => s.subjects);
   460→   460→

   464→   464→
   465→   465→  const todaySlots = timetableSlots
   466→   466→    .filter((s) => s.day === storeDay)
   467→   467→    .sort((a, b) => a.startTime.localeCompare(b.startTime));

   469→   469→  const nowMinutes = today.getHours() * 60 + today.getMinutes();

   588→   588→function NextClassCard() {
   589→   589→  const timetableSlots = useStore((s) => s.timetableSlots);
   590→   590→  const subjects = useStore((s) => s.subjects);
   591→   591→
   592→   592→  const today = new Date();
   593→   593→  const jsDay = today.getDay();
   594→   594→  const storeDay = (jsDay === 0 ? 6 : jsDay - 1) as TimetableSlot['day'];
   595→   595→  const nowMinutes = today.getHours() * 60 + today.getMinutes();
   596→   596→
   597→   597→  const todaySlots = timetableSlots
   598→   598→    .filter((s) => s.day === storeDay)
   599→   599→    .sort((a, b) => a.startTime.localeCompare(b.startTime));
   600→   600→
   601→   601→  const nextSlot = todaySlots.find((s) => timeToMinutes(s.startTime) > nowMinutes);
   602→   602→  const subject = nextSlot ? subjects.find((s) => s.id === nextSlot.subjectId) : null;
   603→   603→
   604→   604→  const [countdown, setCountdown] = useState('');
   605→   605→
   606→   606→  useEffect(() => {
   607→   607→    if (!nextSlot) return;
   608→   608→
   609→   609→    const update = () => {
   610→   610→      const now = new Date();
   611→   611→      const nowMin = now.getHours() * 60 + now.getMinutes();
   612→   612→      const startMin = timeToMinutes(nextSlot.startTime);
   613→   613→      const diff = startMin - nowMin;
   614→   614→      if (diff <= 0) {
   615→   615→        setCountdown('Starting now');
   616→   616→      } else if (diff < 60) {
   617→   617→        setCountdown(`Starts in ${diff} min`);
   618→   618→      } else {
   619→   619→        const h = Math.floor(diff / 60);
   620→   620→        const m = diff % 60;
   621→   621→        setCountdown(`Starts in ${h}h ${m}m`);
   622→   622→      }
   623→   623→    };
   624→   624→
   625→   625→    update();
   626→   626→    const interval = setInterval(update, 30000);
   627→   627→    return () => clearInterval(interval);
   628→   628→  }, [nextSlot]);
   629→   629→
   630→   630→  if (!nextSlot || !subject) return null;
   631→   631→
   632→   632→  return (
   633→   633→    <motion.div variants={fadeUp} initial='hidden' animate='show'>
   634→   634→      <Card className='border-primary/20 bg-gradient-to-r from-primary/5 to-transparent'>
   635→   635→        <CardContent className='p-4 flex items-center gap-4'>
   636→   636→          <div
   637→   637→            className='h-12 w-12 rounded-xl flex items-center justify-center shrink-0'
   638→   638→            style={{ backgroundColor: subject.color + '18' }}
   639→   639→          >
   640→   640→            <Timer className='h-6 w-6' style={{ color: subject.color }} />
   641→   641→          </div>
   642→   642→          <div className='flex-1 min-w-0'>
   643→   643→            <p className='text-xs text-muted-foreground font-medium uppercase tracking-wide'>
   644→   644→              Next Class
   645→   645→            </p>
   646→   646→            <p className='font-semibold text-sm mt-0.5 truncate'>{subject.name}</p>
   647→   647→            <div className='flex items-center gap-2 text-xs text-muted-foreground mt-1'>
   648→   648→              <Clock className='h-3 w-3' />
   649→   649→              {formatTime(nextSlot.startTime)} - {formatTime(nextSlot.endTime)}
   650→   650→              {nextSlot.room && (
   651→   651→                <>
   652→   652→                  <span className='text-border'>|</span>
   653→   653→                  <MapPin className='h-3 w-3' />
   654→   654→                  {nextSlot.room}
   655→   655→                </>
   656→   656→              )}
   657→   657→            </div>
   658→   658→          </div>
   659→   659→          <div className='text-right shrink-0'>
   660→   660→            <p className='text-sm font-semibold text-primary'>{countdown}</p>
   661→   661→          </div>
   662→   662→        </CardContent>
   663→   663→      </Card>
   664→   664→    </motion.div>
   665→   665→  );
   666→   666→}
