    75→function examTypeConfig(type: Exam['type']) {
    76→  switch (type) {
    77→    case 'midsem':
    78→      return { label: 'Mid-Sem', className: 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-400 border-violet-200 dark:border-violet-900' };
    79→    case 'endsem':
    80→      return { label: 'End-Sem', className: 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-400 border-red-200 dark:border-red-900' };
    81→    case 'quiz':
    82→      return { label: 'Quiz', className: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-400 border-amber-200 dark:border-amber-900' };
    83→    case 'practical':
    84→      return { label: 'Practical', className: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-400 border-teal-200 dark:border-teal-900' };
    85→    case 'other':
    86→      return { label: 'Other', className: 'bg-gray-100 text-gray-700 dark:bg-gray-950 dark:text-gray-400 border-gray-200 dark:border-gray-900' };
    87→  }

    90→function difficultyConfig(difficulty?: PYQ['difficulty']) {
    91→  switch (difficulty) {
    92→    case 'easy':
    93→      return { label: 'Easy', className: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400' };
    94→    case 'medium':

    97→      return { label: 'Hard', className: 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-400' };

    99→      return null;
   100→  }

   279→// -- Add PYQ Dialog ---------------------------------------------------
   280→function AddPYQDialog({ examId, subjectId, open, onOpenChange }: {
   281→  examId: string;
   282→  subjectId: string;
   283→  open: boolean;
   284→  onOpenChange: (v: boolean) => void;
   285→}) {
   286→  const addPYQ = useStore((s) => s.addPYQ);
   287→  const syllabusUnits = useStore((s) => s.syllabusUnits);
   288→
   289→  const allTopics = useMemo(
   290→    () => syllabusUnits.filter((u) => u.subjectId === subjectId).flatMap((u) => u.topics),
   291→    [syllabusUnits, subjectId]
   292→  );
   293→
   294→  const [year, setYear] = useState('');
   295→  const [question, setQuestion] = useState('');
   296→  const [answer, setAnswer] = useState('');
   297→  const [topicId, setTopicId] = useState('none');
   298→  const [difficulty, setDifficulty] = useState<PYQ['difficulty']>('medium');
   299→  const [attempted, setAttempted] = useState(false);
   300→  const [correct, setCorrect] = useState(false);
   301→
   302→  const handleSubmit = () => {
   303→    if (!year.trim() || !question.trim()) return;
   304→
   305→    addPYQ({
   306→      examId,
   307→      subjectId,
   308→      year: year.trim(),
   309→      question: question.trim(),
   310→      answer: answer.trim() || undefined,
   311→      topicId: topicId === 'none' ? undefined : topicId,
   312→      difficulty,
   313→      attempted,
   314→      correct: attempted ? correct : undefined,
   315→    });
   316→
   317→    onOpenChange(false);
   318→    setYear('');
   319→    setQuestion('');
   320→    setAnswer('');
   321→    setTopicId('none');
   322→    setDifficulty('medium');
   323→    setAttempted(false);
   324→    setCorrect(false);
   325→  };

   327→  return (
   328→    <Dialog open={open} onOpenChange={onOpenChange}>
   329→      <DialogContent className='sm:max-w-lg'>

   342→                onChange={(e) => setYear(e.target.value)}
   343→                autoFocus
   344→              />
   345→            </div>

   357→          </div>

   359→          <div className='space-y-2'>

   431→// -- Exam Detail (expanded) -------------------------------------------
   432→function ExamDetail({
   433→  exam,
   434→  onCollapse,
   435→}: {
   436→  exam: Exam;
   437→  onCollapse: () => void;
   438→}) {
   439→  const subjects = useStore((s) => s.subjects);
   440→  const pyqs = useStore((s) => s.pyqs);
   441→  const updateExam = useStore((s) => s.updateExam);
   442→  const deletePYQ = useStore((s) => s.deletePYQ);
   443→  const deleteExam = useStore((s) => s.deleteExam);
   444→
   445→  const subject = subjects.find((s) => s.id === exam.subjectId);
   446→  const examPyqs = pyqs.filter((p) => p.examId === exam.id);
   447→  const typeConf = examTypeConfig(exam.type);
   448→
   449→  const [prepNotes, setPrepNotes] = useState(exam.preparationNotes ?? '');
   450→  const [pyqDialogOpen, setPyqDialogOpen] = useState(false);
   451→  const [deleteId, setDeleteId] = useState<string | null>(null);
   452→
   453→  const handleSaveNotes = () => {
   454→    updateExam(exam.id, { preparationNotes: prepNotes });
   455→  };
   456→
   457→  const handleDeleteExam = () => {
   458→    deleteExam(exam.id);
   459→    onCollapse();
   460→  };
   461→
   462→  const percentage =
   463→    exam.status === 'completed' && exam.obtainedMarks !== undefined && exam.totalMarks > 0
   464→      ? Math.round((exam.obtainedMarks / exam.totalMarks) * 100)
   465→      : null;
   466→  const grade = percentage !== null ? GRADE_FROM_PERCENTAGE(percentage) : null;
   467→
   468→  return (
   469→    <motion.div
   470→      initial={{ opacity: 0, height: 0 }}
   471→      animate={{ opacity: 1, height: 'auto' }}
   472→      exit={{ opacity: 0, height: 0 }}
   473→      className='overflow-hidden'
   474→    >
   475→      <Card className='border-primary/20'>
   476→        <CardHeader className='pb-3'>
   477→          <div className='flex items-start justify-between gap-3'>
   478→            <div className='space-y-1'>
   479→              <CardTitle className='text-lg'>{exam.name}</CardTitle>
   480→              <div className='flex items-center gap-2 flex-wrap'>
   481→                {subject && (
   482→                  <Badge
   483→                    variant='outline'
   484→                    className='text-xs'
   485→                    style={{
   486→                      borderColor: subject.color + '60',
   487→                      color: subject.color,
   488→                    }}
   489→                  >
   490→                    {subject.code}
   491→                  </Badge>
   492→                )}
   493→                <Badge variant='outline' className={`text-xs border ${typeConf.className}`}>
   494→                  {typeConf.label}
   495→                </Badge>
   496→                <Badge variant='outline' className='text-xs'>
   497→                  {exam.totalMarks} marks
   498→                </Badge>
   499→                <span className='text-xs text-muted-foreground flex items-center gap-1'>
   500→                  <CalendarDays className='h-3 w-3' />
   501→                  {format(parseISO(exam.date), 'MMM d, yyyy')}
   502→                </span>
   503→              </div>
   504→            </div>
   505→            <div className='flex items-center gap-2 shrink-0'>
   506→              <Button variant='ghost' size='sm' onClick={onCollapse}>
   507→                <ChevronDown className='h-4 w-4' />
   508→              </Button>
   509→              <Button
   510→                variant='ghost'
   511→                size='icon'
   512→                className='h-8 w-8 text-destructive hover:text-destructive'
   513→                onClick={() => setDeleteId('exam')}
   514→              >
   515→                <Trash2 className='h-3.5 w-3.5' />
   516→              </Button>
   517→            </div>
   518→          </div>
   519→
   520→          {/* Completed stats row */}
   521→          {exam.status === 'completed' && percentage !== null && (
   522→            <div className='flex items-center gap-4 mt-3'>
   523→              <div className='text-sm'>
   524→                <span className='font-semibold'>{exam.obtainedMarks}</span>
   525→                <span className='text-muted-foreground'> / {exam.totalMarks}</span>
   526→              </div>
   527→              <Badge variant='outline' className='text-xs'>{percentage}%</Badge>
   528→              {grade && (
   529→                <Badge className='text-xs bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400'>
   530→                  {grade}
   531→                </Badge>
   532→              )}
   533→            </div>
   534→          )}
   535→
   536→          {/* Upcoming days remaining */}
   537→          {exam.status === 'upcoming' && (
   538→            <div className='mt-2'>
   539→              <Badge className='text-xs bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-400'>
   540→                UPCOMING
   541→              </Badge>
   542→            </div>
   543→          )}
   544→        </CardHeader>
   545→
   546→        <Separator />
   547→
   548→        <CardContent className='pt-4 space-y-6'>
   549→          {/* PYQ Section */}
   550→          <div>
   551→            <div className='flex items-center justify-between mb-3'>
   552→              <h3 className='text-sm font-semibold flex items-center gap-2'>
   553→                <FileText className='h-4 w-4' />
   554→                Previous Year Questions ({examPyqs.length})
   555→              </h3>
   556→              <Button size='sm' variant='outline' className='gap-1.5 h-8' onClick={() => setPyqDialogOpen(true)}>
   557→                <Plus className='h-3.5 w-3.5' />
   558→                Add PYQ
   559→              </Button>
   560→            </div>
   561→
   562→            {examPyqs.length === 0 ? (
   563→              <p className='text-sm text-muted-foreground py-4 text-center'>
   564→                No previous year questions added yet.
   565→              </p>
   566→            ) : (
   567→              <div className='space-y-2 max-h-96 overflow-y-auto scrollbar-thin'>
   568→                {examPyqs.map((pyq) => {
   569→                  const diffConf = difficultyConfig(pyq.difficulty);
   570→                  return (
   571→                    <div
   572→                      key={pyq.id}
   573→                      className='rounded-lg border border-border p-3 hover:border-primary/20 transition-colors group'
   574→                    >
