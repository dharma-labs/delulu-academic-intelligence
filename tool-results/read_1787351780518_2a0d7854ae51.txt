    76→function examTypeConfig(type: Exam['type']) {
    77→  switch (type) {
    78→    case 'midsem':
    79→      return { label: 'Mid-Sem', signalClass: 'bg-purple-50 text-purple-700 dark:bg-purple-950/50 dark:text-purple-400' };
    80→    case 'endsem':
    81→      return { label: 'End-Sem', signalClass: 'signal-critical' };
    82→    case 'quiz':
    83→      return { label: 'Quiz', signalClass: 'signal-attention' };
    84→    case 'practical':
    85→      return { label: 'Practical', signalClass: 'signal-healthy' };
    86→    case 'other':
    87→      return { label: 'Other', signalClass: 'bg-secondary text-muted-foreground' };
    88→  }
    89→}
    90→
    91→function difficultyConfig(difficulty?: PYQ['difficulty']) {
    92→  switch (difficulty) {
    93→    case 'easy':
    94→      return { label: 'Easy', className: 'signal-healthy' };
    95→    case 'medium':
    96→      return { label: 'Medium', className: 'signal-attention' };
    97→    case 'hard':
    98→      return { label: 'Hard', className: 'signal-critical' };
    99→    default:
   100→      return null;
   101→  }

   253→// -- Add PYQ Dialog ---------------------------------------------------
   254→function AddPYQDialog({ examId, subjectId, open, onOpenChange }: {

   256→  subjectId: string;
   257→  open: boolean;
   258→  onOpenChange: (v: boolean) => void;
   259→}) {
   260→  const addPYQ = useStore((s) => s.addPYQ);
   261→  const syllabusUnits = useStore((s) => s.syllabusUnits);
   262→
   263→  const allTopics = useMemo(
   264→    () => syllabusUnits.filter((u) => u.subjectId === subjectId).flatMap((u) => u.topics),
   265→    [syllabusUnits, subjectId]
   266→  );
   267→
   268→  const [year, setYear] = useState('');
   269→  const [question, setQuestion] = useState('');
   270→  const [answer, setAnswer] = useState('');

   273→  const [attempted, setAttempted] = useState(false);
   274→  const [correct, setCorrect] = useState(false);
   275→
   276→  const handleSubmit = () => {
   277→    if (!year.trim() || !question.trim()) return;
   278→
   279→    addPYQ({
   280→      examId,
   281→      subjectId,
   282→      year: year.trim(),
   283→      question: question.trim(),
   284→      answer: answer.trim() || undefined,
   285→      topicId: topicId === 'none' ? undefined : topicId,
   286→      difficulty,
   287→      attempted,
   288→      correct: attempted ? correct : undefined,
   289→    });
   290→
   291→    onOpenChange(false);
   292→    setYear('');
   293→    setQuestion('');
   294→    setAnswer('');
   295→    setTopicId('none');
   296→    setDifficulty('medium');
   297→    setAttempted(false);
   298→    setCorrect(false);
   299→  };

   301→  return (
   302→    <Dialog open={open} onOpenChange={onOpenChange}>
   303→      <DialogContent className='sm:max-w-lg'>

   405→// -- Exam Detail (expanded) -------------------------------------------
   406→function ExamDetail({
   407→  exam,
   408→  onCollapse,
   409→}: {
   410→  exam: Exam;
   411→  onCollapse: () => void;
   412→}) {
   413→  const subjects = useStore((s) => s.subjects);
   414→  const pyqs = useStore((s) => s.pyqs);
   415→  const updateExam = useStore((s) => s.updateExam);
   416→  const deletePYQ = useStore((s) => s.deletePYQ);
   417→  const deleteExam = useStore((s) => s.deleteExam);
   418→
   419→  const subject = subjects.find((s) => s.id === exam.subjectId);
   420→  const examPyqs = pyqs.filter((p) => p.examId === exam.id);
   421→  const typeConf = examTypeConfig(exam.type);
   422→
   423→  const [prepNotes, setPrepNotes] = useState(exam.preparationNotes ?? '');
   424→  const [pyqDialogOpen, setPyqDialogOpen] = useState(false);
   425→  const [deleteId, setDeleteId] = useState<string | null>(null);
   426→
   427→  const handleSaveNotes = () => {
   428→    updateExam(exam.id, { preparationNotes: prepNotes });
   429→  };
   430→
   431→  const handleDeleteExam = () => {
   432→    deleteExam(exam.id);
   433→    onCollapse();
   434→  };
   435→
   436→  const percentage =
   437→    exam.status === 'completed' && exam.obtainedMarks !== undefined && exam.totalMarks > 0
   438→      ? Math.round((exam.obtainedMarks / exam.totalMarks) * 100)
   439→      : null;
   440→  const grade = percentage !== null ? GRADE_FROM_PERCENTAGE(percentage) : null;
   441→
   442→  return (
   443→    <motion.div
   444→      initial={{ opacity: 0, height: 0 }}
   445→      animate={{ opacity: 1, height: 'auto' }}
   446→      exit={{ opacity: 0, height: 0 }}
   447→      className='overflow-hidden'
   448→    >
   449→      <div className='metric-card p-4 space-y-4'>
   450→        {/* Header */}
   451→        <div className='flex items-start justify-between gap-3'>
   452→          <div className='space-y-2'>
   453→            <h3 className='text-base font-semibold'>{exam.name}</h3>
   454→            <div className='flex items-center gap-2 flex-wrap'>
   455→              {subject && (
   456→                <span
   457→                  className='inline-flex items-center gap-1.5 text-[11px] font-medium'
   458→                  style={{ color: subject.color }}
   459→                >
   460→                  <div className='status-dot' style={{ backgroundColor: subject.color }} />
   461→                  {subject.code}
   462→                </span>
   463→              )}
   464→              <span className={`inline-flex items-center px-1.5 py-0.5 rounded-md text-[10px] font-semibold tracking-wider uppercase ${typeConf.signalClass}`}>
   465→                {typeConf.label}
   466→              </span>
   467→              <span className='text-[11px] text-muted-foreground'>{exam.totalMarks} marks</span>
   468→              <span className='text-[11px] text-muted-foreground flex items-center gap-1'>
   469→                <CalendarDays className='h-3 w-3' />
   470→                {format(parseISO(exam.date), 'MMM d, yyyy')}
   471→              </span>
   472→            </div>
   473→          </div>
   474→          <div className='flex items-center gap-1 shrink-0'>
   475→            <Button variant='ghost' size='sm' className='h-7 w-7 p-0' onClick={onCollapse}>
   476→              <ChevronDown className='h-4 w-4' />
   477→            </Button>
   478→            <Button
   479→              variant='ghost'
   480→              size='icon'
   481→              className='h-7 w-7 text-muted-foreground hover:text-red-500'
   482→              onClick={() => setDeleteId('exam')}
   483→            >
   484→              <Trash2 className='h-3.5 w-3.5' />
   485→            </Button>
   486→          </div>
   487→        </div>
   488→
   489→        {/* Completed stats */}
   490→        {exam.status === 'completed' && percentage !== null && (
   491→          <div className='flex items-center gap-3'>
   492→            <span className='text-sm font-semibold'>{exam.obtainedMarks}</span>
   493→            <span className='text-sm text-muted-foreground'>/ {exam.totalMarks}</span>

   495→              {percentage}%
   496→            </span>
   497→            {grade && (
   498→              <span className='inline-flex items-center px-1.5 py-0.5 rounded-md text-[10px] font-semibold tracking-wider uppercase signal-healthy'>
   499→                {grade}
   500→              </span>
   501→            )}
   502→          </div>
   503→        )}
   504→
   505→        {/* Upcoming badge */}
   506→        {exam.status === 'upcoming' && (
   507→          <span className='inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-semibold tracking-wider uppercase bg-purple-50 text-purple-700 dark:bg-purple-950/50 dark:text-purple-400'>

   518→              Add PYQ
   519→            </Button>
   520→          </div>

   522→          {examPyqs.length === 0 ? (
   523→            <p className='text-xs text-muted-foreground py-4 text-center'>
   524→              No previous year questions added yet.
   525→            </p>
   526→          ) : (
   527→            <div className='space-y-2 max-h-96 overflow-y-auto scrollbar-thin'>
   528→              {examPyqs.map((pyq) => {
   529→                const diffConf = difficultyConfig(pyq.difficulty);
   530→                return (
   531→                  <div
   532→                    key={pyq.id}
   533→                    className='rounded-md border border-border p-3 hover:border-primary/25 transition-colors group'
   534→                  >
   535→                    <div className='flex items-start justify-between gap-2'>
   536→                      <div className='flex-1 min-w-0'>
   537→                        <div className='flex items-center gap-2 flex-wrap mb-1'>
   538→                          <span className='text-[11px] font-medium text-muted-foreground'>{pyq.year}</span>
   539→                          {diffConf && (
   540→                            <span className={`inline-flex items-center px-1.5 py-0.5 rounded-md text-[9px] font-semibold tracking-wider uppercase ${diffConf.className}`}>
   541→                              {diffConf.label}
   542→                            </span>
   543→                          )}
   544→                          {pyq.attempted && (
   545→                            pyq.correct ? (
   546→                              <span className='inline-flex items-center px-1.5 py-0.5 rounded-md text-[9px] font-semibold tracking-wider uppercase signal-healthy'>
   547→                                Correct
   548→                              </span>
   549→                            ) : (
   550→                              <span className='inline-flex items-center px-1.5 py-0.5 rounded-md text-[9px] font-semibold tracking-wider uppercase signal-critical'>
   551→                                Incorrect
   552→                              </span>
   553→                            )
   554→                          )}
   555→                          {!pyq.attempted && (
   556→                            <span className='text-[9px] text-muted-foreground'>Not attempted</span>
   557→                          )}
   558→                        </div>
   559→                        <p className='text-sm leading-relaxed'>{pyq.question}</p>
   560→                        {pyq.answer && (
   561→                          <p className='text-xs text-muted-foreground mt-1.5 line-clamp-2'>{pyq.answer}</p>
   562→                        )}
   563→                      </div>
   564→                      <Button
