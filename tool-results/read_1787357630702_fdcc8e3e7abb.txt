    40→    40→// -- Health status --
    41→    41→function getHealthConfig(score: number) {
    42→    42→  if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    43→    43→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    44→    44→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    45→    45→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    46→    46→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    47→    47→}

    49→    49→function formatStudyTime(seconds: number): string {
    50→    50→  const totalMinutes = Math.floor(seconds / 60);
    51→    51→  const h = Math.floor(totalMinutes / 60);
    52→    52→  const m = totalMinutes % 60;
    53→    53→  if (h === 0) return `${m}m`;
    54→    54→  if (m === 0) return `${h}h`;
    55→    55→  return `${h}h ${m}m`;
    56→    56→}

    58→    58→function signalDotClass(signal: SignalStatus): string {
    59→    59→  const map: Record<SignalStatus, string> = {
    60→    60→    healthy: 'bg-emerald-500', improving: 'bg-blue-500', attention: 'bg-amber-500',
    61→    61→    critical: 'bg-red-500', upcoming: 'bg-purple-500', nodata: 'bg-muted-foreground/40',
    62→    62→  };
    63→    63→  return map[signal];
    64→    64→}

    70→    70→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
    71→    71→  const days = useMemo(() => {
    72→    72→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
    73→    73→    return Array.from({ length: 7 }, (_, i) => {
    74→    74→      const d = addDays(start, i);
    75→    75→      const dateStr = d.toISOString().split('T')[0];
    76→    76→      const daySessions = studySessions.filter((s) => s.date === dateStr);
    77→    77→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
    78→    78→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
    79→    79→    });
    80→    80→  }, [studySessions]);
    81→    81→
    82→    82→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
    83→    83→  const today = new Date().toISOString().split('T')[0];
    84→    84→
    85→    85→  return (
    86→    86→    <div className="grid grid-cols-7 gap-1.5">
    87→    87→      {days.map((d) => {
    88→    88→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
    89→    89→        const isToday = d.dateStr === today;
    90→    90→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
    91→    91→        return (
    92→    92→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
    93→    93→            <div
    94→    94→              className={`w-full aspect-square rounded-md transition-colors ${
    95→    95→                isToday ? 'ring-2 ring-primary/30' : ''
    96→    96→              }`}
    97→    97→              style={{ backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` }}
    98→    98→            />
    99→    99→            <span className={`text-[10px] ${isToday ? 'font-semibold text-foreground' : 'text-muted-foreground'}`}>{d.label}</span>
   100→   100→            {d.count > 0 && (
   101→   101→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   102→   102→            )}
   103→   103→          </div>
   104→   104→        );
   105→   105→      })}
   106→   106→    </div>
   107→   107→  );
   108→   108→}
