   816→// SM-2 Revision Review Helper
   817→// ═════════════════════════════════════════════════════════════════════
   818→/**
   819→ * Review a revision item using the SM-2 algorithm.
   820→ * @param store - the zustand store
   821→ * @param itemId - the revision item id
   822→ * @param quality - quality of recall (0-5). >= 3 means pass.
   823→ * @returns void (mutates the store)
   824→ */
   825→export function reviewRevisionItem(
   826→  store: ReturnType<typeof useStore>,
   827→  itemId: string,
   828→  quality: number
   829→) {
   830→  const item = store.getState().revisionItems.find((r) => r.id === itemId);
   831→  if (!item) return;
   832→
   833→  const q = Math.max(0, Math.min(5, quality));
   834→  let { easeFactor, interval, repetitions } = item;
   835→
   836→  if (q >= 3) {
   837→    // Good response
   838→    if (repetitions === 0) {
   839→      interval = 1;
   840→    } else if (repetitions === 1) {
   841→      interval = 6;
   842→    } else {
   843→      interval = Math.round(interval * easeFactor);
   844→    }
   845→    repetitions += 1;
   846→    easeFactor =
   847→      Math.max(1.3, easeFactor + 0.1 - (5 - q) * (0.08 + (5 - q) * 0.02));
   848→  } else {
   849→    // Bad response - reset
   850→    repetitions = 0;
   851→    interval = 1;
   852→  }
   853→
   854→  const nextReviewDate = new Date();
   855→  nextReviewDate.setDate(nextReviewDate.getDate() + interval);
   856→  const nextReview = nextReviewDate.toISOString().split('T')[0];
   857→
   858→  store.getState().updateRevisionItem(itemId, {
   859→    easeFactor,
   860→    interval,
   861→    repetitions,
   862→    nextReview,
   863→    lastReview: todayStr(),
   864→  });
   865→}

   871→/** Syllabus completion % for a subject */
   872→export function getSubjectProgress(
   873→  state: Pick<AppState, 'syllabusUnits'>,
   874→  subjectId: string
   875→): number {
   876→  const units = state.syllabusUnits.filter((u) => u.subjectId === subjectId);
   877→  const allTopics = units.flatMap((u) => u.topics);
   878→  if (allTopics.length === 0) return 0;
   879→  const completed = allTopics.filter((t) => t.completed).length;
   880→  return Math.round((completed / allTopics.length) * 100);
   881→}

   883→/** Attendance stats for a subject */
   884→export function getSubjectAttendance(
   885→  state: Pick<AppState, 'attendance'>,
   886→  subjectId: string
   887→): { present: number; total: number; percentage: number } {
   888→  const records = state.attendance.filter((a) => a.subjectId === subjectId);
   889→  const total = records.reduce((sum, r) => sum + r.totalClasses, 0);
   890→  const present = records.filter((r) => r.present).reduce((sum, r) => sum + r.totalClasses, 0);
   891→  const percentage = total > 0 ? Math.round((present / total) * 100) : 0;
   892→  return { present, total, percentage };

   916→/** Current semester SGPA */
   917→export function calculateSGPA(state: Pick<AppState, 'subjects' | 'assessments'>): number {
   918→  const activeSubjects = state.subjects.filter((s) => !s.archived);
   919→  if (activeSubjects.length === 0) return 0;
   920→
   921→  let totalWeightedPoints = 0;
   922→  let totalCredits = 0;
   923→
   924→  for (const subject of activeSubjects) {
   925→    const grade = getSubjectGrade(state, subject.id);
   926→    const gradePoint = GRADE_POINTS[grade] ?? 0;
   927→    totalWeightedPoints += gradePoint * subject.credits;
   928→    totalCredits += subject.credits;
   929→  }
   930→
   931→  return totalCredits > 0
   932→    ? Math.round((totalWeightedPoints / totalCredits) * 100) / 100
   933→    : 0;
   934→}

   969→/** Health signal for a subject */
   970→export function getSubjectSignal(
   971→  state: AppState,
   972→  subjectId: string
   973→): SignalStatus {
   974→  const att = getSubjectAttendance(state, subjectId);
   975→  const progress = getSubjectProgress(state, subjectId);
   976→  const marks = getSubjectMarks(state, subjectId);
   977→  const dueRevisions = getDueRevisionItems(state).filter(
   978→    (r) => r.subjectId === subjectId
   979→  );
   980→
   981→  // No data at all
   982→  if (att.total === 0 && marks.max === 0 && progress === 0) {
   983→    return 'nodata';
   984→  }
   985→
   986→  // Critical: attendance below threshold or very low marks
   987→  if (
   988→    att.total > 0 && att.percentage < 60
   989→  ) {
   990→    return 'critical';
   991→  }
   992→
   993→  if (
   994→    marks.max > 0 && marks.percentage < 40
   995→  ) {
   996→    return 'critical';
   997→  }
   998→
   999→  // Attention: attendance near threshold, many overdue revisions, or low marks
  1000→  if (
  1001→    (att.total > 0 && att.percentage < state.profile.attendanceThreshold) ||
  1002→    dueRevisions.length >= 3 ||
  1003→    (marks.max > 0 && marks.percentage < 55)
  1004→  ) {
  1005→    return 'attention';
  1006→  }
  1007→
  1008→  // Improving: some progress but not great
  1009→  if (
  1010→    progress > 0 && progress < 50 &&
  1011→    (marks.max === 0 || marks.percentage < 70)
  1012→  ) {
  1013→    return 'improving';
  1014→  }
  1015→
  1016→  // Healthy: good attendance, decent progress
  1017→  if (
  1018→    (att.total === 0 || att.percentage >= state.profile.attendanceThreshold) &&
  1019→    progress >= 50
  1020→  ) {
  1021→    return 'healthy';
  1022→  }
  1023→
  1024→  return 'improving';
  1025→}

  1027→/** Overall semester health score 0-100 */
  1028→export function getSemesterHealth(state: AppState): number {
  1029→  const activeSubjects = state.subjects.filter((s) => !s.archived);
  1030→  if (activeSubjects.length === 0) return 0;
  1031→
  1032→  let totalScore = 0;
  1033→  let maxScore = 0;
  1034→
  1035→  for (const subject of activeSubjects) {
  1036→    // Attendance score (30 points max)
  1037→    const att = getSubjectAttendance(state, subject.id);
  1038→    const attScore = att.total > 0 ? Math.min(30, (att.percentage / 100) * 30) : 10;
  1039→
  1040→    // Marks score (30 points max)
  1041→    const marks = getSubjectMarks(state, subject.id);
  1042→    const marksScore = marks.max > 0 ? (marks.percentage / 100) * 30 : 10;
  1043→
  1044→    // Syllabus progress (20 points max)
  1045→    const progress = getSubjectProgress(state, subject.id);
  1046→    const progressScore = (progress / 100) * 20;
  1047→
  1048→    // Revision health (20 points max)
  1049→    const subjectRevisions = state.revisionItems.filter(
  1050→      (r) => r.subjectId === subject.id
  1051→    );
  1052→    const overdueRevisions = subjectRevisions.filter(
  1053→      (r) => r.nextReview <= todayStr()
  1054→    ).length;
  1055→    const revisionScore = subjectRevisions.length > 0
  1056→      ? Math.max(0, 20 - overdueRevisions * 5)
  1057→      : 10;
  1058→
  1059→    totalScore += attScore + marksScore + progressScore + revisionScore;
  1060→    maxScore += 100;
  1061→  }
  1062→
  1063→  return Math.round((totalScore / maxScore) * 100);
  1064→}

  1066→/** Total study time this week (in seconds) */
  1067→export function getStudyTimeThisWeek(
  1068→  state: Pick<AppState, 'studySessions'>
  1069→): number {
  1070→  const now = new Date();
  1071→  const dayOfWeek = now.getDay(); // 0=Sun
  1072→   // Start of week: Monday (day 1) or Sunday (day 0) — we use Sunday as start
  1073→  const startOfWeek = new Date(now);
  1074→  startOfWeek.setDate(now.getDate() - dayOfWeek);
  1075→  startOfWeek.setHours(0, 0, 0, 0);
  1076→
  1077→  const startStr = startOfWeek.toISOString().split('T')[0];
  1078→
  1079→  return state.studySessions
  1080→    .filter((s) => s.date >= startStr)
  1081→    .reduce((sum, s) => sum + s.duration, 0);
  1082→}

  1084→/** Current consecutive day study streak (days with >= 1 session) */
  1085→export function getStudyStreak(
  1086→  state: Pick<AppState, 'studySessions'>
  1087→): number {
  1088→  const today = new Date();
  1089→  const sessionsByDate = new Map<string, number>();
  1090→  for (const s of state.studySessions) {
  1091→    sessionsByDate.set(s.date, (sessionsByDate.get(s.date) || 0) + 1);
  1092→  }
  1093→
  1094→  let streak = 0;
  1095→  for (let i = 0; i < 365; i++) {
  1096→    const d = new Date(today);
  1097→    d.setDate(d.getDate() - i);
  1098→    const dateStr = d.toISOString().split('T')[0];
  1099→    if ((sessionsByDate.get(dateStr) || 0) >= 1) {
  1100→      streak++;
  1101→    } else {
  1102→      // If today has no sessions, don't count it as breaking the streak
  1103→      // (streak counts backwards from most recent study day)
  1104→      if (i === 0) continue;
  1105→      break;
  1106→    }
  1107→  }
  1108→  return streak;
  1109→}
