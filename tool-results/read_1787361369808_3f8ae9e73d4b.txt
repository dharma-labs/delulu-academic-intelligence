    42→// -- Health status --
    43→function getHealthConfig(score: number) {
    44→  if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    45→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    46→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    47→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    48→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    49→}

    51→function formatStudyTime(seconds: number): string {
    52→  const totalMinutes = Math.floor(seconds / 60);
    53→  const h = Math.floor(totalMinutes / 60);
    54→  const m = totalMinutes % 60;
    55→  if (h === 0) return `${m}m`;
    56→  if (m === 0) return `${h}h`;
    57→  return `${h}h ${m}m`;
    58→}

    60→function signalDotClass(signal: SignalStatus): string {
    61→  const map: Record<SignalStatus, string> = {
    62→    healthy: 'bg-emerald-500', improving: 'bg-blue-500', attention: 'bg-amber-500',
    63→    critical: 'bg-red-500', upcoming: 'bg-purple-500', nodata: 'bg-muted-foreground/40',
    64→  };
    65→  return map[signal];
    66→}

    71→// ─── Weekly Heatmap ──────────────────────────────────────────────
    72→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
    73→  const days = useMemo(() => {
    74→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
    75→    return Array.from({ length: 7 }, (_, i) => {
    76→      const d = addDays(start, i);
    77→      const dateStr = d.toISOString().split('T')[0];
    78→      const daySessions = studySessions.filter((s) => s.date === dateStr);
    79→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
    80→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
    81→    });
    82→  }, [studySessions]);
    83→
    84→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
    85→  const today = new Date().toISOString().split('T')[0];
    86→
    87→  return (
    88→    <div className="grid grid-cols-7 gap-1.5">
    89→      {days.map((d) => {
    90→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
    91→        const isToday = d.dateStr === today;
    92→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
    93→        return (
    94→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
    95→            <div
    96→              className={`w-full aspect-square rounded-md transition-colors ${
    97→                isToday ? 'ring-2 ring-primary/30' : ''
    98→              }`}
    99→              style={{ backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` }}
   100→            />
   101→            <span className={`text-[10px] ${isToday ? 'font-semibold text-foreground' : 'text-muted-foreground'}`}>{d.label}</span>
   102→            {d.count > 0 && (
   103→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   104→            )}
   105→          </div>
   106→        );
   107→      })}
   108→    </div>
   109→  );
   110→}
