   717→   816→// SM-2 Revision Review Helper
   718→   817→// ═════════════════════════════════════════════════════════════════════
   719→   818→/**
   720→   819→ * Review a revision item using the SM-2 algorithm.
   721→   820→ * @param store - the zustand store
   722→   821→ * @param itemId - the revision item id
   723→   822→ * @param quality - quality of recall (0-5). >= 3 means pass.
   724→   823→ * @returns void (mutates the store)
   725→   824→ */
   726→   825→export function reviewRevisionItem(
   727→   826→  store: ReturnType<typeof useStore>,
   728→   827→  itemId: string,
   729→   828→  quality: number
   730→   829→) {
   731→   830→  const item = store.getState().revisionItems.find((r) => r.id === itemId);
   732→   831→  if (!item) return;
   733→   832→
   734→   833→  const q = Math.max(0, Math.min(5, quality));
   735→   834→  let { easeFactor, interval, repetitions } = item;
   736→   835→
   737→   836→  if (q >= 3) {
   738→   837→    // Good response
   739→   838→    if (repetitions === 0) {
   740→   839→      interval = 1;
   741→   840→    } else if (repetitions === 1) {
   742→   841→      interval = 6;
   743→   842→    } else {
   744→   843→      interval = Math.round(interval * easeFactor);
   745→   844→    }
   746→   845→    repetitions += 1;
   747→   846→    easeFactor =
   748→   847→      Math.max(1.3, easeFactor + 0.1 - (5 - q) * (0.08 + (5 - q) * 0.02));
   749→   848→  } else {
   750→   849→    // Bad response - reset
   751→   850→    repetitions = 0;
   752→   851→    interval = 1;
   753→   852→  }
   754→   853→
   755→   854→  const nextReviewDate = new Date();
   756→   855→  nextReviewDate.setDate(nextReviewDate.getDate() + interval);
   757→   856→  const nextReview = nextReviewDate.toISOString().split('T')[0];
   758→   857→
   759→   858→  store.getState().updateRevisionItem(itemId, {
   760→   859→    easeFactor,
   761→   860→    interval,
   762→   861→    repetitions,
   763→   862→    nextReview,
   764→   863→    lastReview: todayStr(),
   765→   864→  });
   766→   865→}

   773→   872→export function getSubjectProgress(
   774→   873→  state: Pick<AppState, 'syllabusUnits'>,

   817→   916→/** Current semester SGPA */
   818→   917→export function calculateSGPA(state: Pick<AppState, 'subjects' | 'assessments'>): number {
   819→   918→  const activeSubjects = state.subjects.filter((s) => !s.archived);
   820→   919→  if (activeSubjects.length === 0) return 0;
   821→   920→
   822→   921→  let totalWeightedPoints = 0;
   823→   922→  let totalCredits = 0;
   824→   923→
   825→   924→  for (const subject of activeSubjects) {
   826→   925→    const grade = getSubjectGrade(state, subject.id);
   827→   926→    const gradePoint = GRADE_POINTS[grade] ?? 0;
   828→   927→    totalWeightedPoints += gradePoint * subject.credits;
   829→   928→    totalCredits += subject.credits;
   830→   929→  }
   831→   930→
   832→   931→  return totalCredits > 0

   870→   969→/** Health signal for a subject */
   871→   970→export function getSubjectSignal(
   872→   971→  state: AppState,
   873→   972→  subjectId: string
   874→   973→): SignalStatus {
   875→   974→  const att = getSubjectAttendance(state, subjectId);
   876→   975→  const progress = getSubjectProgress(state, subjectId);
   877→   976→  const marks = getSubjectMarks(state, subjectId);
   878→   977→  const dueRevisions = getDueRevisionItems(state).filter(
   879→   978→    (r) => r.subjectId === subjectId
   880→   979→  );
   881→   980→
   882→   981→  // No data at all
   883→   982→  if (att.total === 0 && marks.max === 0 && progress === 0) {
   884→   983→    return 'nodata';
   885→   984→  }
   886→   985→
   887→   986→  // Critical: attendance below threshold or very low marks
   888→   987→  if (
   889→   988→    att.total > 0 && att.percentage < 60
   890→   989→  ) {
   891→   990→    return 'critical';
   892→   991→  }
   893→   992→
   894→   993→  if (
   895→   994→    marks.max > 0 && marks.percentage < 40
   896→   995→  ) {
   897→   996→    return 'critical';
   898→   997→  }
   899→   998→
   900→   999→  // Attention: attendance near threshold, many overdue revisions, or low marks
   901→  1000→  if (
   902→  1001→    (att.total > 0 && att.percentage < state.profile.attendanceThreshold) ||
   903→  1002→    dueRevisions.length >= 3 ||
   904→  1003→    (marks.max > 0 && marks.percentage < 55)
   905→  1004→  ) {
   906→  1005→    return 'attention';
   907→  1006→  }
   908→  1007→
   909→  1008→  // Improving: some progress but not great
   910→  1009→  if (
   911→  1010→    progress > 0 && progress < 50 &&
   912→  1011→    (marks.max === 0 || marks.percentage < 70)
   913→  1012→  ) {
   914→  1013→    return 'improving';
   915→  1014→  }
   916→  1015→
   917→  1016→  // Healthy: good attendance, decent progress
   918→  1017→  if (
   919→  1018→    (att.total === 0 || att.percentage >= state.profile.attendanceThreshold) &&
   920→  1019→    progress >= 50
   921→  1020→  ) {
   922→  1021→    return 'healthy';
   923→  1022→  }
   924→  1023→
   925→  1024→  return 'improving';
   926→  1025→}

   928→  1027→/** Overall semester health score 0-100 */
   929→  1028→export function getSemesterHealth(state: AppState): number {
   930→  1029→  const activeSubjects = state.subjects.filter((s) => !s.archived);
   931→  1030→  if (activeSubjects.length === 0) return 0;
   932→  1031→
   933→  1032→  let totalScore = 0;
   934→  1033→  let maxScore = 0;
   935→  1034→
   936→  1035→  for (const subject of activeSubjects) {
   937→  1036→    // Attendance score (30 points max)
   938→  1037→    const att = getSubjectAttendance(state, subject.id);
   939→  1038→    const attScore = att.total > 0 ? Math.min(30, (att.percentage / 100) * 30) : 10;
   940→  1039→
   941→  1040→    // Marks score (30 points max)
   942→  1041→    const marks = getSubjectMarks(state, subject.id);
   943→  1042→    const marksScore = marks.max > 0 ? (marks.percentage / 100) * 30 : 10;
   944→  1043→
   945→  1044→    // Syllabus progress (20 points max)
   946→  1045→    const progress = getSubjectProgress(state, subject.id);
   947→  1046→    const progressScore = (progress / 100) * 20;
   948→  1047→
   949→  1048→    // Revision health (20 points max)
   950→  1049→    const subjectRevisions = state.revisionItems.filter(
   951→  1050→      (r) => r.subjectId === subject.id
   952→  1051→    );
   953→  1052→    const overdueRevisions = subjectRevisions.filter(
   954→  1053→      (r) => r.nextReview <= todayStr()
   955→  1054→    ).length;
   956→  1055→    const revisionScore = subjectRevisions.length > 0
   957→  1056→      ? Math.max(0, 20 - overdueRevisions * 5)
   958→  1057→      : 10;
   959→  1058→
   960→  1059→    totalScore += attScore + marksScore + progressScore + revisionScore;
   961→  1060→    maxScore += 100;
   962→  1061→  }
   963→  1062→
   964→  1063→  return Math.round((totalScore / maxScore) * 100);
   965→  1064→}

   967→  1066→/** Total study time this week (in seconds) */
   968→  1067→export function getStudyTimeThisWeek(
   969→  1068→  state: Pick<AppState, 'studySessions'>
   970→  1069→): number {
   971→  1070→  const now = new Date();
   972→  1071→  const dayOfWeek = now.getDay(); // 0=Sun
   973→  1072→   // Start of week: Monday (day 1) or Sunday (day 0) — we use Sunday as start
   974→  1073→  const startOfWeek = new Date(now);
   975→  1074→  startOfWeek.setDate(now.getDate() - dayOfWeek);
   976→  1075→  startOfWeek.setHours(0, 0, 0, 0);
   977→  1076→
   978→  1077→  const startStr = startOfWeek.toISOString().split('T')[0];
   979→  1078→
   980→  1079→  return state.studySessions
   981→  1080→    .filter((s) => s.date >= startStr)
   982→  1081→    .reduce((sum, s) => sum + s.duration, 0);
   983→  1082→}

   985→  1084→/** Current consecutive day study streak (days with >= 1 session) */
   986→  1085→export function getStudyStreak(
   987→  1086→  state: Pick<AppState, 'studySessions'>
   988→  1087→): number {
   989→  1088→  const today = new Date();
   990→  1089→  const sessionsByDate = new Map<string, number>();
   991→  1090→  for (const s of state.studySessions) {
   992→  1091→    sessionsByDate.set(s.date, (sessionsByDate.get(s.date) || 0) + 1);
   993→  1092→  }
   994→  1093→
   995→  1094→  let streak = 0;
   996→  1095→  for (let i = 0; i < 365; i++) {
   997→  1096→    const d = new Date(today);
   998→  1097→    d.setDate(d.getDate() - i);
   999→  1098→    const dateStr = d.toISOString().split('T')[0];
  1000→  1099→    if ((sessionsByDate.get(dateStr) || 0) >= 1) {
  1001→  1100→      streak++;
  1002→  1101→    } else {
  1003→  1102→      // If today has no sessions, don't count it as breaking the streak
  1004→  1103→      // (streak counts backwards from most recent study day)
  1005→  1104→      if (i === 0) continue;
  1006→  1105→      break;
  1007→  1106→    }
  1008→  1107→  }
  1009→  1108→  return streak;
  1010→  1109→}
