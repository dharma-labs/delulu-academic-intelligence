   864→/**
   865→ * Review a revision item using the SM-2 algorithm.
   866→ * @param store - the zustand store
   867→ * @param itemId - the revision item id
   868→ * @param quality - quality of recall (0-5). >= 3 means pass.
   869→ * @returns void (mutates the store)
   870→ */
   871→export function reviewRevisionItem(
   872→  store: ReturnType<typeof useStore>,
   873→  itemId: string,
   874→  quality: number
   875→) {
   876→  const item = store.getState().revisionItems.find((r) => r.id === itemId);
   877→  if (!item) return;
   878→
   879→  const q = Math.max(0, Math.min(5, quality));
   880→  let { easeFactor, interval, repetitions } = item;
   881→
   882→  if (q >= 3) {
   883→    // Good response
   884→    if (repetitions === 0) {
   885→      interval = 1;
   886→    } else if (repetitions === 1) {
   887→      interval = 6;
   888→    } else {
   889→      interval = Math.round(interval * easeFactor);
   890→    }
   891→    repetitions += 1;
   892→    easeFactor =
   893→      Math.max(1.3, easeFactor + 0.1 - (5 - q) * (0.08 + (5 - q) * 0.02));
   894→  } else {
   895→    // Bad response - reset
   896→    repetitions = 0;
   897→    interval = 1;
   898→  }
   899→
   900→  const nextReviewDate = new Date();
   901→  nextReviewDate.setDate(nextReviewDate.getDate() + interval);
   902→  const nextReview = nextReviewDate.toISOString().split('T')[0];
   903→
   904→  store.getState().updateRevisionItem(itemId, {
   905→    easeFactor,
   906→    interval,
   907→    repetitions,
   908→    nextReview,
   909→    lastReview: todayStr(),
   910→  });
   911→}

   917→/** Syllabus completion % for a subject */
   918→export function getSubjectProgress(
   919→  state: Pick<AppState, 'syllabusUnits'>,
   920→  subjectId: string

   924→  if (allTopics.length === 0) return 0;

   929→/** Attendance stats for a subject */
   930→export function getSubjectAttendance(
   931→  state: Pick<AppState, 'attendance'>,
   932→  subjectId: string
   933→): { present: number; total: number; percentage: number } {
   934→  const records = state.attendance.filter((a) => a.subjectId === subjectId);
   935→  const total = records.reduce((sum, r) => sum + r.totalClasses, 0);
   936→  const present = records.filter((r) => r.present).reduce((sum, r) => sum + r.totalClasses, 0);
   937→  const percentage = total > 0 ? Math.round((present / total) * 100) : 0;
   938→  return { present, total, percentage };

   962→/** Current semester SGPA */
   963→export function calculateSGPA(state: Pick<AppState, 'subjects' | 'assessments'>): number {
   964→  const activeSubjects = state.subjects.filter((s) => !s.archived);
   965→  if (activeSubjects.length === 0) return 0;
   966→
   967→  let totalWeightedPoints = 0;
   968→  let totalCredits = 0;
   969→
   970→  for (const subject of activeSubjects) {
   971→    const grade = getSubjectGrade(state, subject.id);
   972→    const gradePoint = GRADE_POINTS[grade] ?? 0;
   973→    totalWeightedPoints += gradePoint * subject.credits;
   974→    totalCredits += subject.credits;
   975→  }
   976→
   977→  return totalCredits > 0
   978→    ? Math.round((totalWeightedPoints / totalCredits) * 100) / 100
   979→    : 0;
   980→}

  1015→/** Health signal for a subject */
  1016→export function getSubjectSignal(
  1017→  state: AppState,
  1018→  subjectId: string
  1019→): SignalStatus {
  1020→  const att = getSubjectAttendance(state, subjectId);
  1021→  const progress = getSubjectProgress(state, subjectId);
  1022→  const marks = getSubjectMarks(state, subjectId);
  1023→  const dueRevisions = getDueRevisionItems(state).filter(
  1024→    (r) => r.subjectId === subjectId
  1025→  );
  1026→
  1027→  // No data at all
  1028→  if (att.total === 0 && marks.max === 0 && progress === 0) {
  1029→    return 'nodata';
  1030→  }
  1031→
  1032→  // Critical: attendance below threshold or very low marks
  1033→  if (
  1034→    att.total > 0 && att.percentage < 60
  1035→  ) {
  1036→    return 'critical';
  1037→  }
  1038→
  1039→  if (
  1040→    marks.max > 0 && marks.percentage < 40
  1041→  ) {
  1042→    return 'critical';
  1043→  }
  1044→
  1045→  // Attention: attendance near threshold, many overdue revisions, or low marks
  1046→  if (
  1047→    (att.total > 0 && att.percentage < state.profile.attendanceThreshold) ||
  1048→    dueRevisions.length >= 3 ||
  1049→    (marks.max > 0 && marks.percentage < 55)
  1050→  ) {
  1051→    return 'attention';
  1052→  }
  1053→
  1054→  // Improving: some progress but not great
  1055→  if (
  1056→    progress > 0 && progress < 50 &&
  1057→    (marks.max === 0 || marks.percentage < 70)
  1058→  ) {
  1059→    return 'improving';
  1060→  }
  1061→
  1062→  // Healthy: good attendance, decent progress
  1063→  if (
  1064→    (att.total === 0 || att.percentage >= state.profile.attendanceThreshold) &&
  1065→    progress >= 50
  1066→  ) {
  1067→    return 'healthy';
  1068→  }
  1069→
  1070→  return 'improving';
  1071→}

  1073→/** Overall semester health score 0-100 */
  1074→export function getSemesterHealth(state: AppState): number {
  1075→  const activeSubjects = state.subjects.filter((s) => !s.archived);
  1076→  if (activeSubjects.length === 0) return 0;
  1077→
  1078→  let totalScore = 0;
  1079→  let maxScore = 0;
  1080→
  1081→  for (const subject of activeSubjects) {
  1082→    // Attendance score (30 points max)
  1083→    const att = getSubjectAttendance(state, subject.id);
  1084→    const attScore = att.total > 0 ? Math.min(30, (att.percentage / 100) * 30) : 10;
  1085→
  1086→    // Marks score (30 points max)
  1087→    const marks = getSubjectMarks(state, subject.id);
  1088→    const marksScore = marks.max > 0 ? (marks.percentage / 100) * 30 : 10;
  1089→
  1090→    // Syllabus progress (20 points max)
  1091→    const progress = getSubjectProgress(state, subject.id);
  1092→    const progressScore = (progress / 100) * 20;
  1093→
  1094→    // Revision health (20 points max)
  1095→    const subjectRevisions = state.revisionItems.filter(
  1096→      (r) => r.subjectId === subject.id
  1097→    );
  1098→    const overdueRevisions = subjectRevisions.filter(
  1099→      (r) => r.nextReview <= todayStr()
  1100→    ).length;
  1101→    const revisionScore = subjectRevisions.length > 0
  1102→      ? Math.max(0, 20 - overdueRevisions * 5)
  1103→      : 10;
  1104→
  1105→    totalScore += attScore + marksScore + progressScore + revisionScore;
  1106→    maxScore += 100;
  1107→  }
  1108→
  1109→  return Math.round((totalScore / maxScore) * 100);
  1110→}

  1130→/** Current consecutive day study streak (days with >= 1 session) */
  1131→export function getStudyStreak(
  1132→  state: Pick<AppState, 'studySessions'>
  1133→): number {
  1134→  const today = new Date();
  1135→  const sessionsByDate = new Map<string, number>();
  1136→  for (const s of state.studySessions) {
  1137→    sessionsByDate.set(s.date, (sessionsByDate.get(s.date) || 0) + 1);
  1138→  }
  1139→
  1140→  let streak = 0;
  1141→  for (let i = 0; i < 365; i++) {
  1142→    const d = new Date(today);
  1143→    d.setDate(d.getDate() - i);
  1144→    const dateStr = d.toISOString().split('T')[0];
  1145→    if ((sessionsByDate.get(dateStr) || 0) >= 1) {
  1146→      streak++;
  1147→    } else {
  1148→      // If today has no sessions, don't count it as breaking the streak
  1149→      // (streak counts backwards from most recent study day)
  1150→      if (i === 0) continue;
  1151→      break;
  1152→    }
  1153→  }
  1154→  return streak;
  1155→}
