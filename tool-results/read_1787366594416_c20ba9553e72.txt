    63→type Phase = 'setup' | 'active' | 'completion';

    65→// --- Helpers ---
    66→function formatElapsed(totalSeconds: number): string {
    67→  const m = Math.floor(totalSeconds / 60);
    68→  const s = totalSeconds % 60;
    69→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    70→}
    71→
    72→function formatDuration(totalSeconds: number): string {
    73→  if (totalSeconds < 60) return `${totalSeconds}s`;
    74→  const m = Math.floor(totalSeconds / 60);
    75→  const s = totalSeconds % 60;
    76→  if (m < 60) return s > 0 ? `${m}m ${s}s` : `${m}m`;
    77→  const h = Math.floor(m / 60);
    78→  const rm = m % 60;
    79→  return rm > 0 ? `${h}h ${rm}m` : `${h}h`;
    80→}
    81→
    82→function todayDateStr(): string {
    83→  return new Date().toISOString().split('T')[0];
    84→}
