   135→function SidebarNavItem({
   136→  item,
   137→  collapsed,
   138→}: {
   139→  item: NavItem;
   140→  collapsed: boolean;
   141→}) {
   142→  const currentView = useStore((s) => s.currentView);
   143→  const navigate = useStore((s) => s.navigate);
   144→  const isActive = currentView === item.id;
   145→  const Icon = item.icon;
   146→
   147→  return (
   148→    <button
   149→      onClick={() => navigate(item.id as ViewId)}
   150→      title={collapsed ? item.label : undefined}
   151→      className={cn(
   152→        'flex w-full items-center gap-2.5 rounded-md text-[13px] transition-all duration-200',
   153→        collapsed ? 'justify-center px-2 py-2' : 'px-2.5 py-1.5 border-l-2',
   154→        isActive
   155→          ? 'bg-primary/10 text-primary font-medium border-l-primary'
   156→          : 'text-muted-foreground hover:bg-accent hover:text-foreground border-l-transparent'
   157→      )}
   158→    >
   159→      <Icon className="size-[16px] shrink-0" />
   160→      {!collapsed && <span className="truncate">{item.label}</span>}
   161→    </button>
   162→  );
   163→}

   316→    <CommandDialog open={commandOpen} onOpenChange={setCommandOpen}>
   317→      <CommandInput placeholder="Search subjects, tasks, notes, or type a command..." />

   417→                  ...t,

   595→  const studyToday = useMemo(() => getStudyTimeToday({ studySessions }), [studySessions]);
   596→
   597→  const hour = new Date().getHours();
   598→  const greeting = hour < 12 ? 'Good morning' : hour < 17 ? 'Good afternoon' : 'Good evening';

   600→  const todayMinutes = Math.floor(studyToday / 60);
   601→  const todayH = Math.floor(todayMinutes / 60);
   602→  const todayM = todayMinutes % 60;
   603→  const todayStr = todayH === 0 ? `${todayM}m` : todayM === 0 ? `${todayH}h` : `${todayH}h ${todayM}m`;
   604→
   605→  return (
   606→    <div className="top-bar hidden lg:flex items-center gap-4 px-6 py-2">

   608→      <span className="font-semibold text-foreground/80">{greeting}, {profile.name}</span>
   609→      <span className="text-border">|</span>
   610→      <span>{format(new Date(), 'EEE, d MMM · HH:mm')}</span>
   611→      <div className="flex-1" />
   612→      <div className="flex items-center gap-3">
   613→        <span className={cn(
   614→          'inline-flex items-center gap-1 px-1.5 py-0.5 rounded-md text-[10px] font-semibold tracking-wide',
   615→          healthScore >= 75 ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400' : healthScore >= 50 ? 'bg-blue-500/10 text-blue-600 dark:text-blue-400' : healthScore >= 30 ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400' : 'bg-red-500/10 text-red-600 dark:text-red-400'
   616→        )}>
   617→          {healthScore}
   618→        </span>
   619→        {streak > 0 && (
   620→          <span className="inline-flex items-center gap-1 text-orange-500 dark:text-orange-400">
   621→            <Flame className="size-3" />
   622→            <span className="font-semibold tabular-nums">{streak}</span>

   625→        <span className="inline-flex items-center gap-1">
   626→          <Clock className="size-3" />

   628→          <span>today</span>
   629→        </span>
   630→      </div>
   631→    </div>

   637→const FAB_ACTIONS: { label: string; icon: LucideIcon; view: ViewId; color: string }[] = [
   638→  { label: 'New Subject', icon: BookOpen, view: 'subjects', color: 'bg-violet-500 dark:bg-violet-400' },
   639→  { label: 'New Task', icon: CheckSquare, view: 'tasks', color: 'bg-blue-500 dark:bg-blue-400' },

   643→
   644→function MobileFAB() {
   645→  const [open, setOpen] = useState(false);
   646→  const navigate = useStore((s) => s.navigate);
   647→  const currentView = useStore((s) => s.currentView);
   648→  const fabRef = useRef<HTMLDivElement>(null);
   649→
   650→  // Auto-close when navigating away
   651→  const prevViewRef = useRef(currentView);
   652→  useEffect(() => {
   653→    if (prevViewRef.current !== currentView) {
   654→      setOpen(false);
   655→      prevViewRef.current = currentView;
   656→    }
   657→  }, [currentView]);
   658→
   659→  // Close on outside click
   660→  useEffect(() => {
   661→    if (!open) return;
   662→    const handleClick = (e: MouseEvent) => {
   663→      if (fabRef.current && !fabRef.current.contains(e.target as Node)) {
   664→        setOpen(false);
   665→      }
   666→    };
   667→    // Delay to avoid immediate close from the toggle click
   668→    const timer = setTimeout(() => {
   669→      document.addEventListener('mousedown', handleClick);
   670→    }, 0);
   671→    return () => {
   672→      clearTimeout(timer);
   673→      document.removeEventListener('mousedown', handleClick);
   674→    };
   675→  }, [open]);
   676→
   677→  const handleAction = (view: ViewId) => {
   678→    setOpen(false);
   679→    navigate(view);
   680→  };
   681→
   682→  return (
   683→    <div ref={fabRef} className="fixed bottom-[76px] right-4 z-30 md:hidden">
   684→      {/* Backdrop overlay when open */
   685→      <AnimatePresence>
   686→        {open && (
   687→          <motion.div
   688→            initial={{ opacity: 0 }}
   689→            animate={{ opacity: 1 }}
   690→            exit={{ opacity: 0 }}
   691→            transition={{ duration: 0.15 }}
   692→            className="fixed inset-0 z-[-1]"
   693→            onClick={() => setOpen(false)}
   694→          />
   695→        )}
   696→      </AnimatePresence>
   697→
   698→      {/* Radial menu items */
   699→      <AnimatePresence>
   700→        {open && (
   701→          <div className="absolute bottom-16 right-0 flex flex-col items-end gap-2.5">
   702→            {FAB_ACTIONS.map((action, i) => {
   703→              const Icon = action.icon;
   704→              return (
   705→                <motion.div
   706→                  key={action.label}
   707→                  initial={{ opacity: 0, scale: 0.5, y: 10 }}
   708→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   709→                  exit={{ opacity: 0, scale: 0.5, y: 10 }}
   710→                  transition={{ duration: 0.2, delay: i * 0.04 }}
   711→                  className="flex items-center gap-2.5"
   712→                >
   713→                  <span className="text-xs font-medium text-foreground whitespace-nowrap bg-card/90 backdrop-blur-sm border border-border rounded-lg px-2.5 py-1.5 shadow-sm">
   714→                    {action.label}
   715→                  </span>
   716→                  <button
   717→                    onClick={() => handleAction(action.view)}
   718→                    className={cn(
   719→                      'h-10 w-10 rounded-full flex items-center justify-center shadow-lg press-scale transition-colors',
   720→                      action.color,
   721→                      'text-white'
   722→                    )}
   723→                    aria-label={action.label}
   724→                  >
   725→                    <Icon className="size-4" />
   726→                  </button>
   727→                </motion.div>
   728→              );
   729→            })}
   730→          </div>
   731→        )}
   732→      </AnimatePresence>
   733→
   734→      {/* Main FAB button */
   735→      <motion.button
   736→        onClick={() => setOpen((prev) => !prev)}
   737→        whileTap={{ scale: 0.92 }}
   738→        className="h-12 w-12 rounded-full bg-primary text-primary-foreground shadow-lg press-scale flex items-center justify-center"
   739→        aria-label={open ? 'Close quick actions' : 'Open quick actions'}
   740→        style={{ rotate: open ? 45 : 0 }}
   741→        transition={{ duration: 0.2 }}
   742→      >
   743→        <Plus className="size-5" />
   744→      </motion.button>
   745→    </div>
   746→  );
   747→}

   751→export function AppShell() {
   752→  const subjects = useStore((s) => s.subjects);
   753→  const setCommandOpen = useStore((s) => s.setCommandOpen);
   754→  const [moreSheetOpen, setMoreSheetOpen] = useState(false);
   755→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   756→
   757→  const showShortcuts = useCallback(() => {
   758→    setCommandOpen(false);
   759→    setShortcutsOpen(true);
   760→  }, [setCommandOpen]);
   761→
   762→  useEffect(() => {
   763→    const handleKeyDown = (e: KeyboardEvent) => {
   764→      // Don't capture when typing in inputs
   765→      const tag = (e.target as HTMLElement)?.tagName;
   766→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   767→
   768→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   769→        e.preventDefault();
   770→        setCommandOpen(true);
   771→      }
   772→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   773→        e.preventDefault();
   774→        setShortcutsOpen((prev) => !prev);
   775→      }
   776→    };
   777→    document.addEventListener('keydown', handleKeyDown);
   778→    return () => document.removeEventListener('keydown', handleKeyDown);
   779→  }, [setCommandOpen]);
   780→
   781→  return (
   782→    <ToastProvider>
   783→    <MobileFAB />
   784→    <div className="flex min-h-screen">
   785→      <DesktopSidebar />
   786→
   787→      <main className="flex-1 min-w-0 flex flex-col">
   788→        <DesktopTopBar />
   789→        <div className="content-area flex-1 px-4 md:px-5 lg:px-6 py-5 pb-[76px] md:pb-6">
   790→          {/* Mobile header */}
   791→          <div className="flex items-center justify-between mb-5 md:mb-0">
   792→            <div className="flex items-center gap-2">
   793→              <span className="md:hidden text-sm font-bold tracking-tight text-foreground">
   794→                DELULU
   795→              </span>
   796→              <MobileStreakBadge />
   797→            </div>
   798→            <div className="flex items-center gap-1.5 ml-auto">
   799→              <ThemeToggle />
   800→              <button
   801→                onClick={() => setCommandOpen(true)}
   802→                className="md:hidden flex items-center justify-center h-9 w-9 rounded-lg border border-border bg-background/50 text-muted-foreground hover:bg-accent hover:text-foreground active:scale-95 transition-all duration-150"
   803→                aria-label="Search"
   804→              >
   805→                <Search className="size-4" />
   806→              </button>
   807→            </div>
   808→          </div>
   809→
   810→          {/* Onboarding — shows when no subjects exist */}
   811→          {subjects.length === 0 && <Onboarding />}
   812→
   813→          <ViewRouter />
   814→        </div>
   815→      </main>
   816→
   817→      <MobileBottomNav onMoreOpen={() => setMoreSheetOpen(true)} />
   818→      <MobileMoreSheet open={moreSheetOpen} onOpenChange={setMoreSheetOpen} />
   819→      <CommandPalette onShowShortcuts={showShortcuts} />
   820→      <ShortcutsOverlay open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   821→    </div>
   822→    </ToastProvider>
   823→  );
   824→}
