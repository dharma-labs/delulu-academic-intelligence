    42→// -- Health status --
    43→function getHealthConfig(score: number) {
    44→   if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    45→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    46→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    47→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    48→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    49→}

    51→function formatStudyTime(seconds: number): string {
    52→  const totalMinutes = Math.floor(seconds / 60);
    53→  const h = Math.floor(totalMinutes / 60);
    54→  const m = totalMinutes % 60;
    55→  if (h === 0) return `${m}m`;
    56→  if (m === 0) return `${h}h`;
    57→  return `${h}h ${m}m`;
    58→}

    63→    critical: 'bg-red-500', upcoming: 'bg-purple-500', nodata: 'bg-muted-foreground/40',

    96→// ─── Weekly Heatmap ──────────────────────────────────────────────
    97→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
    98→  const days = useMemo(() => {
    99→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
   100→    return Array.from({ length: 7 }, (_, i) => {
   101→      const d = addDays(start, i);
   102→      const dateStr = d.toISOString().split('T')[0];
   103→      const daySessions = studySessions.filter((s) => s.date === dateStr);
   104→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
   105→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
   106→    });
   107→  }, [studySessions]);
   108→
   109→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
   110→  const today = new Date().toISOString().split('T')[0];
   111→
   112→  return (
   113→    <div className="grid grid-cols-7 gap-1.5">
   114→      {days.map((d) => {
   115→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
   116→        const isToday = d.dateStr === today;
   117→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
   118→        return (
   119→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
   120→            <div
   121→              className={cn('w-full aspect-square rounded-md transition-colors', isToday && 'ring-2 ring-primary/30')}
   122→              style={{ backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` }}
   123→            />
   124→            <span className={cn('text-[10px]', isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>{d.label}</span>
   125→            {d.count > 0 && (
   126→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   127→            )}
   128→          </div>
   129→        );
   130→      })}
   131→    </div>
   132→  );
   133→}
