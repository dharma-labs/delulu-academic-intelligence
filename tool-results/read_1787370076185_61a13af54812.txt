    43→// -- Health status --
    44→function getHealthConfig(score: number) {
    45→   if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    46→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    47→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    48→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    49→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    50→}

    52→function formatStudyTime(seconds: number): string {
    53→  const totalMinutes = Math.floor(seconds / 60);
    54→  const h = Math.floor(totalMinutes / 60);
    55→  const m = totalMinutes % 60;
    56→  if (h === 0) return `${m}m`;
    57→  if (m === 0) return `${h}h`;
    58→  return `${h}h ${m}m`;
    59→}

    97→// ─── Weekly Heatmap ──────────────────────────────────────────────
    98→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
    99→  const days = useMemo(() => {
   100→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
   101→    return Array.from({ length: 7 }, (_, i) => {
   102→      const d = addDays(start, i);
   103→      const dateStr = d.toISOString().split('T')[0];
   104→      const daySessions = studySessions.filter((s) => s.date === dateStr);
   105→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
   106→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
   107→    });
   108→  }, [studySessions]);
   109→
   110→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
   111→  const today = new Date().toISOString().split('T')[0];
   112→
   113→  return (
   114→    <div className="grid grid-cols-7 gap-1.5">
   115→      {days.map((d) => {
   116→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
   117→        const isToday = d.dateStr === today;
   118→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
   119→        return (
   120→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
   121→            <div
   122→              className={cn('w-full aspect-square rounded-md transition-colors', isToday && 'ring-2 ring-primary/30')}
   123→              style={{ backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` }}
   124→            />
   125→            <span className={cn('text-[10px]', isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>{d.label}</span>
   126→            {d.count > 0 && (
   127→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   128→            )}
   129→          </div>
   130→        );
   131→      })}
   132→    </div>
   133→  );
   134→}
