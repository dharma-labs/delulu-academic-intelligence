   125→function signalToStatus(signal: SignalStatus): 'healthy' | 'improving' | 'attention' | 'critical' | 'upcoming' | 'nodata' {
   126→  if (signal === 'nodata') return 'nodata';
   127→  return signal as 'healthy' | 'improving' | 'attention' | 'critical' | 'upcoming';
   128→}
   129→
   130→function signalLabel(signal: SignalStatus): string {
   131→  switch (signal) {
   132→    case 'healthy': return 'Healthy';
   133→    case 'improving': return 'Improving';
   134→    case 'attention': return 'Needs Attention';
   135→    case 'critical': return 'Critical';
   136→    case 'upcoming': return 'Upcoming';
   137→    default: return 'No Data';
   138→  }
   139→}
   140→
   141→function pctColor(pct: number): string {
   142→  if (pct >= 75) return 'text-[var(--delulu-success)]';
   143→  if (pct >= 50) return 'text-[var(--delulu-info)]';
   144→  if (pct >= 30) return 'text-[var(--delulu-warning)]';
   145→  return 'text-[var(--delulu-danger)]';
   146→}
   147→
   148→function formatDuration(seconds: number): string {
   149→  const h = Math.floor(seconds / 3600);
   150→  const m = Math.floor((seconds % 3600) / 60);
   151→  if (h > 0) return `${h}h ${m}m`;
   152→  return `${m}m`;
   153→}

   163→function OverviewTab({ subjectId }: { subjectId: string }) {
   164→  const subjects = useStore((s) => s.subjects);
   165→  const allSyllabusUnits = useStore((s) => s.syllabusUnits);
   166→  const allAssessments = useStore((s) => s.assessments);
   167→  const allAttendance = useStore((s) => s.attendance);
   168→  const allStudySessions = useStore((s) => s.studySessions);
   169→  const allExams = useStore((s) => s.exams);
   170→  const profile = useStore((s) => s.profile);
   171→  const subject = subjects.find((x) => x.id === subjectId);
   172→  const syllabusUnits = useMemo(() => allSyllabusUnits.filter((u) => u.subjectId === subjectId), [allSyllabusUnits, subjectId]);
   173→  const assessments = useMemo(() => allAssessments.filter((a) => a.subjectId === subjectId), [allAssessments, subjectId]);
   174→  const attendance = useMemo(() => allAttendance.filter((a) => a.subjectId === subjectId), [allAttendance, subjectId]);
   175→  const studySessions = useMemo(() => allStudySessions.filter((s) => s.subjectId === subjectId), [allStudySessions, subjectId]);
   176→  const exams = useMemo(() => allExams.filter((e) => e.subjectId === subjectId), [allExams, subjectId]);
   177→
   178→  const progress = getSubjectProgress({ syllabusUnits }, subjectId);
   179→  const att = getSubjectAttendance({ attendance }, subjectId);
   180→  const marks = getSubjectMarks({ assessments }, subjectId);
   181→  const grade = getSubjectGrade({ assessments }, subjectId);
   182→
   183→  const caPct = assessments.length > 0
   184→    ? Math.round(assessments.reduce((sum, a) => sum + (a.maxMarks > 0 ? (a.obtainedMarks / a.maxMarks) * 100 : 0), 0) / assessments.length)
   185→    : 0;
   186→
   187→  const nextExam = exams
   188→    .filter((e) => e.status === 'upcoming')
   189→    .sort((a, b) => a.date.localeCompare(b.date))[0];
   190→
   191→  const recommendation = useMemo(() => {
   192→    const recs: string[] = [];
   193→    if (att.total > 0 && att.percentage < profile.attendanceThreshold) {
   194→      recs.push(`Attendance is below ${profile.attendanceThreshold}% threshold. Focus on attending classes.`);
   195→    }
   196→    if (progress < 50 && progress > 0) {
   197→      recs.push('Syllabus is less than halfway done. Prioritize completing more topics.');
   198→    }
   199→    if (progress === 0) {
   200→      recs.push('No syllabus progress yet. Start by adding topics to your units.');
   201→    }
   202→    if (marks.percentage < 50 && marks.max > 0) {
   203→      recs.push('Assessment scores need improvement. Review weak areas before the next test.');
   204→    }
   205→    if (nextExam) {
   206→      const daysUntil = Math.ceil((parseISO(nextExam.date).getTime() - Date.now()) / 86400000);
   207→      if (daysUntil <= 7) {
   208→        recs.push(`${nextExam.name} is in ${daysUntil} day${daysUntil !== 1 ? 's' : ''}. Intensive revision recommended.`);
   209→      }
   210→    }
   211→    if (recs.length === 0) {
   212→      if (progress >= 80 && marks.percentage >= 70) {
   213→        recs.push('Great progress! Keep up the consistency and start revising for exams.');
   214→      } else {
   215→        recs.push('Add more data (attendance, marks, study sessions) to get personalized recommendations.');
   216→      }
   217→    }
   218→    return recs[0];
   219→  }, [att, progress, marks, nextExam, profile.attendanceThreshold]);
   220→
   221→  const recentActivity = useMemo(() => {
   222→    const items: { type: string; label: string; date: string }[] = [];
   223→    studySessions.forEach((s) =>
   224→      items.push({ type: 'session', label: `Studied for ${formatDuration(s.duration)}`, date: s.date })
   225→    );
   226→    assessments.forEach((a) =>
   227→      items.push({ type: 'assessment', label: `${a.name}: ${a.obtainedMarks}/${a.maxMarks}`, date: a.date })
   228→    );
   229→    return items.sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5);
   230→  }, [studySessions, assessments]);
   231→
   232→  // Determine recommendation type for insight card
   233→  const recType = useMemo(() => {
   234→    if (att.total > 0 && att.percentage < profile.attendanceThreshold) return 'critical' as const;
   235→    if (marks.percentage < 50 && marks.max > 0) return 'warning' as const;
   236→    if (nextExam && Math.ceil((parseISO(nextExam.date).getTime() - Date.now()) / 86400000) <= 7) return 'warning' as const;
   237→    if (progress >= 80 && marks.percentage >= 70) return 'positive' as const;
   238→    return 'info' as const;
   239→  }, [att, progress, marks, nextExam, profile.attendanceThreshold]);
   240→
   241→  if (!subject) return null;
   242→
   243→  return (
   244→    <div className="space-y-5">
   245→      {/* 4 metric cards */}
   246→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   247→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   248→          <MetricCard
   249→            label="Current Grade"
   250→            value={marks.max > 0 ? grade : '--'}
   251→            context={marks.max > 0 ? `${marks.percentage}% average` : 'No assessments yet'}
   252→            icon={GraduationCap}
   253→            valueColor={marks.max > 0 ? pctColor(marks.percentage) : 'text-muted-foreground'}
   254→          />
   255→        </motion.div>
   256→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   257→          <MetricCard
   258→            label="Attendance"
   259→            value={att.total > 0 ? `${att.percentage}%` : '--'}
   260→            context={att.total > 0 ? `${att.present}/${att.total} classes` : 'No records yet'}
   261→            icon={ClipboardCheck}
   262→            valueColor={att.total > 0
   263→              ? att.percentage >= profile.attendanceThreshold
   264→                ? 'text-[var(--delulu-success)]'
   265→                : 'text-[var(--delulu-danger)]'
   266→              : 'text-muted-foreground'}
   267→          />
   268→        </motion.div>
   269→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   270→          <MetricCard
   271→            label="Syllabus"
   272→            value={`${progress}%`}
   273→            context={progress > 0 ? 'topics completed' : 'No topics yet'}
   274→            icon={BookOpen}
   275→            valueColor={progress >= 75 ? 'text-[var(--delulu-success)]' : progress >= 40 ? 'text-[var(--delulu-info)]' : 'text-muted-foreground'}
   276→          />
   277→        </motion.div>
   278→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   279→          <MetricCard
   280→            label="CA Performance"
   281→            value={caPct > 0 ? `${caPct}%` : '--'}
   282→            context={caPct > 0 ? `across ${assessments.length} assessment${assessments.length !== 1 ? 's' : ''}` : 'No assessments yet'}
   283→            icon={BarChart3}
   284→            valueColor={caPct > 0 ? pctColor(caPct) : 'text-muted-foreground'}
   285→          />
   286→        </motion.div>
   287→      </div>
   288→
   289→      {/* Next Assessment + Recommendation */}
   290→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   291→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   292→          <div className="metric-card">
   293→            <div className="flex items-center gap-2 mb-3">
   294→              <CalendarDays className="size-4 text-muted-foreground" />
   295→              <span className="section-label">Next Assessment</span>
   296→            </div>
   297→            {nextExam ? (
   298→              <div>
   299→                <p className="font-medium text-sm">{nextExam.name}</p>
   300→                <p className="text-xs text-muted-foreground mt-1">
   301→                  {format(parseISO(nextExam.date), 'dd MMM yyyy')}
   302→                </p>
   303→                <Badge variant="secondary" className="mt-2 text-[10px] capitalize font-medium">{nextExam.type}</Badge>
   304→              </div>
   305→            ) : (
   306→              <p className="text-sm text-muted-foreground">No upcoming assessments</p>
   307→            )}
   308→          </div>
   309→        </motion.div>
   310→
   311→        <motion.div variants={fadeUp} initial="hidden" animate="show">
   312→          <InsightCard
   313→            type={recType}
   314→            icon={Target}
   315→            title="Recommended Action"
   316→            description={recommendation}
   317→          />
   318→        </motion.div>
   319→      </div>
   320→
   321→      {/* Recent Activity */}
   322→      <motion.div variants={fadeUp} initial="hidden" animate="show">
   323→        <div className="metric-card">
   324→          <div className="flex items-center justify-between mb-3">
   325→            <span className="section-label">Recent Activity</span>
   326→          </div>
   327→          {recentActivity.length > 0 ? (
   328→            <div className="space-y-2.5 max-h-48 overflow-y-auto scrollbar-thin">
   329→              {recentActivity.map((item, i) => (
   330→                <div key={i} className="flex items-center gap-3">
   331→                  <div className="h-7 w-7 rounded-lg bg-secondary flex items-center justify-center shrink-0">
   332→                    {item.type === 'session' ? (
   333→                      <Timer className="size-3.5 text-muted-foreground" />
   334→                    ) : (
   335→                      <BarChart3 className="size-3.5 text-muted-foreground" />
   336→                    )}
   337→                  </div>
   338→                  <div className="min-w-0 flex-1">
   339→                    <p className="text-sm truncate">{item.label}</p>
   340→                    <p className="text-xs text-muted-foreground">{item.date}</p>
   341→                  </div>
   342→                </div>
   343→              ))}
   344→            </div>
   345→          ) : (
   346→            <p className="text-sm text-muted-foreground">No recent activity recorded</p>
   347→          )}
   348→        </div>
   349→      </motion.div>
   350→    </div>
   351→  );
   352→}

   357→function SyllabusTab({ subjectId }: { subjectId: string }) {
   358→  const allUnits = useStore((s) => s.syllabusUnits);
   359→  const units = useMemo(() => allUnits.filter((u) => u.subjectId === subjectId).sort((a, b) => a.order - b.order), [allUnits, subjectId]);
   360→  const addSyllabusUnit = useStore((s) => s.addSyllabusUnit);
   361→  const updateSyllabusUnit = useStore((s) => s.updateSyllabusUnit);
   362→  const deleteSyllabusUnit = useStore((s) => s.deleteSyllabusUnit);
   363→  const addSyllabusTopic = useStore((s) => s.addSyllabusTopic);
   364→  const updateSyllabusTopic = useStore((s) => s.updateSyllabusTopic);
   365→  const deleteSyllabusTopic = useStore((s) => s.deleteSyllabusTopic);
   366→  const toggleTopicComplete = useStore((s) => s.toggleTopicComplete);
   367→
   368→  const [newTopicMap, setNewTopicMap] = useState<Record<string, string>>({});
   369→  const [editingUnitId, setEditingUnitId] = useState<string | null>(null);
   370→  const [editingTopicId, setEditingTopicId] = useState<string | null>(null);
   371→  const [editValue, setEditValue] = useState('');
   372→  const [showAddUnit, setShowAddUnit] = useState(false);
   373→  const [newUnitName, setNewUnitName] = useState('');
   374→
   375→  const totalTopics = units.reduce((sum, u) => sum + u.topics.length, 0);
   376→  const completedTopics = units.reduce((sum, u) => sum + u.topics.filter((t) => t.completed).length, 0);
   377→  const overallPct = totalTopics > 0 ? Math.round((completedTopics / totalTopics) * 100) : 0;
   378→
   379→  const handleAddTopic = (unitId: string) => {
   380→    const name = newTopicMap[unitId]?.trim();
   381→    if (!name) return;
   382→    const unit = units.find((u) => u.id === unitId);
   383→    addSyllabusTopic({
   384→      unitId,
   385→      subjectId,
   386→      name,
   387→      completed: false,
   388→      order: unit ? unit.topics.length + 1 : 1,
   389→    });
   390→    setNewTopicMap((prev) => ({ ...prev, [unitId]: '' }));
   391→  };
   392→
   393→  const handleAddUnit = () => {
   394→    const name = newUnitName.trim();
   395→    if (!name) return;
   396→    addSyllabusUnit({
   397→      subjectId,
   398→      name,
   399→      order: units.length + 1,
   400→    });
   401→    setNewUnitName('');
   402→    setShowAddUnit(false);
   403→  };
   404→
   405→  const startEditUnit = (unit: SyllabusUnit) => {
   406→    setEditingUnitId(unit.id);
   407→    setEditValue(unit.name);
   408→  };
   409→
   410→  const saveEditUnit = () => {
   411→    if (editingUnitId && editValue.trim()) {
   412→      updateSyllabusUnit(editingUnitId, { name: editValue.trim() });
   413→    }
   414→    setEditingUnitId(null);
   415→    setEditValue('');
   416→  };
   417→
   418→  const startEditTopic = (topic: SyllabusTopic) => {
   419→    setEditingTopicId(topic.id);
   420→    setEditValue(topic.name);
   421→  };
   422→
   423→  const saveEditTopic = () => {
   424→    if (editingTopicId && editValue.trim()) {
   425→      updateSyllabusTopic(editingTopicId, { name: editValue.trim() });
   426→    }
   427→    setEditingTopicId(null);
   428→    setEditValue('');
   429→  };
   430→
   431→  return (
   432→    <div className="space-y-4">
   433→      {/* Overall progress */}
   434→      {totalTopics > 0 && (
   435→        <div className="metric-card">
   436→          <CompactProgress
   437→            label="Overall Progress"
   438→            value={overallPct}
   439→            displayValue={`${completedTopics}/${totalTopics} topics (${overallPct}%)`}
   440→          />
   441→        </div>
   442→      )}
   443→
   444→      {/* Units */}
   445→      {units.length === 0 && !showAddUnit ? (
   446→        <EmptyState
   447→          icon={BookOpen}
   448→          title="No syllabus units yet"
   449→          description="Add units to start tracking your syllabus progress"
   450→          action={
   451→            <Button size="sm" onClick={() => setShowAddUnit(true)}>
   452→              <Plus className="h-4 w-4 mr-1" /> Add Unit
   453→            </Button>
   454→          }
   455→        />
   456→      ) : (
   457→        <Accordion type="multiple" className="space-y-2">
   458→          {units.map((unit) => {
   459→            const completed = unit.topics.filter((t) => t.completed).length;
   460→            const total = unit.topics.length;
   461→            const pct = total > 0 ? Math.round((completed / total) * 100) : 0;
   462→
   463→            return (
   464→              <AccordionItem key={unit.id} value={unit.id} className="border rounded-lg px-0">
   465→                <AccordionTrigger className="px-4 py-3 hover:no-underline hover:bg-secondary/50 rounded-t-lg">
   466→                  <div className="flex items-center gap-3 flex-1 text-left">
   467→                    {editingUnitId === unit.id ? (
   468→                      <div className="flex items-center gap-2 flex-1">
   469→                        <Input
   470→                          value={editValue}
   471→                          onChange={(e) => setEditValue(e.target.value)}
   472→                          onKeyDown={(e) => {
   473→                            if (e.key === 'Enter') saveEditUnit();
   474→                            if (e.key === 'Escape') setEditingUnitId(null);
   475→                          }}
   476→                          className="h-8 text-sm"
   477→                          autoFocus
   478→                          onClick={(e) => e.stopPropagation()}
   479→                        />
   480→                        <Button size="icon" variant="ghost" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); saveEditUnit(); }}>
   481→                          <Check className="size-3.5" />
   482→                        </Button>
   483→                      </div>
   484→                    ) : (
   485→                      <>
   486→                        <span className="font-medium text-sm flex-1" onDoubleClick={(e) => { e.stopPropagation(); startEditUnit(unit); }}>{unit.name}</span>
   487→                        <span className="text-xs text-muted-foreground tabular-nums">{completed}/{total}</span>
   488→                        <div className="w-16">
   489→                          <div className="progress-thin">
   490→                            <div
   491→                              className={pct >= 75 ? 'bg-emerald-500 dark:bg-emerald-400' : pct >= 40 ? 'bg-blue-500 dark:bg-blue-400' : 'bg-amber-500 dark:bg-amber-400'}
   492→                              style={{ width: `${pct}%` }}
   493→                            />
   494→                          </div>
   495→                        </div>
   496→                      </>
   497→                    )}
   498→                  </div>
   499→                </AccordionTrigger>
   500→                <AccordionContent className="px-4 pb-3">
   501→                  <div className="space-y-1">
   502→                    {unit.topics.map((topic) => (
   503→                      <div
   504→                        key={topic.id}
   505→                        className="flex items-center gap-2 group py-1.5 px-2 rounded-md hover:bg-secondary/50"
   506→                      >
   507→                        <Checkbox
   508→                          checked={topic.completed}
   509→                          onCheckedChange={() => toggleTopicComplete(topic.id)}
   510→                          className="shrink-0"
   511→                        />
   512→
   513→                        {editingTopicId === topic.id ? (
   514→                          <div className="flex items-center gap-2 flex-1">
   515→                            <Input
   516→                              value={editValue}
   517→                              onChange={(e) => setEditValue(e.target.value)}
   518→                              onKeyDown={(e) => {
   519→                                if (e.key === 'Enter') saveEditTopic();
   520→                                if (e.key === 'Escape') setEditingTopicId(null);
   521→                              }}

   939→function AttendanceTab({ subjectId }: { subjectId: string }) {
   940→  const allRecords = useStore((s) => s.attendance);
   941→  const records = useMemo(() => allRecords.filter((a) => a.subjectId === subjectId), [allRecords, subjectId]);
   942→  const profile = useStore((s) => s.profile);
   943→  const addAttendance = useStore((s) => s.addAttendance);
   944→  const deleteAttendance = useStore((s) => s.deleteAttendance);
   945→
   946→  const att = getSubjectAttendance({ attendance: records }, subjectId);
   947→  const today = new Date().toISOString().split('T')[0];
   948→  const todayRecord = records.find((r) => r.date === today);
   949→
   950→  const status = att.total === 0
   951→    ? 'NO DATA'
   952→    : att.percentage >= profile.attendanceThreshold
   953→      ? 'SAFE'
   954→      : att.percentage >= profile.attendanceThreshold - 10
   955→        ? 'WATCH'
   956→        : 'RISK';
   957→
   958→  const statusColor =
   959→    status === 'SAFE' ? 'text-[var(--delulu-success)]' :
   960→    status === 'WATCH' ? 'text-[var(--delulu-warning)]' :
   961→    status === 'RISK' ? 'text-[var(--delulu-danger)]' :
   962→    'text-muted-foreground';
   963→
   964→  const canMiss = useMemo(() => {
   965→    if (att.total === 0) return 'N/A';
   966→    const threshold = profile.attendanceThreshold;
   967→    const maxMissable = Math.floor((att.present * 100) / threshold) - att.total;
   968→    return Math.max(0, maxMissable);
   969→  }, [att, profile.attendanceThreshold]);
   970→
   971→  const handleQuickMark = (present: boolean) => {
   972→    if (todayRecord) return;
   973→    addAttendance({
   974→      subjectId,
   975→      date: today,
   976→      present,
   977→      totalClasses: 1,
   978→    });
   979→  };
   980→
   981→  return (
   982→    <div className="space-y-5">
   983→      {/* Stats metric cards */}
   984→      <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   985→        <MetricCard
   986→          label="Present"
   987→          value={att.present}
   988→          valueColor="text-[var(--delulu-success)]"
   989→          icon={CheckCircle2}
   990→        />
   991→        <MetricCard
   992→          label="Absent"
   993→          value={att.total - att.present}
   994→          valueColor="text-[var(--delulu-danger)]"
   995→          icon={X}
   996→        />
   997→        <MetricCard
   998→          label="Total"
   999→          value={att.total}
  1000→          icon={ClipboardCheck}
  1001→        />
  1002→        <MetricCard
  1003→          label="Percentage"
  1004→          value={`${att.percentage}%`}
  1005→          valueColor={statusColor}
  1006→          icon={BarChart3}
  1007→        />
  1008→        <div className="metric-card col-span-2 md:col-span-1">
  1009→          <span className="metric-label">Status</span>
  1010→          <p className={`text-lg font-bold mt-2 ${statusColor}`}>{status}</p>
  1011→        </div>
  1012→      </div>
  1013→
  1014→      {/* Can miss insight */}
  1015→      {typeof canMiss === 'number' ? (
  1016→        <InsightCard
  1017→          type={canMiss > 0 ? (canMiss <= 2 ? 'warning' : 'positive') : 'critical'}
  1018→          icon={AlertTriangle}
  1019→          title={canMiss > 0 ? `You can miss ${canMiss} more class${canMiss !== 1 ? 'es' : ''}` : 'Attendance at threshold'}
  1020→          description={canMiss > 0
  1021→            ? `Before falling below the ${profile.attendanceThreshold}% threshold.`
  1022→            : 'You cannot miss any more classes. Your attendance is already at or below the threshold.'}
  1023→        />
  1024→      ) : (
