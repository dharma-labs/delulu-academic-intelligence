    43→// -- Health status --
    44→function getHealthConfig(score: number) {
    45→   if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    46→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    47→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    48→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    49→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    50→}

    52→function formatStudyTime(seconds: number): string {
    53→  const totalMinutes = Math.floor(seconds / 60);
    54→  const h = Math.floor(totalMinutes / 60);
    55→  const m = totalMinutes % 60;
    56→  if (h === 0) return `${m}m`;
    57→  if (m === 0) return `${h}h`;
    58→  return `${h}h ${m}m`;
    59→}

    94→// -- Weekly Mini Chart (mobile sparkline) --
    95→function WeeklyMiniChart({ studySessions }: { studySessions: { date: string; duration: number }[] }) {
    96→  const bars = useMemo(() => {
    97→    const days: { label: string; minutes: number; isToday: boolean }[] = [];
    98→    for (let i = 6; i >= 0; i--) {
    99→      const d = new Date();
   100→      d.setDate(d.getDate() - i);
   101→      const dateStr = d.toISOString().split('T')[0];
   102→      const dayMinutes = Math.floor(studySessions
   103→        .filter(s => s.date === dateStr)
   104→        .reduce((sum, s) => sum + s.duration / 60, 0));
   105→      days.push({
   106→        label: format(d, 'EEE').charAt(0),
   107→        minutes: dayMinutes,
   108→        isToday: i === 0,
   109→      });
   110→    }
   111→    return days;
   112→  }, [studySessions]);
   113→
   114→  const maxMin = Math.max(...bars.map(b => b.minutes), 1);
   115→
   116→  return (
   117→    <div className="flex items-end gap-1.5 h-12">
   118→      {bars.map((b, i) => (
   119→        <div key={i} className="flex-1 flex flex-col items-center gap-1">
   120→          <div className="w-full relative" style={{ height: '32px' }}>
   121→            <div
   122→              className={cn(
   123→                'absolute bottom-0 left-1/2 -translate-x-1/2 w-3/4 rounded-t-sm transition-all duration-500',
   124→                b.isToday ? 'bg-primary' : 'bg-primary/25',
   125→              )}
   126→              style={{ height: `${Math.max(4, (b.minutes / maxMin) * 100)}%` }}
   127→            />
   128→          </div>
   129→          <span className={cn('text-[8px] tabular-nums', b.isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>
   130→            {b.label}
   131→          </span>
   132→        </div>
   133→      ))}
   134→    </div>
   135→  );
   136→}

   141→// ─── Weekly Heatmap ──────────────────────────────────────────────
   142→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
   143→  const days = useMemo(() => {
   144→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
   145→    return Array.from({ length: 7 }, (_, i) => {
   146→      const d = addDays(start, i);
   147→      const dateStr = d.toISOString().split('T')[0];
   148→      const daySessions = studySessions.filter((s) => s.date === dateStr);
   149→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
   150→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
   151→    });
   152→  }, [studySessions]);
   153→
   154→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
   155→  const today = new Date().toISOString().split('T')[0];
   156→
   157→  return (
   158→    <div className="grid grid-cols-7 gap-1.5">
   159→      {days.map((d) => {
   160→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
   161→        const isToday = d.dateStr === today;
   162→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
   163→        return (
   164→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
   165→            <div
   166→              className={cn('w-full aspect-square rounded-md transition-colors', isToday && 'ring-2 ring-primary/30')}
   167→              style={{ backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` }}
   168→            />
   169→            <span className={cn('text-[10px]', isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>{d.label}</span>
   170→            {d.count > 0 && (
   171→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   172→            )}
   173→          </div>
   174→        );
   175→      })}
   176→    </div>
   177→  );
   178→}
