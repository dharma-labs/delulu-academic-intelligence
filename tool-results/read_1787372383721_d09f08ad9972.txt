    68→    68→function formatElapsed(totalSeconds: number): string {
    69→    69→  const m = Math.floor(totalSeconds / 60);

    71→    71→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    72→    72→}
    73→    73→
    74→    74→function formatDuration(totalSeconds: number): string {
    75→    75→  if (totalSeconds < 60) return `${totalSeconds}s`;
    76→    76→  const m = Math.floor(totalSeconds / 60);
    77→    77→  const s = totalSeconds % 60;
    78→    78→  if (m < 60) return s > 0 ? `${m}m ${s}s` : `${m}m`;
    79→    79→  const h = Math.floor(m / 60);
    80→    80→  const rm = m % 60;
    81→    81→  return rm > 0 ? `${h}h ${rm}m` : `${h}h`;
    82→    82→}
    83→    83→
    84→    84→function todayDateStr(): string {
    85→    85→  return new Date().toISOString().split('T')[0];
