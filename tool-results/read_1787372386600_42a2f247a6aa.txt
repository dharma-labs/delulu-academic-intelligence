    67→// --- Helpers ---
    68→function formatElapsed(totalSeconds: number): string {
    69→  const m = Math.floor(totalSeconds / 60);
    70→  const s = totalSeconds % 60;
    71→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    72→}
    73→
    74→function formatDuration(totalSeconds: number): string {
    75→  if (totalSeconds < 60) return `${totalSeconds}s`;
    76→  const m = Math.floor(totalSeconds / 60);
    77→  const s = totalSeconds % 60;
    78→  if (m < 60) return s > 0 ? `${m}m ${s}s` : `${m}m`;
    79→  const h = Math.floor(m / 60);
    80→  const rm = m % 60;
    81→  return rm > 0 ? `${h}h ${rm}m` : `${h}h`;
    82→}
    83→
    84→function todayDateStr(): string {
    85→  return new Date().toISOString().split('T')[0];
    86→}

   169→
   170→  // --- Timer logic ---

   175→        useStore.setState({ focusElapsed: elapsedRef.current });

   183→          const topicName = selectedTopic?.name || '';

   186→            setTimerPulse(true);

   191→              setCompletionTopicName(topicName);
   192→              setCompletionNotes('');
   193→              setCompletedNaturally(true);
   194→              setPhase('completion');
   195→              setTimerPulse(false);
   196→            }, 500);

   198→        }
   199→      }, 1000);
   200→    }

   202→      if (intervalRef.current) clearInterval(intervalRef.current);

   255→    setIsPaused(false);
   256→    setPhase('active');
   257→    setCompletedNaturally(false);
   258→    setTimerPulse(false);
   259→    completionToastShownRef.current = false;
   260→    naturalCompletionTriggeredRef.current = false;

   275→    stopFocus();
   276→    setCompletionDuration(duration);
   277→    setCompletionTopicName(selectedTopic?.name || '');
   278→    setCompletionNotes('');
   279→    setCompletedNaturally(false);
   280→    setPhase('completion');

   287→      topicName: selectedTopic?.name,
   288→      duration: completionDuration,
   289→      date: todayDateStr(),
   290→      type: 'focus',

   292→    });

   294→    completionToastShownRef.current = false;
   295→    naturalCompletionTriggeredRef.current = false;
   296→    setPhase('setup');
   297→    setSelectedSubjectId('');
   298→    setSelectedTopicId('');
   299→    setGoalMinutes('');
   300→    setCompletionNotes('');

   303→    selectedSubjectId, selectedTopicId, selectedTopic,

   305→  ]);

   309→    completionToastShownRef.current = false;
   310→    naturalCompletionTriggeredRef.current = false;
   311→    setPhase('setup');
   312→    setSelectedSubjectId('');
   313→    setSelectedTopicId('');

   315→    setCompletionNotes('');
   316→    setIsPaused(false);

   330→        handleStopClick();

   332→    };
   333→    window.addEventListener('keydown', handleKeyDown);
   334→    return () => window.removeEventListener('keydown', handleKeyDown);
   335→  }, [phase, handlePauseResume, handleStopClick]);
   336→

   351→              exit={{ opacity: 0, y: -8 }}
   352→              transition={{ duration: 0.2 }}

   354→            >
   355→              {/* Compact inline session start */}
   356→              <div className='flex flex-wrap gap-2 items-end'>
   357→                <div className='flex-1 min-w-[120px]'>
   358→                  <span className='section-label text-[10px]'>Subject</span>
   359→                  <Select
   360→                    value={selectedSubjectId}
   361→                    onValueChange={(v) => {
   362→                      setSelectedSubjectId(v);
   363→                      setSelectedTopicId('');
   364→                    }}
   365→                  >
   366→                    <SelectTrigger className='h-8 text-xs'>
   367→                      <SelectValue placeholder='Subject' />
   368→                    </SelectTrigger>
   369→                    <SelectContent>
   370→                      {activeSubjects.map((s) => (
   371→                        <SelectItem key={s.id} value={s.id}>
   372→                          <span className='flex items-center gap-1.5'>
   373→                            <span className='status-dot' style={{ backgroundColor: s.color }} />
   374→                            {s.name}
   375→                          </span>
   376→                        </SelectItem>

   379→                  </Select>

   384→                    <Select value={selectedTopicId} onValueChange={setSelectedTopicId}>

   411→                  size='sm'
   412→                  disabled={!selectedSubjectId}
   413→                  onClick={handleStart}

   423→                <div>

   425→                    <span className='section-label'>Today</span>
   426→                    <span className='text-[10px] text-muted-foreground tabular-nums'>{todayTotalMinutes}m</span>
   427→                  </div>
   428→                  <div className='flex gap-2 overflow-x-auto pb-1 -mx-1 px-1 scrollbar-none snap-x snap-mandatory'>
   429→                    {todaySessions.slice(0, 6).map((session) => {
   430→                      const subj = subjects.find((s) => s.id === session.subjectId);
   431→                      return (
   432→                        <motion.div
   433→                          key={session.id}
   434→                          initial={{ opacity: 0, scale: 0.95 }}
   435→                          animate={{ opacity: 1, scale: 1 }}

   438→                          <div className='flex items-center gap-1.5 mb-1.5'>
   439→                            <span

   441→                              style={{ backgroundColor: subj?.color || 'var(--muted-foreground)' }}
   442→                            />
   443→                            <span className='text-[11px] font-medium truncate'>
   444→                              {subj?.name || 'Unknown'}
   445→                            </span>
   446→                          </div>

   448→                            {formatDuration(session.duration)}

   454→                </div>

   457→              {todaySessions.length === 0 && (
   458→                <EmptyState
   459→                  icon={Clock}
   460→                  title='No sessions today'
   461→                  description='Tap Start above to begin your first focus session.'
   462→                />
   463→              )}
   464→            </motion.div>
   465→          )}
   466→
   467→          {/* --- Mobile: Active Timer Phase --- */}
   468→          {phase === 'active' && focusActive && (
   469→            <motion.div
   470→              key='mobile-active'
   471→              initial={{ opacity: 0, scale: 0.97 }}
   472→              animate={{ opacity: 1, scale: 1 }}
   473→              exit={{ opacity: 0, scale: 0.97 }}
   474→              transition={{ duration: 0.3, ease: 'easeOut' }}
   475→              className='flex flex-col items-center gap-4'
   476→            >
   477→              {/* Timer digits centered */}
   478→              <div className='text-center'>

   480→                  key={focusElapsed}

   482→                    'text-5xl font-bold tracking-tighter leading-none tabular-nums',

   485→                  initial={{ scale: 1.01 }}
   486→                  animate={{ scale: 1 }}
   487→                  transition={{ duration: 0.1 }}
   488→                >

   494→                    'inline-flex items-center px-1.5 py-0.5 rounded-md text-[10px] font-semibold tracking-wider uppercase',

   498→                  )}>
   499→                    {isPaused ? 'PAUSED' : 'RUNNING'}
   500→                  </span>
   501→                  {todaySessions.length > 0 && (
   502→                    <span className='text-[10px] text-muted-foreground'>

   505→                  )}
   506→                </div>
   507→              </div>
   508→
   509→              {/* Timer Ring — full width, larger */}
   510→              <div className={cn('relative flex items-center justify-center w-full max-w-[300px]', timerPulse && 'timer-completion-pulse')}>
   511→                <svg
   512→                  width={MOBILE_TIMER_RADIUS * 2 + 24}

   514→                  className={cn('transform -rotate-90', !isPaused && 'timer-glow')}
   515→                >
   516→                  <circle
   517→                    cx={MOBILE_TIMER_RADIUS + 12}
   518→                    cy={MOBILE_TIMER_RADIUS + 12}
   519→                    r={MOBILE_TIMER_RADIUS}
   520→                    fill='none'
   521→                    stroke='var(--border)'
   522→                    strokeWidth='4'
   523→                  />
   524→                  {goalSeconds > 0 && (
   525→                    <motion.circle
   526→                      cx={MOBILE_TIMER_RADIUS + 12}
   527→                      cy={MOBILE_TIMER_RADIUS + 12}
   528→                      r={MOBILE_TIMER_RADIUS}
   529→                      fill='none'
   530→                      stroke='var(--color-primary)'
   531→                      strokeWidth='4'
   532→                      strokeLinecap='round'
   533→                      strokeDasharray={MOBILE_TIMER_CIRCUMFERENCE}
   534→                      strokeDashoffset={mobileStrokeDashoffset}
   535→                      transition={{ duration: 0.5, ease: 'easeOut' }}
   536→                    />
   537→                  )}
   538→                  {goalSeconds === 0 && (
   539→                    <circle
   540→                      cx={MOBILE_TIMER_RADIUS + 12}
   541→                      cy={MOBILE_TIMER_RADIUS + 12}
   542→                      r={MOBILE_TIMER_RADIUS - 8}
   543→                      fill='none'
   544→                      stroke='var(--border)'
   545→                      strokeWidth='1'
   546→                      strokeDasharray='4 8'
   547→                      opacity='0.5'
   548→                    />
   549→                  )}
   550→                </svg>
   551→              </div>
   552→
   553→              {/* Topic display */}
   554→              {(selectedSubject || selectedTopic) && (

   560→
   561→              {/* Controls — large circular buttons */}
   562→              <div className='flex items-center gap-6 mt-2'>
