    61→// ─── Notification badges hook ──────────────────────────────────────
    62→
    63→function useNavBadges() {
    64→  const tasks = useStore((s) => s.tasks);
    65→  const revisionItems = useStore((s) => s.revisionItems);
    66→  const exams = useStore((s) => s.exams);
    67→  const assignments = useStore((s) => s.assignments);
    68→
    69→  return useMemo(() => {
    70→    const badges: Partial<Record<ViewId, number>> = {};
    71→
    72→    // Overdue tasks
    73→    const today = new Date().toISOString().split('T')[0];
    74→    const overdueTasks = tasks.filter(t => !t.completed && t.dueDate && t.dueDate < today);
    75→    if (overdueTasks.length > 0) badges['tasks'] = overdueTasks.length;
    76→
    77→    // Due revisions
    78→    const dueRevisions = revisionItems.filter(r => {
    79→      const nextReview = new Date(r.nextReview);
    80→      return nextReview <= new Date();
    81→    });
    82→    if (dueRevisions.length > 0) badges['revision'] = dueRevisions.length;
    83→
    84→    // Upcoming exams (within 7 days)
    85→    const weekFromNow = new Date();
    86→    weekFromNow.setDate(weekFromNow.getDate() + 7);
    87→    const upcomingExams = exams.filter(e => e.status === 'upcoming' && e.date >= today && e.date <= weekFromNow.toISOString().split('T')[0]);
    88→    if (upcomingExams.length > 0) badges['exams'] = upcomingExams.length;
    89→
    90→    // Upcoming assignments (within 3 days)
    91→    const threeDays = new Date();
    92→    threeDays.setDate(threeDays.getDate() + 3);
    93→    const upcomingAssign = assignments.filter(a => a.status !== 'completed' && a.deadline >= today && a.deadline <= threeDays.toISOString().split('T')[0]);
    94→    if (upcomingAssign.length > 0) badges['assignments'] = upcomingAssign.length;
    95→
    96→    return badges;
    97→  }, [tasks, revisionItems, exams, assignments]);
    98→}

   109→  );

   715→function MobileFAB() {
   716→  const [open, setOpen] = useState(false);
   717→  const navigate = useStore((s) => s.navigate);
   718→  const currentView = useStore((s) => s.currentView);
   719→  const fabRef = useRef<HTMLDivElement>(null);
   720→
   721→  // Auto-close when navigating away (store currentView is external state)
   722→  const prevViewRef = useRef(currentView);
   723→  /* eslint-disable react-hooks/set-state-in-effect */
   724→  useEffect(() => {
   725→    if (prevViewRef.current !== currentView) {
   726→      prevViewRef.current = currentView;
   727→      setOpen(false);
   728→    }
   729→  }, [currentView]);
   730→  /* eslint-enable react-hooks/set-state-in-effect */
   731→
   732→  const handleClose = useCallback(() => setOpen(false), []);
   733→
   734→  // Close on outside click
   735→  useEffect(() => {
   736→    if (!open) return;
   737→    const handleClick = (e: MouseEvent) => {
   738→      if (fabRef.current && !fabRef.current.contains(e.target as Node)) {
   739→        setOpen(false);
   740→      }
   741→    };
   742→    // Delay to avoid immediate close from the toggle click
   743→    const timer = setTimeout(() => {
   744→      document.addEventListener('mousedown', handleClick);
   745→    }, 0);
   746→    return () => {
   747→      clearTimeout(timer);
   748→      document.removeEventListener('mousedown', handleClick);
   749→    };
   750→  }, [open]);
   751→
   752→  const handleAction = (view: ViewId) => {
   753→    setOpen(false);
   754→    navigate(view);
   755→  };
   756→
   757→  return (
   758→    <div ref={fabRef} className="fixed bottom-[76px] right-4 z-30 md:hidden">
   759→      {/* Backdrop overlay when open */}
   760→      <AnimatePresence>
   761→        {open && (
   762→          <motion.div
   763→            initial={{ opacity: 0 }}
   764→            animate={{ opacity: 1 }}
   765→            exit={{ opacity: 0 }}
   766→            transition={{ duration: 0.15 }}
   767→            className="fixed inset-0 z-[-1]"
   768→            onClick={() => setOpen(false)}
   769→          />
   770→        )}
   771→      </AnimatePresence>
   772→
   773→      {/* Radial menu items */}
   774→      <AnimatePresence>
   775→        {open && (
   776→          <div className="absolute bottom-16 right-0 flex flex-col items-end gap-2.5">
   777→            {FAB_ACTIONS.map((action, i) => {
   778→              const Icon = action.icon;
   779→              return (
   780→                <motion.div
   781→                  key={action.label}
   782→                  initial={{ opacity: 0, scale: 0.5, y: 10 }}
   783→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   784→                  exit={{ opacity: 0, scale: 0.5, y: 10 }}
   785→                  transition={{ duration: 0.2, delay: i * 0.04 }}
   786→                  className="flex items-center gap-2.5"
   787→                >
   788→                  <span className="text-xs font-medium text-foreground whitespace-nowrap bg-card/90 backdrop-blur-sm border border-border rounded-lg px-2.5 py-1.5 shadow-sm">
   789→                    {action.label}
   790→                  </span>
   791→                  <button
   792→                    onClick={() => handleAction(action.view)}
   793→                    className={cn(
   794→                      'h-10 w-10 rounded-full flex items-center justify-center shadow-lg press-scale transition-colors',
   795→                      action.color,
   796→                      'text-white'
   797→                    )}
   798→                    aria-label={action.label}
   799→                  >
   800→                    <Icon className="size-4" />
   801→                  </button>
   802→                </motion.div>
   803→              );
   804→            })}
   805→          </div>
   806→        )}
   807→      </AnimatePresence>
   808→
   809→      {/* Main FAB button */}
   810→      <motion.button
   811→        onClick={() => setOpen((prev) => !prev)}
   812→        whileTap={{ scale: 0.92 }}
   813→        className="h-12 w-12 rounded-full bg-primary text-primary-foreground fab-shadow press-scale flex items-center justify-center"
   814→        aria-label={open ? 'Close quick actions' : 'Open quick actions'}
   815→        style={{ rotate: open ? 45 : 0 }}
   816→        transition={{ duration: 0.2 }}
   817→      >
   818→        <Plus className="size-5" />
   819→      </motion.button>
   820→    </div>
   821→  );
   822→}

   826→export function AppShell() {
   827→  const subjects = useStore((s) => s.subjects);
   828→  const setCommandOpen = useStore((s) => s.setCommandOpen);
   829→  const [moreSheetOpen, setMoreSheetOpen] = useState(false);
   830→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   831→  const { theme, setTheme } = useTheme();
   832→
   833→  const showShortcuts = useCallback(() => {
   834→    setCommandOpen(false);
   835→    setShortcutsOpen(true);
   836→  }, [setCommandOpen]);
   837→
   838→  useEffect(() => {
   839→    const NUMBER_KEY_VIEWS: Record<string, ViewId> = {
   840→      '1': 'dashboard',
   841→      '2': 'subjects',
   842→      '3': 'marks',
   843→      '4': 'attendance',
   844→      '5': 'exams',
   845→      '6': 'focus',
   846→      '7': 'revision',
   847→      '8': 'notes',
   848→      '9': 'tasks',
   849→    };
   850→
   851→    const handleKeyDown = (e: KeyboardEvent) => {
   852→      // Don't capture when typing in inputs
   853→      const tag = (e.target as HTMLElement)?.tagName;
   854→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   855→      const hasModifier = e.metaKey || e.ctrlKey || e.altKey;
   856→
   857→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   858→        e.preventDefault();
   859→        setCommandOpen(true);
   860→      }
   861→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'd' || e.key === 'D')) {
   862→        e.preventDefault();
   863→        setTheme(theme === 'dark' ? 'light' : 'dark');
   864→      }
   865→      if ((e.metaKey || e.ctrlKey) && e.key === ',') {
   866→        e.preventDefault();
   867→        useStore.getState().navigate('settings');
   868→      }
   869→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   870→        e.preventDefault();
   871→        setShortcutsOpen((prev) => !prev);
   872→      }
   873→      // Number keys 1-9 for view navigation (no modifiers, not in input)
   874→      if (!isInput && !hasModifier && !e.shiftKey && NUMBER_KEY_VIEWS[e.key]) {
   875→        e.preventDefault();
   876→        useStore.getState().navigate(NUMBER_KEY_VIEWS[e.key]);
   877→      }
   878→      // D key for theme toggle (no modifiers, not in input)
   879→      if (!isInput && !hasModifier && e.key === 'd') {
   880→        e.preventDefault();
   881→        setTheme(theme === 'dark' ? 'light' : 'dark');
   882→      }
   883→      // N key for context-sensitive new action (no modifiers, not in input)
   884→      if (!isInput && !hasModifier && e.key === 'n') {
   885→        e.preventDefault();
   886→        document.dispatchEvent(new CustomEvent('shortcut:new-action'));
   887→      }
   888→    };
   889→    document.addEventListener('keydown', handleKeyDown);
   890→    return () => document.removeEventListener('keydown', handleKeyDown);
   891→  }, [setCommandOpen, theme, setTheme]);
   892→
   893→  return (
   894→    <ToastProvider>
   895→    <MobileFAB />
   896→    <div className="flex min-h-screen">
   897→      <DesktopSidebar />
   898→
   899→      <main className="flex-1 min-w-0 flex flex-col">
   900→        <DesktopTopBar />
   901→        <div className="content-area flex-1 px-4 md:px-5 lg:px-6 py-5 pb-[76px] md:pb-6">
   902→          {/* Mobile header */}
   903→          <div className="flex items-center justify-between mb-5 md:mb-0">
   904→            <div className="flex items-center gap-2">
   905→              <span className="md:hidden text-sm font-bold tracking-tight text-foreground">
   906→                DELULU
   907→              </span>
   908→              <MobileStreakBadge />
   909→            </div>
   910→            <div className="flex items-center gap-1.5 ml-auto">
   911→              <ThemeToggle />
   912→              <button
   913→                onClick={() => setCommandOpen(true)}
   914→                className="md:hidden flex items-center justify-center h-9 w-9 rounded-lg border border-border bg-background/50 text-muted-foreground hover:bg-accent hover:text-foreground active:scale-95 transition-all duration-150"
   915→                aria-label="Search"
   916→              >
   917→                <Search className="size-4" />
   918→              </button>
   919→            </div>
   920→          </div>
   921→
   922→          {/* Onboarding — shows when no subjects exist */}
   923→          {subjects.length === 0 && <Onboarding />}
   924→
   925→          <ViewRouter />
   926→        </div>
   927→      </main>
   928→
   929→      <MobileBottomNav onMoreOpen={() => setMoreSheetOpen(true)} />
   930→      <MobileMoreSheet open={moreSheetOpen} onOpenChange={setMoreSheetOpen} />
   931→      <CommandPalette onShowShortcuts={showShortcuts} />
   932→      <ShortcutsOverlay open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   933→    </div>
   934→    </ToastProvider>
   935→  );
   936→}
