    61→    61→// ─── Notification badges hook ──────────────────────────────────────

    63→    63→function useNavBadges() {
    64→    64→  const tasks = useStore((s) => s.tasks);
    65→    65→  const revisionItems = useStore((s) => s.revisionItems);
    66→    66→  const exams = useStore((s) => s.exams);
    67→    67→  const assignments = useStore((s) => s.assignments);
    68→    68→
    69→    69→  return useMemo(() => {
    70→    70→    const badges: Partial<Record<ViewId, number>> = {};
    71→    71→
    72→    72→    // Overdue tasks
    73→    73→    const today = new Date().toISOString().split('T')[0];
    74→    74→    const overdueTasks = tasks.filter(t => !t.completed && t.dueDate && t.dueDate < today);
    75→    75→    if (overdueTasks.length > 0) badges['tasks'] = overdueTasks.length;
    76→    76→
    77→    77→    // Due revisions
    78→    78→    const dueRevisions = revisionItems.filter(r => {
    79→    79→      const nextReview = new Date(r.nextReview);
    80→    80→      return nextReview <= new Date();
    81→    81→    });
    82→    82→    if (dueRevisions.length > 0) badges['revision'] = dueRevisions.length;
    83→    83→
    84→    84→    // Upcoming exams (within 7 days)
    85→    85→    const weekFromNow = new Date();
    86→    86→    weekFromNow.setDate(weekFromNow.getDate() + 7);
    87→    87→    const upcomingExams = exams.filter(e => e.status === 'upcoming' && e.date >= today && e.date <= weekFromNow.toISOString().split('T')[0]);
    88→    88→    if (upcomingExams.length > 0) badges['exams'] = upcomingExams.length;
    89→    89→
    90→    90→    // Upcoming assignments (within 3 days)
    91→    91→    const threeDays = new Date();
    92→    92→    threeDays.setDate(threeDays.getDate() + 3);
    93→    93→    const upcomingAssign = assignments.filter(a => a.status !== 'completed' && a.deadline >= today && a.deadline <= threeDays.toISOString().split('T')[0]);
    94→    94→    if (upcomingAssign.length > 0) badges['assignments'] = upcomingAssign.length;
    95→    95→
    96→    96→    return badges;
    97→    97→  }, [tasks, revisionItems, exams, assignments]);
    98→    98→}

   834→   834→    setCommandOpen(false);
   835→   835→    setShortcutsOpen(true);
   836→   836→  }, [setCommandOpen]);
   837→   837→
   838→   838→  useEffect(() => {
   839→   839→    const NUMBER_KEY_VIEWS: Record<string, ViewId> = {
   840→   840→      '1': 'dashboard',
   841→   841→      '2': 'subjects',
   842→   842→      '3': 'marks',
   843→   843→      '4': 'attendance',
   844→   844→      '5': 'exams',
   845→   845→      '6': 'focus',
   846→   846→      '7': 'revision',
   847→   847→      '8': 'notes',

   849→   849→    };
   850→   850→
   851→   851→    const handleKeyDown = (e: KeyboardEvent) => {
   852→   852→      // Don't capture when typing in inputs
   853→   853→      const tag = (e.target as HTMLElement)?.tagName;
   854→   854→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   855→   855→      const hasModifier = e.metaKey || e.ctrlKey || e.altKey;
   856→   856→
   857→   857→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   858→   858→        e.preventDefault();
   859→   859→        setCommandOpen(true);
   860→   860→      }
   861→   861→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'd' || e.key === 'D')) {
   862→   862→        e.preventDefault();
   863→   863→        setTheme(theme === 'dark' ? 'light' : 'dark');
   864→   864→      }
   865→   865→      if ((e.metaKey || e.ctrlKey) && e.key === ',') {
   866→   866→        e.preventDefault();
   867→   867→        useStore.getState().navigate('settings');
   868→   868→      }
   869→   869→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   870→   870→        e.preventDefault();
   871→   871→        setShortcutsOpen((prev) => !prev);
   872→   872→      }
   873→   873→      // Number keys 1-9 for view navigation (no modifiers, not in input)
   874→   874→      if (!isInput && !hasModifier && !e.shiftKey && NUMBER_KEY_VIEWS[e.key]) {
   875→   875→        e.preventDefault();
   876→   876→        useStore.getState().navigate(NUMBER_KEY_VIEWS[e.key]);
   877→   877→      }
   878→   878→      // D key for theme toggle (no modifiers, not in input)
   879→   879→      if (!isInput && !hasModifier && e.key === 'd') {
   880→   880→        e.preventDefault();
   881→   881→        setTheme(theme === 'dark' ? 'light' : 'dark');
   882→   882→      }
   883→   883→      // N key for context-sensitive new action (no modifiers, not in input)
   884→   884→      if (!isInput && !hasModifier && e.key === 'n') {
   885→   885→        e.preventDefault();
   886→   886→        document.dispatchEvent(new CustomEvent('shortcut:new-action'));
   887→   887→      }
   888→   888→    };
   889→   889→    document.addEventListener('keydown', handleKeyDown);
   890→   890→    return () => document.removeEventListener('keydown', handleKeyDown);
   891→   891→  }, [setCommandOpen, theme, setTheme]);
   892→   892→
   893→   893→  return (
   894→   894→    <ToastProvider>
   895→   895→    <MobileFAB />
   896→   896→    <div className="flex min-h-screen">
   897→   897→      <DesktopSidebar />
   898→   898→
   899→   899→      <main className="flex-1 min-w-0 flex flex-col">
   900→   900→        <DesktopTopBar />
   901→   901→        <div className="content-area flex-1 px-4 md:px-5 lg:px-6 py-5 pb-[76px] md:pb-6">
   902→   902→          {/* Mobile header */}
   903→   903→          <div className="flex items-center justify-between mb-5 md:mb-0">
