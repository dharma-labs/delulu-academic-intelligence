    46→// -- Health status --
    47→function getHealthConfig(score: number) {
    48→   if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    49→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    50→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    51→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    52→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    53→}

    55→function formatStudyTime(seconds: number): string {
    56→  const totalMinutes = Math.floor(seconds / 60);
    57→  const h = Math.floor(totalMinutes / 60);
    58→  const m = totalMinutes % 60;
    59→  if (h === 0) return `${m}m`;
    60→  if (m === 0) return `${h}h`;
    61→  return `${h}h ${m}m`;
    62→}

    65→  const map: Record<SignalStatus, string> = {

    67→    critical: 'bg-red-500', upcoming: 'bg-purple-500', nodata: 'bg-muted-foreground/40',
    68→  };
    69→  return map[signal];

    81→      <svg width="112" height="112" viewBox="0 0 112 112" className="-rotate-90">

    84→          cx="56" cy="56" r={radius} fill="none" stroke={strokeColor} strokeWidth="5"

    97→// -- Weekly Mini Chart (mobile sparkline) --
    98→function WeeklyMiniChart({ studySessions }: { studySessions: { date: string; duration: number }[] }) {
    99→  const bars = useMemo(() => {
   100→    const days: { label: string; minutes: number; isToday: boolean }[] = [];
   101→    for (let i = 6; i >= 0; i--) {
   102→      const d = new Date();
   103→      d.setDate(d.getDate() - i);
   104→      const dateStr = d.toISOString().split('T')[0];
   105→      const dayMinutes = Math.floor(studySessions
   106→        .filter(s => s.date === dateStr)
   107→        .reduce((sum, s) => sum + s.duration / 60, 0));
   108→      days.push({
   109→        label: format(d, 'EEE').charAt(0),
   110→        minutes: dayMinutes,
   111→        isToday: i === 0,
   112→      });
   113→    }
   114→    return days;
   115→  }, [studySessions]);
   116→
   117→  const maxMin = Math.max(...bars.map(b => b.minutes), 1);
   118→
   119→  return (
   120→    <div className="flex items-end gap-1.5 h-12">
   121→      {bars.map((b, i) => (
   122→        <div key={i} className="flex-1 flex flex-col items-center gap-1">
   123→          <div className="w-full relative" style={{ height: '32px' }}>
   124→            <div
   125→              className={cn(
   126→                'absolute bottom-0 left-1/2 -translate-x-1/2 w-3/4 rounded-t-sm transition-all duration-500',
   127→                b.isToday ? 'bg-primary' : 'bg-primary/25',
   128→              )}
   129→              style={{ height: `${Math.max(4, (b.minutes / maxMin) * 100)}%` }}
   130→            />
   131→          </div>
   132→          <span className={cn('text-[8px] tabular-nums', b.isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>
   133→            {b.label}
   134→          </span>
   135→        </div>
   136→      ))}
   137→    </div>
   138→  );
   139→}

   144→// ─── Weekly Heatmap ──────────────────────────────────────────────
   145→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
   146→  const days = useMemo(() => {
   147→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
   148→    return Array.from({ length: 7 }, (_, i) => {
   149→      const d = addDays(start, i);
   150→      const dateStr = d.toISOString().split('T')[0];
   151→      const daySessions = studySessions.filter((s) => s.date === dateStr);
   152→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
   153→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
   154→    });
   155→  }, [studySessions]);
   156→
   157→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
   158→  const today = new Date().toISOString().split('T')[0];
   159→
   160→  return (
   161→    <div className="grid grid-cols-7 gap-1.5">
   162→      {days.map((d) => {
   163→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
   164→        const isToday = d.dateStr === today;
   165→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
   166→        return (
   167→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
   168→            <div
   169→              className={cn('w-full aspect-square rounded-md transition-colors', isToday && 'ring-2 ring-primary/30', d.count === 0 && 'chart-empty-cell')}
   170→              style={d.count > 0 ? { backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` } : undefined}
   171→            />
   172→            <span className={cn('text-[10px]', isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>{d.label}</span>
   173→            {d.count > 0 && (
   174→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   175→            )}
   176→          </div>
   177→        );
   178→      })}
   179→    </div>
   180→  );
   181→}
