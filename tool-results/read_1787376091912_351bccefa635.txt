    63→function useHiddenWidgets(): [string[], (id: string) => void] {
    64→  const [hidden, setHidden] = useState<string[]>(() => {
    65→    if (typeof window === 'undefined') return [];
    66→    try {
    67→      const stored = localStorage.getItem('dashboard-hidden-widgets');
    68→      return stored ? JSON.parse(stored) : [];
    69→    } catch { return []; }
    70→  });
    71→  useEffect(() => {
    72→    localStorage.setItem('dashboard-hidden-widgets', JSON.stringify(hidden));
    73→  }, [hidden]);
    74→  const toggle = (id: string) => {
    75→    setHidden((prev) => prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]);
    76→  };
    77→  return [hidden, toggle];
    78→}

    85→// -- Health status --
    86→function getHealthConfig(score: number) {
    87→   if (score >= 80) return { label: 'HEALTHY', status: 'healthy' as SignalStatus, description: 'Academics on track' };
    88→  if (score >= 60) return { label: 'GOOD', status: 'improving' as SignalStatus, description: 'Solid progress' };
    89→  if (score >= 40) return { label: 'NEEDS ATTENTION', status: 'attention' as SignalStatus, description: 'Some areas need focus' };
    90→  if (score > 0) return { label: 'CRITICAL', status: 'critical' as SignalStatus, description: 'Immediate action required' };
    91→  return { label: 'NO DATA', status: 'nodata' as SignalStatus, description: 'Add subjects to begin' };
    92→}

    94→function formatStudyTime(seconds: number): string {
    95→  const totalMinutes = Math.floor(seconds / 60);
    96→  const h = Math.floor(totalMinutes / 60);
    97→  const m = totalMinutes % 60;
    98→  if (h === 0) return `${m}m`;
    99→  if (m === 0) return `${h}h`;
   100→  return `${h}h ${m}m`;
   101→}

   103→function signalDotClass(signal: SignalStatus): string {

   106→    critical: 'bg-red-500', upcoming: 'bg-purple-500', nodata: 'bg-muted-foreground/40',

   108→  return map[signal];

   136→// -- Weekly Mini Chart (mobile sparkline) --
   137→function WeeklyMiniChart({ studySessions }: { studySessions: { date: string; duration: number }[] }) {
   138→  const bars = useMemo(() => {
   139→    const days: { label: string; minutes: number; isToday: boolean }[] = [];
   140→    for (let i = 6; i >= 0; i--) {
   141→      const d = new Date();
   142→      d.setDate(d.getDate() - i);
   143→      const dateStr = d.toISOString().split('T')[0];
   144→      const dayMinutes = Math.floor(studySessions
   145→        .filter(s => s.date === dateStr)
   146→        .reduce((sum, s) => sum + s.duration / 60, 0));
   147→      days.push({
   148→        label: format(d, 'EEE').charAt(0),
   149→        minutes: dayMinutes,
   150→        isToday: i === 0,
   151→      });
   152→    }
   153→    return days;
   154→  }, [studySessions]);
   155→
   156→  const maxMin = Math.max(...bars.map(b => b.minutes), 1);
   157→
   158→  return (
   159→    <div className="flex items-end gap-1.5 h-12">
   160→      {bars.map((b, i) => (
   161→        <div key={i} className="flex-1 flex flex-col items-center gap-1">
   162→          <div className="w-full relative" style={{ height: '32px' }}>
   163→            <div
   164→              className={cn(
   165→                'absolute bottom-0 left-1/2 -translate-x-1/2 w-3/4 rounded-t-sm transition-all duration-500',
   166→                b.isToday ? 'bg-primary' : 'bg-primary/25',
   167→              )}
   168→              style={{ height: `${Math.max(4, (b.minutes / maxMin) * 100)}%` }}
   169→            />
   170→          </div>
   171→          <span className={cn('text-[8px] tabular-nums', b.isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>
   172→            {b.label}
   173→          </span>
   174→        </div>
   175→      ))}
   176→    </div>
   177→  );
   178→}

   183→// ─── Weekly Heatmap ──────────────────────────────────────────────
   184→function WeeklyHeatmap({ studySessions }: { studySessions: { date: string; duration: number; subjectId: string }[] }) {
   185→  const days = useMemo(() => {
   186→    const start = startOfWeek(new Date(), { weekStartsOn: 1 });
   187→    return Array.from({ length: 7 }, (_, i) => {
   188→      const d = addDays(start, i);
   189→      const dateStr = d.toISOString().split('T')[0];
   190→      const daySessions = studySessions.filter((s) => s.date === dateStr);
   191→      const totalMinutes = Math.floor(daySessions.reduce((sum, s) => sum + s.duration / 60, 0));
   192→      return { date: d, dateStr, label: format(d, 'EEE'), totalMinutes, count: daySessions.length };
   193→    });
   194→  }, [studySessions]);
   195→
   196→  const maxMinutes = Math.max(...days.map((d) => d.totalMinutes), 1);
   197→  const today = new Date().toISOString().split('T')[0];
   198→
   199→  return (
   200→    <div className="grid grid-cols-7 gap-1.5">
   201→      {days.map((d) => {
   202→        const intensity = Math.min(d.totalMinutes / maxMinutes, 1);
   203→        const isToday = d.dateStr === today;
   204→        const bgOpacity = d.count === 0 ? 0.08 : Math.max(0.15, intensity);
   205→        return (
   206→          <div key={d.dateStr} className="flex flex-col items-center gap-1">
   207→            <div
   208→              className={cn('w-full aspect-square rounded-md transition-colors', isToday && 'ring-2 ring-primary/30', d.count === 0 && 'chart-empty-cell')}
   209→              style={d.count > 0 ? { backgroundColor: `rgba(var(--primary-rgb), ${bgOpacity})` } : undefined}
   210→            />
   211→            <span className={cn('text-[10px]', isToday ? 'font-semibold text-foreground' : 'text-muted-foreground')}>{d.label}</span>
   212→            {d.count > 0 && (
   213→              <span className="text-[9px] text-muted-foreground tabular-nums">{d.totalMinutes}m</span>
   214→            )}
   215→          </div>
   216→        );
   217→      })}
   218→    </div>
   219→  );
   220→}
