    62→// ─── Notification badges hook ──────────────────────────────────────
    63→
    64→function useNavBadges() {
    65→  const tasks = useStore((s) => s.tasks);
    66→  const revisionItems = useStore((s) => s.revisionItems);
    67→  const exams = useStore((s) => s.exams);
    68→  const assignments = useStore((s) => s.assignments);
    69→
    70→  return useMemo(() => {
    71→    const badges: Partial<Record<ViewId, number>> = {};
    72→
    73→    // Overdue tasks
    74→    const today = new Date().toISOString().split('T')[0];
    75→    const overdueTasks = tasks.filter(t => !t.completed && t.dueDate && t.dueDate < today);
    76→    if (overdueTasks.length > 0) badges['tasks'] = overdueTasks.length;
    77→
    78→    // Due revisions
    79→    const dueRevisions = revisionItems.filter(r => {
    80→      const nextReview = new Date(r.nextReview);
    81→      return nextReview <= new Date();
    82→    });
    83→    if (dueRevisions.length > 0) badges['revision'] = dueRevisions.length;
    84→
    85→    // Upcoming exams (within 7 days)
    86→    const weekFromNow = new Date();
    87→    weekFromNow.setDate(weekFromNow.getDate() + 7);
    88→    const upcomingExams = exams.filter(e => e.status === 'upcoming' && e.date >= today && e.date <= weekFromNow.toISOString().split('T')[0]);
    89→    if (upcomingExams.length > 0) badges['exams'] = upcomingExams.length;
    90→
    91→    // Upcoming assignments (within 3 days)
    92→    const threeDays = new Date();
    93→    threeDays.setDate(threeDays.getDate() + 3);
    94→    const upcomingAssign = assignments.filter(a => a.status !== 'completed' && a.deadline >= today && a.deadline <= threeDays.toISOString().split('T')[0]);
    95→    if (upcomingAssign.length > 0) badges['assignments'] = upcomingAssign.length;
    96→
    97→    return badges;
    98→  }, [tasks, revisionItems, exams, assignments]);
    99→}

   109→    </span>
   110→  );
   111→}

   717→  { label: 'Quick Note', icon: StickyNote, color: 'bg-amber-500 dark:bg-amber-400', customAction: 'quick-note' },

   721→];
   722→
   723→function MobileFAB() {
   724→  const [open, setOpen] = useState(false);
   725→  const [quickNoteOpen, setQuickNoteOpen] = useState(false);
   726→  const navigate = useStore((s) => s.navigate);
   727→  const currentView = useStore((s) => s.currentView);
   728→  const fabRef = useRef<HTMLDivElement>(null);
   729→
   730→  // Auto-close when navigating away (store currentView is external state)
   731→  const prevViewRef = useRef(currentView);
   732→  /* eslint-disable react-hooks/set-state-in-effect */
   733→  useEffect(() => {
   734→    if (prevViewRef.current !== currentView) {
   735→      prevViewRef.current = currentView;
   736→      setOpen(false);
   737→    }
   738→  }, [currentView]);
   739→  /* eslint-enable react-hooks/set-state-in-effect */
   740→
   741→  const handleClose = useCallback(() => setOpen(false), []);
   742→
   743→  // Close on outside click
   744→  useEffect(() => {
   745→    if (!open) return;
   746→    const handleClick = (e: MouseEvent) => {
   747→      if (fabRef.current && !fabRef.current.contains(e.target as Node)) {
   748→        setOpen(false);
   749→      }
   750→    };
   751→    // Delay to avoid immediate close from the toggle click
   752→    const timer = setTimeout(() => {
   753→      document.addEventListener('mousedown', handleClick);
   754→    }, 0);
   755→    return () => {
   756→      clearTimeout(timer);
   757→      document.removeEventListener('mousedown', handleClick);
   758→    };
   759→  }, [open]);
   760→
   761→  const handleAction = (action: typeof FAB_ACTIONS[number]) => {
   762→    setOpen(false);
   763→    if (action.customAction === 'quick-note') {
   764→      setQuickNoteOpen(true);
   765→    } else if (action.view) {
   766→      navigate(action.view);
   767→    }
   768→  };
   769→
   770→  return (
   771→    <div ref={fabRef} className="fixed bottom-[76px] right-4 z-30 md:hidden">
   772→      {/* Backdrop overlay when open */}
   773→      <AnimatePresence>
   774→        {open && (
   775→          <motion.div
   776→            initial={{ opacity: 0 }}
   777→            animate={{ opacity: 1 }}
   778→            exit={{ opacity: 0 }}
   779→            transition={{ duration: 0.15 }}
   780→            className="fixed inset-0 z-[-1]"
   781→            onClick={() => setOpen(false)}
   782→          />
   783→        )}
   784→      </AnimatePresence>
   785→
   786→      {/* Radial menu items */}
   787→      <AnimatePresence>
   788→        {open && (
   789→          <div className="absolute bottom-16 right-0 flex flex-col items-end gap-2.5">
   790→            {FAB_ACTIONS.map((action, i) => {
   791→              const Icon = action.icon;
   792→              return (
   793→                <motion.div
   794→                  key={action.label}
   795→                  initial={{ opacity: 0, scale: 0.5, y: 10 }}
   796→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   797→                  exit={{ opacity: 0, scale: 0.5, y: 10 }}
   798→                  transition={{ duration: 0.2, delay: i * 0.04 }}
   799→                  className="flex items-center gap-2.5"
   800→                >
   801→                  <span className="text-xs font-medium text-foreground whitespace-nowrap bg-card/90 backdrop-blur-sm border border-border rounded-lg px-2.5 py-1.5 shadow-sm">
   802→                    {action.label}
   803→                  </span>
   804→                  <button
   805→                    onClick={() => handleAction(action)}
   806→                    className={cn(
   807→                      'h-10 w-10 rounded-full flex items-center justify-center shadow-lg press-scale transition-colors',
   808→                      action.color,
   809→                      'text-white'
   810→                    )}
   811→                    aria-label={action.label}
   812→                  >
   813→                    <Icon className="size-4" />
   814→                  </button>
   815→                </motion.div>
   816→              );
   817→            })}
   818→          </div>
   819→        )}
   820→      </AnimatePresence>
   821→
   822→      {/* Main FAB button */}
   823→      <motion.button
   824→        onClick={() => setOpen((prev) => !prev)}
   825→        whileTap={{ scale: 0.92 }}
   826→        className="h-12 w-12 rounded-full bg-primary text-primary-foreground fab-shadow press-scale flex items-center justify-center"
   827→        aria-label={open ? 'Close quick actions' : 'Open quick actions'}
   828→        style={{ rotate: open ? 45 : 0 }}
   829→        transition={{ duration: 0.2 }}
   830→      >
   831→        <Plus className="size-5" />
   832→      </motion.button>
   833→
   834→      <QuickNoteDialog open={quickNoteOpen} onOpenChange={setQuickNoteOpen} />
   835→    </div>
   836→  );
   837→}

   841→export function AppShell() {
   842→  const subjects = useStore((s) => s.subjects);
   843→  const setCommandOpen = useStore((s) => s.setCommandOpen);
   844→  const [moreSheetOpen, setMoreSheetOpen] = useState(false);
   845→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   846→  const { theme, setTheme } = useTheme();
   847→
   848→  const showShortcuts = useCallback(() => {
   849→    setCommandOpen(false);
   850→    setShortcutsOpen(true);
   851→  }, [setCommandOpen]);
   852→
   853→  useEffect(() => {
   854→    const NUMBER_KEY_VIEWS: Record<string, ViewId> = {
   855→      '1': 'dashboard',
   856→      '2': 'subjects',
   857→      '3': 'marks',
   858→      '4': 'attendance',
   859→      '5': 'exams',
   860→      '6': 'focus',
   861→      '7': 'revision',
   862→      '8': 'notes',
   863→      '9': 'tasks',
   864→    };
   865→
   866→    const handleKeyDown = (e: KeyboardEvent) => {
   867→      // Don't capture when typing in inputs
   868→      const tag = (e.target as HTMLElement)?.tagName;
   869→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   870→      const hasModifier = e.metaKey || e.ctrlKey || e.altKey;
   871→
   872→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   873→        e.preventDefault();
   874→        setCommandOpen(true);
   875→      }
   876→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'd' || e.key === 'D')) {
   877→        e.preventDefault();
   878→        setTheme(theme === 'dark' ? 'light' : 'dark');
   879→      }
   880→      if ((e.metaKey || e.ctrlKey) && e.key === ',') {
   881→        e.preventDefault();
   882→        useStore.getState().navigate('settings');
   883→      }
   884→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   885→        e.preventDefault();
   886→        setShortcutsOpen((prev) => !prev);
   887→      }
   888→      // Number keys 1-9 for view navigation (no modifiers, not in input)
   889→      if (!isInput && !hasModifier && !e.shiftKey && NUMBER_KEY_VIEWS[e.key]) {
   890→        e.preventDefault();
   891→        useStore.getState().navigate(NUMBER_KEY_VIEWS[e.key]);
   892→      }
   893→      // D key for theme toggle (no modifiers, not in input)
   894→      if (!isInput && !hasModifier && e.key === 'd') {
   895→        e.preventDefault();
   896→        setTheme(theme === 'dark' ? 'light' : 'dark');
   897→      }
   898→      // N key for context-sensitive new action (no modifiers, not in input)
   899→      if (!isInput && !hasModifier && e.key === 'n') {
   900→        e.preventDefault();
   901→        document.dispatchEvent(new CustomEvent('shortcut:new-action'));
   902→      }
   903→    };
   904→    document.addEventListener('keydown', handleKeyDown);
   905→    return () => document.removeEventListener('keydown', handleKeyDown);
   906→  }, [setCommandOpen, theme, setTheme]);
   907→
   908→  return (
   909→    <ToastProvider>
   910→    <MobileFAB />
   911→    <div className="flex min-h-screen">
   912→      <DesktopSidebar />
   913→
   914→      <main className="flex-1 min-w-0 flex flex-col">
   915→        <DesktopTopBar />
   916→        <div className="content-area flex-1 px-4 md:px-5 lg:px-6 py-5 pb-[76px] md:pb-6">
   917→          {/* Mobile header */}
   918→          <div className="flex items-center justify-between mb-5 md:mb-0">
   919→            <div className="flex items-center gap-2">
   920→              <span className="md:hidden text-sm font-bold tracking-tight text-foreground">
   921→                DELULU
   922→              </span>
   923→              <MobileStreakBadge />
   924→            </div>
   925→            <div className="flex items-center gap-1.5 ml-auto">
   926→              <ThemeToggle />
   927→              <button
   928→                onClick={() => setCommandOpen(true)}
   929→                className="md:hidden flex items-center justify-center h-9 w-9 rounded-lg border border-border bg-background/50 text-muted-foreground hover:bg-accent hover:text-foreground active:scale-95 transition-all duration-150"
   930→                aria-label="Search"
   931→              >
   932→                <Search className="size-4" />
   933→              </button>
   934→            </div>
   935→          </div>
   936→
   937→          {/* Onboarding — shows when no subjects exist */}
   938→          {subjects.length === 0 && <Onboarding />}
   939→
   940→          <ViewRouter />
   941→        </div>
   942→      </main>
   943→
   944→      <MobileBottomNav onMoreOpen={() => setMoreSheetOpen(true)} />
   945→      <MobileMoreSheet open={moreSheetOpen} onOpenChange={setMoreSheetOpen} />
   946→      <CommandPalette onShowShortcuts={showShortcuts} />
   947→      <ShortcutsOverlay open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
   948→    </div>
   949→    </ToastProvider>
   950→  );
   951→}
