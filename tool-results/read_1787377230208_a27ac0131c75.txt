    62→    62→// ─── Notification badges hook ──────────────────────────────────────
    63→    63→
    64→    64→function useNavBadges() {
    65→    65→  const tasks = useStore((s) => s.tasks);
    66→    66→  const revisionItems = useStore((s) => s.revisionItems);
    67→    67→  const exams = useStore((s) => s.exams);
    68→    68→  const assignments = useStore((s) => s.assignments);
    69→    69→
    70→    70→  return useMemo(() => {
    71→    71→    const badges: Partial<Record<ViewId, number>> = {};
    72→    72→
    73→    73→    // Overdue tasks
    74→    74→    const today = new Date().toISOString().split('T')[0];
    75→    75→    const overdueTasks = tasks.filter(t => !t.completed && t.dueDate && t.dueDate < today);
    76→    76→    if (overdueTasks.length > 0) badges['tasks'] = overdueTasks.length;
    77→    77→
    78→    78→    // Due revisions
    79→    79→    const dueRevisions = revisionItems.filter(r => {
    80→    80→      const nextReview = new Date(r.nextReview);
    81→    81→      return nextReview <= new Date();
    82→    82→    });
    83→    83→    if (dueRevisions.length > 0) badges['revision'] = dueRevisions.length;
    84→    84→
    85→    85→    // Upcoming exams (within 7 days)
    86→    86→    const weekFromNow = new Date();
    87→    87→    weekFromNow.setDate(weekFromNow.getDate() + 7);
    88→    88→    const upcomingExams = exams.filter(e => e.status === 'upcoming' && e.date >= today && e.date <= weekFromNow.toISOString().split('T')[0]);
    89→    89→    if (upcomingExams.length > 0) badges['exams'] = upcomingExams.length;
    90→    90→
    91→    91→    // Upcoming assignments (within 3 days)
    92→    92→    const threeDays = new Date();
    93→    93→    threeDays.setDate(threeDays.getDate() + 3);
    94→    94→    const upcomingAssign = assignments.filter(a => a.status !== 'completed' && a.deadline >= today && a.deadline <= threeDays.toISOString().split('T')[0]);
    95→    95→    if (upcomingAssign.length > 0) badges['assignments'] = upcomingAssign.length;
    96→    96→
    97→    97→    return badges;
    98→    98→  }, [tasks, revisionItems, exams, assignments]);
    99→    99→}

   841→   841→export function AppShell() {
   842→   842→  const subjects = useStore((s) => s.subjects);

   844→   844→  const [moreSheetOpen, setMoreSheetOpen] = useState(false);

   848→   848→  const showShortcuts = useCallback(() => {
   849→   849→    setCommandOpen(false);
   850→   850→    setShortcutsOpen(true);
   851→   851→  }, [setCommandOpen]);
   852→   852→
   853→   853→  useEffect(() => {
   854→   854→    const NUMBER_KEY_VIEWS: Record<string, ViewId> = {
   855→   855→      '1': 'dashboard',
   856→   856→      '2': 'subjects',
   857→   857→      '3': 'marks',
   858→   858→      '4': 'attendance',
   859→   859→      '5': 'exams',
   860→   860→      '6': 'focus',
   861→   861→      '7': 'revision',
   862→   862→      '8': 'notes',
   863→   863→      '9': 'tasks',
   864→   864→    };
   865→   865→
   866→   866→    const handleKeyDown = (e: KeyboardEvent) => {
   867→   867→      // Don't capture when typing in inputs
   868→   868→      const tag = (e.target as HTMLElement)?.tagName;
   869→   869→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   870→   870→      const hasModifier = e.metaKey || e.ctrlKey || e.altKey;
   871→   871→
   872→   872→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   873→   873→        e.preventDefault();
   874→   874→        setCommandOpen(true);
   875→   875→      }
   876→   876→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'd' || e.key === 'D')) {
   877→   877→        e.preventDefault();
   878→   878→        setTheme(theme === 'dark' ? 'light' : 'dark');
   879→   879→      }
   880→   880→      if ((e.metaKey || e.ctrlKey) && e.key === ',') {
   881→   881→        e.preventDefault();
   882→   882→        useStore.getState().navigate('settings');
   883→   883→      }
   884→   884→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   885→   885→        e.preventDefault();
   886→   886→        setShortcutsOpen((prev) => !prev);
   887→   887→      }
   888→   888→      // Number keys 1-9 for view navigation (no modifiers, not in input)
   889→   889→      if (!isInput && !hasModifier && !e.shiftKey && NUMBER_KEY_VIEWS[e.key]) {
   890→   890→        e.preventDefault();
   891→   891→        useStore.getState().navigate(NUMBER_KEY_VIEWS[e.key]);
   892→   892→      }
   893→   893→      // D key for theme toggle (no modifiers, not in input)
   894→   894→      if (!isInput && !hasModifier && e.key === 'd') {
   895→   895→        e.preventDefault();
   896→   896→        setTheme(theme === 'dark' ? 'light' : 'dark');
   897→   897→      }
   898→   898→      // N key for context-sensitive new action (no modifiers, not in input)
   899→   899→      if (!isInput && !hasModifier && e.key === 'n') {
   900→   900→        e.preventDefault();
   901→   901→        document.dispatchEvent(new CustomEvent('shortcut:new-action'));
   902→   902→      }
