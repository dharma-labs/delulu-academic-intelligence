    74→// --- Helpers ---
    75→function formatElapsed(totalSeconds: number): string {
    76→  const m = Math.floor(totalSeconds / 60);
    77→  const s = totalSeconds % 60;
    78→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    79→}
    80→
    81→function formatDuration(totalSeconds: number): string {
    82→  if (totalSeconds < 60) return `${totalSeconds}s`;
    83→  const m = Math.floor(totalSeconds / 60);
    84→  const s = totalSeconds % 60;
    85→  if (m < 60) return s > 0 ? `${m}m ${s}s` : `${m}m`;
    86→  const h = Math.floor(m / 60);
    87→  const rm = m % 60;
    88→  return rm > 0 ? `${h}h ${rm}m` : `${h}h`;
    89→}
    90→
    91→function todayDateStr(): string {
    92→  return new Date().toISOString().split('T')[0];
    93→}
