     1→'use client';
     2→
     3→import { useEffect, useState, useMemo } from 'react';
     4→import { useStore, getSubjectAttendance, getSubjectProgress, getDueRevisionItems } from '@/lib/store';
     5→import { Clock, BarChart3, BrainCircuit, Target, Flame, Activity, Download } from 'lucide-react';
     6→import { format, subDays, startOfWeek, eachDayOfInterval, parseISO } from 'date-fns';
     7→import { motion } from 'framer-motion';
     8→import {
     9→  BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer,
    10→  LineChart, Line, PieChart, Pie, Cell,
    11→} from 'recharts';
    12→import { PageHeader, MetricCard, SectionHeader, EmptyState, InsightCard, CompactProgress } from '@/components/shared';
    13→import { Button } from '@/components/ui/button';
    14→import { exportAnalyticsCSV } from '@/lib/csv-export';
    15→import { progressColorClass } from '@/components/shared';
    16→
    17→type SubjectSummary = { id: string; name: string; color: string; percentage: number };
    18→type HeatmapDay = { dateStr: string; minutes: number };
    19→type StudyDistItem = { subjectId: string; name: string; color: string; minutes: number };
    20→type AssessSummary = { id: string; name: string; obtainedMarks: number; maxMarks: number };
    21→
    22→export default function AnalyticsView() {
    23→  const [ready, setReady] = useState(false);
    24→  const [weekMinutes, setWeekMinutes] = useState(0);
    25→  const [weekSessions, setWeekSessions] = useState(0);
    26→  const [avgSession, setAvgSession] = useState(0);
    27→  const [uniqueSubjects, setUniqueSubjects] = useState(0);
    28→  const [assessmentAvgPct, setAssessmentAvgPct] = useState(0);
    29→  const [assessmentCount, setAssessmentCount] = useState(0);
    30→  const [dueRevisionCount, setDueRevisionCount] = useState(0);
    31→  const [heatmap, setHeatmap] = useState<HeatmapDay[]>([]);
    32→  const [attendanceTrend, setAttendanceTrend] = useState<SubjectSummary[]>([]);
    33→  const [syllabusTrend, setSyllabusTrend] = useState<SubjectSummary[]>([]);
    34→  const [studyDist, setStudyDist] = useState<StudyDistItem[]>([]);
    35→  const [recentAssessments, setRecentAssessments] = useState<AssessSummary[]>([]);
    36→  const [completionRate, setCompletionRate] = useState(0);
    37→  const [threshold, setThreshold] = useState(75);
    38→  const [timeRange, setTimeRange] = useState<'week' | 'month' | 'semester'>('week');
    39→  const [chartTheme, setChartTheme] = useState({ primary: '#3B82F6', primaryRgb: '59, 130, 246', card: '#FFFFFF', foreground: '#0F172A', border: '#E2E8F0' });
    40→
    41→  useEffect(() => {
    42→    const compute = () => {
    43→      const s = useStore.getState();
    44→      const active = s.subjects.filter((x) => !x.archived);
    45→      const days = eachDayOfInterval({ start: subDays(new Date(), 34), end: new Date() });
    46→      const hm = days.map((day) => {
    47→        const ds = format(day, 'yyyy-MM-dd');
    48→        const secs = s.studySessions.filter((x) => x.date === ds).reduce((a, x) => a + x.duration, 0);
    49→        return { dateStr: ds, minutes: Math.round(secs / 60) };
    50→      });
    51→      setHeatmap(hm);
    52→      const ws = startOfWeek(new Date(), { weekStartsOn: 1 });
    53→      const wsList = s.studySessions.filter((x) => parseISO(x.date) >= ws);
    54→      const totalSec = wsList.reduce((a, x) => a + x.duration, 0);
    55→      setWeekMinutes(Math.round(totalSec / 60));
    56→      setWeekSessions(wsList.length);
    57→      setAvgSession(wsList.length > 0 ? Math.round(totalSec / wsList.length / 60) : 0);
    58→      setUniqueSubjects(new Set(wsList.map((x) => x.subjectId)).size);
    59→      const aCount = s.assessments.length;
    60→      setAssessmentCount(aCount);
    61→      setAssessmentAvgPct(aCount > 0 ? Math.round(s.assessments.reduce((a, x) => a + (x.obtainedMarks / x.maxMarks) * 100, 0) / aCount) : 0);
    62→      setRecentAssessments([...s.assessments].sort((a, b) => b.date.localeCompare(a.date)).slice(0, 5));
    63→      setAttendanceTrend(active.map((sub) => {
    64→        const att = getSubjectAttendance({ attendance: s.attendance }, sub.id);
    65→        return { id: sub.id, name: sub.name, color: sub.color, percentage: att.percentage };
    66→      }));
    67→      setSyllabusTrend(active.map((sub) => {
    68→        const p = getSubjectProgress({ syllabusUnits: s.syllabusUnits }, sub.id);
    69→        return { id: sub.id, name: sub.name, color: sub.color, percentage: p };
    70→      }));
    71→      setStudyDist(
    72→        active
    73→          .map((sub) => ({
    74→            subjectId: sub.id, name: sub.name, color: sub.color,
    75→            minutes: Math.round(s.studySessions.filter((x) => x.subjectId === sub.id && parseISO(x.date) >= ws).reduce((a, x) => a + x.duration, 0) / 60),
    76→          }))
    77→          .filter((d) => d.minutes > 0)
    78→          .sort((a, b) => b.minutes - a.minutes),
    79→      );
    80→      setDueRevisionCount(getDueRevisionItems({ revisionItems: s.revisionItems }).length);
    81→      const syllabusProgresses = active.map((sub) => getSubjectProgress({ syllabusUnits: s.syllabusUnits }, sub.id));
    82→      setCompletionRate(active.length > 0 ? Math.round(syllabusProgresses.reduce((a, p) => a + p, 0) / active.length) : 0);
    83→      setThreshold(s.profile.attendanceThreshold);
    84→      setReady(true);
    85→    };
    86→    compute();
    87→    const unsub = useStore.subscribe(compute);
    88→    return () => unsub();
    89→  }, []);
    90→
    91→  const getColor = (pct: number) => {
    92→    if (pct >= 75) return 'text-[var(--delulu-success)]';
    93→    if (pct >= 50) return 'text-[var(--delulu-info)]';
    94→    if (pct >= 30) return 'text-[var(--delulu-warning)]';
    95→    return 'text-[var(--delulu-danger)]';
    96→  };
    97→
    98→  const getHeatColor = (m: number) => {
    99→    if (m === 0) return 'bg-border';
   100→    if (m < 15) return 'bg-primary/20';
   101→    if (m < 30) return 'bg-primary/40';
   102→    if (m < 60) return 'bg-primary/60';
   103→    return 'bg-primary';
   104→  };
   105→
   106→  // Read CSS variables for Recharts (SVG doesn't support var() in attributes)
   107→  useEffect(() => {
   108→    const read = () => {
   109→      const s = getComputedStyle(document.documentElement);
   110→      setChartTheme({
   111→        primary: s.getPropertyValue('--primary').trim() || '#3B82F6',
   112→        primaryRgb: s.getPropertyValue('--primary-rgb').trim() || '59, 130, 246',
   113→        card: s.getPropertyValue('--card').trim() || '#FFFFFF',
   114→        foreground: s.getPropertyValue('--foreground').trim() || '#0F172A',
   115→        border: s.getPropertyValue('--border').trim() || '#E2E8F0',
   116→      });
   117→    };
   118→    read();
   119→    const observer = new MutationObserver(read);
   120→    observer.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
   121→    return () => observer.disconnect();
   122→  }, []);
   123→
   124→  // Chart data derived from existing computed state
   125→  const weekBarData = useMemo(() => {
   126→    const last7 = heatmap.slice(-7);
   127→    return last7.map((d) => ({
   128→      day: format(parseISO(d.dateStr), 'EEE'),
   129→      minutes: d.minutes,
   130→    }));
   131→  }, [heatmap]);
   132→
   133→  const cgpaTrendData = useMemo(() => {
   134→    return [...recentAssessments]
   135→      .sort((a, b) => a.date.localeCompare(b.date))
   136→      .map((a) => ({
   137→        name: a.name.length > 14 ? a.name.slice(0, 14) + '…' : a.name,
   138→        percentage: Math.round((a.obtainedMarks / a.maxMarks) * 100),
   139→      }));
   140→  }, [recentAssessments]);
   141→
   142→  const pieData = useMemo(() => {
   143→    return studyDist.map((d) => ({
   144→      name: d.name.length > 16 ? d.name.slice(0, 16) + '…' : d.name,
   145→      value: d.minutes,
   146→      color: d.color,
   147→    }));
   148→  }, [studyDist]);
   149→
   150→  const tooltipStyle = useMemo(() => ({
   151→    backgroundColor: chartTheme.card,
   152→    borderColor: chartTheme.border,
   153→    color: chartTheme.foreground,
   154→    border: `1px solid ${chartTheme.border}`,
   155→    borderRadius: '8px',
   156→    fontSize: '12px',
   157→    boxShadow: '0 4px 12px rgba(0,0,0,0.08)',
   158→  }), [chartTheme]);
   159→
   160→  const belowThresholdCount = attendanceTrend.filter((a) => a.percentage < threshold).length;
   161→  const hasInsights = weekMinutes > 0 || uniqueSubjects > 0 || belowThresholdCount > 0;
   162→
   163→  if (!ready) {
   164→    return (
   165→      <div className="p-6">
   166→        <div className="h-6 w-40 bg-muted animate-pulse rounded" />
   167→        <div className="h-4 w-56 bg-muted animate-pulse rounded mt-2" />
   168→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mt-6">
   169→          {[1, 2, 3, 4].map((i) => (
   170→            <div key={i} className="metric-card" />
   171→          ))}
   172→        </div>
   173→      </div>
   174→    );
   175→  }
   176→
   177→  return (
   178→    <div className="p-4 md:p-6 content-area animate-fade-slide-in">
   179→      <PageHeader
   180→        title="Analytics"
   181→        subtitle="Academic performance insights and trends"
   182→        actions={
   183→          <div className="flex items-center gap-2">
   184→            <Button
   185→              variant="outline"
   186→              size="sm"
   187→              className="text-xs"
   188→              onClick={() => {
   189→                const s = useStore.getState();
   190→                exportAnalyticsCSV(s.studySessions, s.subjects);
   191→              }}
   192→            >
   193→              <Download className="h-3.5 w-3.5 mr-1.5" />
   194→              Export Report
   195→            </Button>
   196→            <div className="flex items-center bg-secondary rounded-lg p-0.5">
   197→              {(['week', 'month', 'semester'] as const).map((range) => (
   198→                <button
   199→                  key={range}
   200→                  onClick={() => setTimeRange(range)}
   201→                  className={`px-3 py-1.5 text-xs font-medium rounded-md transition-colors ${
   202→                    timeRange === range
   203→                      ? 'bg-card text-foreground shadow-sm'
   204→                      : 'text-muted-foreground hover:text-foreground'
   205→                  }`}
   206→                >
   207→                  {range.charAt(0).toUpperCase() + range.slice(1)}
   208→                </button>
   209→              ))}
   210→            </div>
   211→          </div>
   212→        }
   213→      />
   214→
   215→      {/* Mobile KPI Strip */}
   216→      <div className="md:hidden flex gap-2 overflow-x-auto scrollbar-none -mx-4 px-4 pb-3">
   217→        {[
   218→          { label: 'This Week', value: `${(weekMinutes / 60).toFixed(1)}h`, sub: 'study time' },
   219→          { label: 'Avg Session', value: `${avgSession}m`, sub: 'per session' },
   220→          { label: 'Subjects', value: `${uniqueSubjects}`, sub: 'studied' },
   221→          { label: 'Completion', value: `${completionRate}%`, sub: 'syllabus' },
   222→        ].map(kpi => (
   223→          <div key={kpi.label} className="shrink-0 bg-card border border-border/50 rounded-xl px-3 py-2.5 min-w-[100px]">
   224→            <p className="text-sm font-bold tracking-tight tabular-nums">{kpi.value}</p>
   225→            <p className="text-[9px] text-muted-foreground font-medium tracking-wider uppercase mt-0.5">{kpi.label}</p>
   226→            <p className="text-[10px] text-muted-foreground mt-0.5">{kpi.sub}</p>
   227→          </div>
   228→        ))}
   229→      </div>
   230→
   231→      {/* Summary Metric Cards */}
   232→      <motion.div
   233→        initial={{ opacity: 0, y: 8 }}
   234→        animate={{ opacity: 1, y: 0 }}
   235→        transition={{ duration: 0.3 }}
   236→        className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4 mb-4 md:mb-6"
   237→      >
   238→        <MetricCard
   239→          label="Study Time"
   240→          value={`${weekMinutes}m`}
   241→          context={`${weekSessions} sessions this week`}
   242→          icon={Clock}
   243→          trend={weekMinutes > 120 ? 'up' : 'neutral'}
   244→          trendValue={weekMinutes > 0 ? `${uniqueSubjects} subjects` : undefined}
   245→        />
   246→        <MetricCard
   247→          label="Avg Session"
   248→          value={`${avgSession}m`}
   249→          context="minutes per session"
   250→          icon={Target}
   251→        />
   252→        <MetricCard
   253→          label="CA Average"
   254→          value={`${assessmentAvgPct}%`}
   255→          context={`${assessmentCount} assessments`}
   256→          icon={BarChart3}
   257→          valueColor={getColor(assessmentAvgPct)}
   258→        />
   259→        <MetricCard
   260→          label="Revision Queue"
   261→          value={dueRevisionCount}
   262→          context="items due now"
   263→          icon={BrainCircuit}
   264→          valueColor={dueRevisionCount > 5 ? 'text-[var(--delulu-danger)]' : undefined}
   265→        />
   266→      </motion.div>
   267→
   268→      {/* Study Activity Heatmap */}
   269→      <motion.div
   270→        initial={{ opacity: 0, y: 8 }}
   271→        animate={{ opacity: 1, y: 0 }}
   272→        transition={{ duration: 0.3, delay: 0.05 }}
   273→        className="metric-card mb-4 md:mb-6"
   274→      >
   275→        <SectionHeader title="Study Activity" subtitle="Last 35 days" />
   276→        <div className="flex flex-wrap gap-[3px]">
   277→          {heatmap.map((d) => (
   278→            <div
   279→              key={d.dateStr}
   280→              className={`w-3 h-3 rounded-[2px] ${getHeatColor(d.minutes)} transition-colors`}
   281→              title={`${d.dateStr}: ${d.minutes}m`}
   282→            />
   283→          ))}
   284→        </div>
   285→        <div className="flex items-center gap-2.5 mt-3 text-[10px] text-muted-foreground">
   286→          <span>Less</span>
   287→          <div className="flex gap-[3px]">
   288→            <div className="w-2.5 h-2.5 rounded-[2px] bg-border" />
   289→            <div className="w-2.5 h-2.5 rounded-[2px] bg-primary/20" />
   290→            <div className="w-2.5 h-2.5 rounded-[2px] bg-primary/40" />
   291→            <div className="w-2.5 h-2.5 rounded-[2px] bg-primary/60" />
   292→            <div className="w-2.5 h-2.5 rounded-[2px] bg-primary" />
   293→          </div>
   294→          <span>More</span>
   295→        </div>
   296→      </motion.div>
   297→
   298→      {/* Recharts Visualizations */}
   299→      <div className="mb-4 md:mb-6">
   300→        <SectionHeader title="Visualizations" subtitle="Interactive charts powered by your study data" />
   301→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 md:gap-4">
   302→          {/* Weekly Study Time Bar Chart */}
   303→          <motion.div
   304→            initial={{ opacity: 0, y: 8 }}
   305→            animate={{ opacity: 1, y: 0 }}
   306→            transition={{ duration: 0.3, delay: 0.06 }}
   307→            className="metric-card min-w-0"
   308→          >
   309→            <SectionHeader title="Weekly Study Time" subtitle="Minutes per day — last 7 days" />
   310→            {weekBarData.every((d) => d.minutes === 0) ? (
   311→              <EmptyState
   312→                icon={BarChart3}
   313→                title="No study data this week"
   314→                description="Log study sessions to see your daily chart"
   315→                className="py-8"
   316→              />
   317→            ) : (
   318→              <ResponsiveContainer width="100%" height={200} maintainAspectRatio={false}>
   319→                <BarChart data={weekBarData} margin={{ top: 4, right: 4, left: -16, bottom: 0 }}>
   320→                  <CartesianGrid strokeDasharray="3 3" stroke={`rgba(${chartTheme.primaryRgb}, 0.08)`} vertical={false} />
   321→                  <XAxis
   322→                    dataKey="day"
   323→                    tick={{ fontSize: 11, fill: chartTheme.foreground }}
   324→                    axisLine={false}
   325→                    tickLine={false}
   326→                  />
   327→                  <YAxis
   328→                    tick={{ fontSize: 11, fill: chartTheme.foreground }}
   329→                    axisLine={false}
   330→                    tickLine={false}
   331→                  />
   332→                  <Tooltip
   333→                    contentStyle={tooltipStyle}
   334→                    cursor={{ fill: `rgba(${chartTheme.primaryRgb}, 0.06)` }}
   335→                    formatter={(value: number) => [`${value}m`, 'Study time']}
   336→                  />
   337→                  <Bar
   338→                    dataKey="minutes"
   339→                    fill={chartTheme.primary}
   340→                    radius={[6, 6, 0, 0]}
   341→                    maxBarSize={40}
   342→                  />
   343→                </BarChart>
   344→              </ResponsiveContainer>
   345→            )}
   346→          </motion.div>
   347→
   348→          {/* CGPA Trend Line Chart */}
   349→          <motion.div
   350→            initial={{ opacity: 0, y: 8 }}
   351→            animate={{ opacity: 1, y: 0 }}
   352→            transition={{ duration: 0.3, delay: 0.08 }}
   353→            className="metric-card min-w-0"
   354→          >
   355→            <SectionHeader title="Assessment Performance" subtitle="Score trend across assessments" />
   356→            {cgpaTrendData.length === 0 ? (
   357→              <EmptyState
   358→                icon={BarChart3}
   359→                title="No assessments yet"
   360→                description="Record CA marks to see your performance trend"
   361→                className="py-8"
   362→              />
   363→            ) : (
   364→              <ResponsiveContainer width="100%" height={200} maintainAspectRatio={false}>
   365→                <LineChart data={cgpaTrendData} margin={{ top: 4, right: 4, left: -16, bottom: 0 }}>
   366→                  <CartesianGrid strokeDasharray="3 3" stroke={`rgba(${chartTheme.primaryRgb}, 0.08)`} vertical={false} />
   367→                  <XAxis
   368→                    dataKey="name"
   369→                    tick={{ fontSize: 10, fill: chartTheme.foreground }}
   370→                    axisLine={false}
   371→                    tickLine={false}
   372→                  />
   373→                  <YAxis
   374→                    domain={[0, 100]}
   375→                    tick={{ fontSize: 11, fill: chartTheme.foreground }}
   376→                    axisLine={false}
   377→                    tickLine={false}
   378→                  />
   379→                  <Tooltip
   380→                    contentStyle={tooltipStyle}
   381→                    formatter={(value: number) => [`${value}%`, 'Score']}
   382→                  />
   383→                  <Line
   384→                    type="monotone"
   385→                    dataKey="percentage"
   386→                    stroke={chartTheme.primary}
   387→                    strokeWidth={2.5}
   388→                    dot={{ r: 4, fill: chartTheme.primary, stroke: chartTheme.card, strokeWidth: 2 }}
   389→                    activeDot={{ r: 6, fill: chartTheme.primary, stroke: chartTheme.card, strokeWidth: 2 }}
   390→                  />
   391→                </LineChart>
   392→              </ResponsiveContainer>
   393→            )}
   394→          </motion.div>
   395→
   396→          {/* Study Distribution Donut Chart */}
   397→          <motion.div
   398→            initial={{ opacity: 0, y: 8 }}
   399→            animate={{ opacity: 1, y: 0 }}
   400→            transition={{ duration: 0.3, delay: 0.1 }}
   401→            className="metric-card min-w-0"
   402→          >
   403→            <SectionHeader title="Study Distribution" subtitle="Time per subject this week" />
   404→            {pieData.length === 0 ? (
   405→              <EmptyState
   406→                icon={BarChart3}
   407→                title="No study data"
   408→                description="Start a study session to see your distribution"
   409→                className="py-8"
   410→              />
   411→            ) : (
   412→              <div className="flex items-center gap-4">
   413→                <ResponsiveContainer width="50%" height={200} maintainAspectRatio={false}>
   414→                  <PieChart>
   415→                    <Pie
   416→                      data={pieData}
   417→                      cx="50%"
   418→                      cy="50%"
   419→                      innerRadius={50}
   420→                      outerRadius={80}
   421→                      paddingAngle={2}
   422→                      dataKey="value"
   423→                    >
   424→                      {pieData.map((entry, index) => (
   425→                        <Cell key={index} fill={entry.color} stroke="none" />
   426→                      ))}
   427→                    </Pie>
   428→                    <Tooltip
   429→                      contentStyle={tooltipStyle}
   430→                      formatter={(value: number) => [`${value}m`, 'Study time']}
   431→                    />
   432→                  </PieChart>
   433→                </ResponsiveContainer>
   434→                <div className="flex-1 space-y-2">
   435→                  {pieData.slice(0, 5).map((d) => (
   436→                    <div key={d.name} className="flex items-center gap-2">
   437→                      <span className="status-dot" style={{ backgroundColor: d.color }} />
   438→                      <span className="text-xs text-muted-foreground truncate flex-1">{d.name}</span>
   439→                      <span className="text-xs font-semibold tabular-nums">{d.value}m</span>
   440→                    </div>
   441→                  ))}
   442→                </div>
   443→              </div>
   444→            )}
   445→          </motion.div>
   446→        </div>
   447→      </div>
   448→
   449→      {/* Charts Grid */}
   450→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 md:gap-4 mb-4 md:mb-6">
   451→        {/* Study Distribution */}
   452→        <motion.div
   453→          initial={{ opacity: 0, y: 8 }}
   454→          animate={{ opacity: 1, y: 0 }}
   455→          transition={{ duration: 0.3, delay: 0.1 }}
   456→          className="metric-card"
   457→        >
   458→          <SectionHeader title="Study Distribution" subtitle="This week by subject" />
   459→          {studyDist.length === 0 ? (
   460→            <EmptyState
   461→              icon={BarChart3}
   462→              title="No study data"
   463→              description="Start a study session to see your distribution"
   464→              className="py-8"
   465→            />
   466→          ) : (
   467→            <div className="space-y-3">
   468→              {studyDist.map((d) => {
   469→                const mx = Math.max(...studyDist.map((x) => x.minutes), 1);
   470→                return (
   471→                  <div key={d.subjectId}>
   472→                    <div className="flex justify-between items-center mb-1.5">
   473→                      <div className="flex items-center gap-2">
   474→                        <span className="status-dot" style={{ backgroundColor: d.color }} />
   475→                        <span className="text-xs font-medium truncate max-w-[140px]">{d.name}</span>
   476→                      </div>
   477→                      <span className="text-xs font-semibold tabular-nums">{d.minutes}m</span>
   478→                    </div>
   479→                    <div className="progress-thin">
   480→                      <div style={{ width: `${(d.minutes / mx) * 100}%`, backgroundColor: d.color }} />
   481→                    </div>
   482→                  </div>
   483→                );
   484→              })}
   485→            </div>
   486→          )}
   487→        </motion.div>
   488→
   489→        {/* Attendance Overview */}
   490→        <motion.div
   491→          initial={{ opacity: 0, y: 8 }}
   492→          animate={{ opacity: 1, y: 0 }}

   518→        {/* Syllabus Coverage */}

   619→          {uniqueSubjects === 0 && weekMinutes === 0 && (
