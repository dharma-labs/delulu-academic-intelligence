   802→  const currentView = useStore((s) => s.currentView);
   803→  const fabRef = useRef<HTMLDivElement>(null);
   804→

   806→  const prevViewRef = useRef(currentView);
   807→  /* eslint-disable react-hooks/set-state-in-effect */
   808→  useEffect(() => {
   809→    if (prevViewRef.current !== currentView) {
   810→      prevViewRef.current = currentView;
   811→      setOpen(false);
   812→    }
   813→  }, [currentView]);
   814→  /* eslint-enable react-hooks/set-state-in-effect */

   820→    if (!open) return;

   822→      if (fabRef.current && !fabRef.current.contains(e.target as Node)) {
   823→        setOpen(false);
   824→      }
   825→    };
   826→    // Delay to avoid immediate close from the toggle click
   827→    const timer = setTimeout(() => {
   828→      document.addEventListener('mousedown', handleClick);
   829→    }, 0);
   830→    return () => {
   831→      clearTimeout(timer);
   832→      document.removeEventListener('mousedown', handleClick);
   833→    };
   834→  }, [open]);
   835→
   836→  const handleAction = (action: typeof FAB_ACTIONS[number]) => {
   837→    setOpen(false);
   838→    if (action.customAction === 'quick-note') {
   839→      setQuickNoteOpen(true);

   841→      navigate(action.view);
   842→    }
   843→  };
   844→
   845→  return (
   846→    <div ref={fabRef} className="fixed bottom-[76px] right-4 z-30 md:hidden">
   847→      {/* Backdrop overlay when open */}
   848→      <AnimatePresence>
   849→        {open && (
   850→          <motion.div
   851→            initial={{ opacity: 0 }}
   852→            animate={{ opacity: 1 }}
   853→            exit={{ opacity: 0 }}
   854→            transition={{ duration: 0.15 }}
   855→            className="fixed inset-0 z-[-1]"
   856→            onClick={() => setOpen(false)}
   857→          />
   858→        )}
   859→      </AnimatePresence>
   860→
   861→      {/* Radial menu items */}
   862→      <AnimatePresence>
   863→        {open && (
   864→          <div className="absolute bottom-16 right-0 flex flex-col items-end gap-2.5">
   865→            {FAB_ACTIONS.map((action, i) => {
   866→              const Icon = action.icon;
   867→              return (
   868→                <motion.div
   869→                  key={action.label}
   870→                  initial={{ opacity: 0, scale: 0.5, y: 10 }}
   871→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   872→                  exit={{ opacity: 0, scale: 0.5, y: 10 }}
   873→                  transition={{ duration: 0.2, delay: i * 0.04 }}

   916→export function AppShell() {
   917→  const subjects = useStore((s) => s.subjects);
   918→  const setCommandOpen = useStore((s) => s.setCommandOpen);
   919→  const [moreSheetOpen, setMoreSheetOpen] = useState(false);
   920→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
   921→  const { theme, setTheme } = useTheme();
   922→
   923→  const showShortcuts = useCallback(() => {
   924→    setCommandOpen(false);
   925→    setShortcutsOpen(true);
   926→  }, [setCommandOpen]);
   927→
   928→  // ── Auto-collapse sidebar on tablet (768–1024) ──
   929→  useEffect(() => {
   930→    const mql = window.matchMedia('(min-width: 768px) and (max-width: 1024px)');
   931→    const handler = (e: MediaQueryListEvent) => {
   932→      if (e.matches) {
   933→        useStore.setState({ sidebarCollapsed: true });
   934→      }
   935→    };
   936→    // Collapse immediately if already in tablet range
   937→    if (mql.matches) {
   938→      useStore.setState({ sidebarCollapsed: true });
   939→    }
   940→    mql.addEventListener('change', handler);
   941→    return () => mql.removeEventListener('change', handler);
   942→  }, []);
   943→
   944→  useEffect(() => {
   945→    const NUMBER_KEY_VIEWS: Record<string, ViewId> = {
   946→      '1': 'dashboard',
   947→      '2': 'subjects',
   948→      '3': 'marks',
   949→      '4': 'attendance',
   950→      '5': 'exams',
   951→      '6': 'focus',
   952→      '7': 'revision',
   953→      '8': 'notes',
   954→      '9': 'tasks',
   955→    };
   956→
   957→    const handleKeyDown = (e: KeyboardEvent) => {
   958→      // Don't capture when typing in inputs
   959→      const tag = (e.target as HTMLElement)?.tagName;
   960→      const isInput = tag === 'INPUT' || tag === 'TEXTAREA' || (e.target as HTMLElement)?.isContentEditable;
   961→      const hasModifier = e.metaKey || e.ctrlKey || e.altKey;
   962→
   963→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   964→        e.preventDefault();
   965→        setCommandOpen(true);
   966→      }
   967→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'd' || e.key === 'D')) {
   968→        e.preventDefault();
   969→        setTheme(theme === 'dark' ? 'light' : 'dark');
   970→      }
   971→      if ((e.metaKey || e.ctrlKey) && e.key === ',') {
   972→        e.preventDefault();
   973→        useStore.getState().navigate('settings');
   974→      }
   975→      if (!isInput && (e.key === '?' || (e.shiftKey && e.key === '/'))) {
   976→        e.preventDefault();
   977→        setShortcutsOpen((prev) => !prev);
   978→      }
   979→      // Number keys 1-9 for view navigation (no modifiers, not in input)
   980→      if (!isInput && !hasModifier && !e.shiftKey && NUMBER_KEY_VIEWS[e.key]) {
   981→        e.preventDefault();
   982→        useStore.getState().navigate(NUMBER_KEY_VIEWS[e.key]);
   983→      }
   984→      // D key for theme toggle (no modifiers, not in input)
   985→      if (!isInput && !hasModifier && e.key === 'd') {
   986→        e.preventDefault();
   987→        setTheme(theme === 'dark' ? 'light' : 'dark');
   988→      }
   989→      // N key for context-sensitive new action (no modifiers, not in input)
   990→      if (!isInput && !hasModifier && e.key === 'n') {
   991→        e.preventDefault();
   992→        document.dispatchEvent(new CustomEvent('shortcut:new-action'));
   993→      }
   994→    };
   995→    document.addEventListener('keydown', handleKeyDown);
   996→    return () => document.removeEventListener('keydown', handleKeyDown);
   997→  }, [setCommandOpen, theme, setTheme]);
   998→
   999→  return (
  1000→    <ToastProvider>
  1001→    <MobileFAB />
  1002→    <div className="flex min-h-screen">
  1003→      <DesktopSidebar />
  1004→
  1005→      <main className="flex-1 min-w-0 flex flex-col">
  1006→        <DesktopTopBar />
  1007→        <div className="content-area flex-1 px-4 md:px-5 lg:px-6 py-5 pb-[76px] md:pb-6">
  1008→          {/* Mobile header */}
  1009→          <div className="flex items-center justify-between mb-5 md:mb-0">
  1010→            <div className="flex items-center gap-2">
  1011→              <span className="md:hidden text-sm font-bold tracking-tight text-foreground">
  1012→                DELULU
  1013→              </span>
  1014→              <MobileStreakBadge />
  1015→            </div>
  1016→            <div className="flex items-center gap-1.5 ml-auto">
  1017→              <ThemeToggle />
  1018→              <button
  1019→                onClick={() => setCommandOpen(true)}
  1020→                className="md:hidden flex items-center justify-center h-9 w-9 rounded-lg border border-border bg-background/50 text-muted-foreground hover:bg-accent hover:text-foreground active:scale-95 transition-all duration-150"
  1021→                aria-label="Search"
  1022→              >
  1023→                <Search className="size-4" />
  1024→              </button>
  1025→            </div>
  1026→          </div>
  1027→
  1028→          {/* Onboarding — shows when no subjects exist */}
  1029→          {subjects.length === 0 && <Onboarding />}
  1030→
  1031→          <ViewRouter />
  1032→        </div>
  1033→      </main>
  1034→
  1035→      <MobileBottomNav onMoreOpen={() => setMoreSheetOpen(true)} />
  1036→      <MobileMoreSheet open={moreSheetOpen} onOpenChange={setMoreSheetOpen} />
  1037→      <CommandPalette onShowShortcuts={showShortcuts} />
  1038→      <ShortcutsOverlay open={shortcutsOpen} onOpenChange={setShortcutsOpen} />
  1039→    </div>
  1040→    </ToastProvider>
  1041→  );
  1042→}
