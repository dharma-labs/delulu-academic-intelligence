    25→function escapeField(value: string): string {
    26→  if (/[",\n\r]/.test(value)) return `"${value.replace(/"/g, '""')}"`;
    27→  return value;
    28→}

    81→      }));

    87→          }))
    88→          .filter((d) => d.minutes > 0)
    89→          .sort((a, b) => b.minutes - a.minutes),

    93→      setCompletionRate(active.length > 0 ? Math.round(syllabusProgresses.reduce((a, p) => a + p, 0) / active.length) : 0);
    94→      setThreshold(s.profile.attendanceThreshold);
    95→
    96→      // Compute summary card data

    98→      setTotalStudyHours(Math.round(allTotalSec / 3600 * 10) / 10);

   100→      const avgAtt = active.length > 0

   102→            const att = getSubjectAttendance({ attendance: s.attendance }, sub.id);

   104→          }, 0) / active.length)

   469→                </LineChart>
   470→              </ResponsiveContainer>
   471→            )}
   472→          </motion.div>
   473→
   474→          {/* Study Distribution Donut Chart */}
   475→          <motion.div
   476→            initial={{ opacity: 0, y: 8 }}
   477→            animate={{ opacity: 1, y: 0 }}
   478→            transition={{ duration: 0.3, delay: 0.1 }}
   479→            className="metric-card min-w-0"
   480→          >
   481→            <SectionHeader title="Study Distribution" subtitle="Time per subject this week" />
   482→            {pieData.length === 0 ? (
   483→              <EmptyState
   484→                icon={BarChart3}
   485→                title="No study data"
   486→                description="Start a study session to see your distribution"
   487→                className="py-8"
   488→              />
   489→            ) : (
   490→              <div className="flex items-center gap-4">
   491→                <ResponsiveContainer width="50%" height={200} maintainAspectRatio={false}>
   492→                  <PieChart>
   493→                    <Pie
   494→                      data={pieData}
   495→                      cx="50%"
   496→                      cy="50%"
   497→                      innerRadius={50}
   498→                      outerRadius={80}
   499→                      paddingAngle={2}
   500→                      dataKey="value"
   501→                    >
   502→                      {pieData.map((entry, index) => (
   503→                        <Cell key={index} fill={entry.color} stroke="none" />
   504→                      ))}
   505→                    </Pie>
   506→                    <Tooltip
   507→                      contentStyle={tooltipStyle}
   508→                      formatter={(value: number) => [`${value}m`, 'Study time']}
   509→                    />
   510→                  </PieChart>
   511→                </ResponsiveContainer>
   512→                <div className="flex-1 space-y-2">
   513→                  {pieData.slice(0, 5).map((d) => (
   514→                    <div key={d.name} className="flex items-center gap-2">
   515→                      <span className="status-dot" style={{ backgroundColor: d.color }} />
   516→                      <span className="text-xs text-muted-foreground truncate flex-1">{d.name}</span>
   517→                      <span className="text-xs font-semibold tabular-nums">{d.value}m</span>
   518→                    </div>
   519→                  ))}
   520→                </div>
   521→              </div>
   522→            )}
   523→          </motion.div>
   524→        </div>
   525→      </div>
   526→
   527→      {/* Charts Grid */}
   528→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 md:gap-4 mb-4 md:mb-6">
   529→        {/* Study Distribution */}
   530→        <motion.div
   531→          initial={{ opacity: 0, y: 8 }}
   532→          animate={{ opacity: 1, y: 0 }}
   533→          transition={{ duration: 0.3, delay: 0.1 }}
   534→          className="metric-card"
   535→        >
   536→          <SectionHeader title="Study Distribution" subtitle="This week by subject" />
   537→          {studyDist.length === 0 ? (
   538→            <EmptyState
   539→              icon={BarChart3}
   540→              title="No study data"
   541→              description="Start a study session to see your distribution"
   542→              className="py-8"
   543→            />
   544→          ) : (
   545→            <div className="space-y-3">
   546→              {studyDist.map((d) => {
   547→                const mx = Math.max(...studyDist.map((x) => x.minutes), 1);
   548→                return (
   549→                  <div key={d.subjectId}>
   550→                    <div className="flex justify-between items-center mb-1.5">
   551→                      <div className="flex items-center gap-2">
   552→                        <span className="status-dot" style={{ backgroundColor: d.color }} />
   553→                        <span className="text-xs font-medium truncate max-w-[140px]">{d.name}</span>
   554→                      </div>
   555→                      <span className="text-xs font-semibold tabular-nums">{d.minutes}m</span>
   556→                    </div>
   557→                    <div className="progress-thin">
   558→                      <div style={{ width: `${(d.minutes / mx) * 100}%`, backgroundColor: d.color }} />
   559→                    </div>
   560→                  </div>
   561→                );
   562→              })}
   563→            </div>
   564→          )}
   565→        </motion.div>
   566→
   567→        {/* Attendance Overview */}
   568→        <motion.div
   569→          initial={{ opacity: 0, y: 8 }}
   570→          animate={{ opacity: 1, y: 0 }}
   571→          transition={{ duration: 0.3, delay: 0.12 }}
   572→          className="metric-card"
   573→        >
   574→          <SectionHeader title="Attendance Overview" subtitle="Per subject" />
   575→          {attendanceTrend.length === 0 ? (
   576→            <EmptyState
   577→              icon={Activity}
   578→              title="No attendance data"
   579→              description="Mark your attendance to track trends"
   580→              className="py-8"
   581→            />
   582→          ) : (
   583→            <div className="space-y-3">
   584→              {attendanceTrend.map((a) => (
   585→                <CompactProgress
   586→                  key={a.id}
   587→                  label={a.name}
   588→                  value={a.percentage}
   589→                  color={progressColorClass(a.percentage)}
   590→                />
   591→              ))}
   592→            </div>
   593→          )}
   594→        </motion.div>
   595→
   596→        {/* Syllabus Coverage */}
   597→        <motion.div
   598→          initial={{ opacity: 0, y: 8 }}
   599→          animate={{ opacity: 1, y: 0 }}
   600→          transition={{ duration: 0.3, delay: 0.14 }}
   601→          className="metric-card"
   602→        >
   603→          <SectionHeader title="Syllabus Coverage" subtitle="Per subject" />
   604→          {syllabusTrend.length === 0 ? (
   605→            <EmptyState
   606→              icon={Target}
   607→              title="No subjects"
   608→              description="Add subjects to track syllabus progress"
   609→              className="py-8"
   610→            />
   611→          ) : (
   612→            <div className="space-y-3">
   613→              {syllabusTrend.map((s) => (
   614→                <CompactProgress
   615→                  key={s.id}
   616→                  label={s.name}
   617→                  value={s.percentage}
   618→                  color={progressColorClass(s.percentage)}
   619→                />
   620→              ))}
   621→            </div>
   622→          )}
   623→        </motion.div>
   624→
   625→        {/* Recent Assessments */}
   626→        <motion.div
   627→          initial={{ opacity: 0, y: 8 }}
   628→          animate={{ opacity: 1, y: 0 }}
   629→          transition={{ duration: 0.3, delay: 0.16 }}
   630→          className="metric-card"
   631→        >
   632→          <SectionHeader title="Recent Assessments" subtitle="Last 5 performance scores" />
   633→          {recentAssessments.length === 0 ? (
   634→            <EmptyState
   635→              icon={BarChart3}
   636→              title="No assessments yet"
   637→              description="Record your CA marks to see performance trends"
   638→              className="py-8"
   639→            />
   640→          ) : (
   641→            <div className="space-y-3">
   642→              {recentAssessments.map((a, i) => {
   643→                const pct = (a.obtainedMarks / a.maxMarks) * 100;
   644→                return (
   645→                  <div key={a.id}>
   646→                    <div className="flex justify-between items-center mb-1.5">
   647→                      <div className="flex items-center gap-2">
   648→                        <span className="text-[10px] text-muted-foreground tabular-nums w-4">{i + 1}.</span>
   649→                        <span className="text-xs font-medium truncate max-w-[140px]">{a.name}</span>
   650→                      </div>
   651→                      <span className={`text-xs font-semibold tabular-nums ${getColor(pct)}`}>
   652→                        {pct.toFixed(0)}%
   653→                      </span>
   654→                    </div>
   655→                    <div className="progress-thin">
   656→                      <div
   657→                        className={progressColorClass(pct) === 'green' ? 'bg-emerald-500 dark:bg-emerald-400' :
   658→                          progressColorClass(pct) === 'blue' ? 'bg-blue-500 dark:bg-blue-400' :
   659→                          progressColorClass(pct) === 'amber' ? 'bg-amber-500 dark:bg-amber-400' :
   660→                          'bg-red-500 dark:bg-red-400'}
   661→                        style={{ width: `${pct}%` }}
   662→                      />
   663→                    </div>
   664→                  </div>
   665→                );
   666→              })}
   667→            </div>
   668→          )}
   669→        </motion.div>
   670→      </div>
   671→
   672→      {/* Insights */}
   673→      {hasInsights && (
   674→        <motion.div
   675→          initial={{ opacity: 0, y: 8 }}
   676→          animate={{ opacity: 1, y: 0 }}
   677→          transition={{ duration: 0.3, delay: 0.2 }}
   678→          className="space-y-2"
   679→        >
   680→          <SectionHeader title="Insights" subtitle="Based on your data" />
   681→          {weekMinutes > 0 && (
   682→            <InsightCard
   683→              type="info"
   684→              icon={Activity}
   685→              title={`Studied ${weekMinutes} minutes this week`}
   686→              description={`Across ${weekSessions} sessions covering ${uniqueSubjects} subject${uniqueSubjects > 1 ? 's' : ''}`}
   687→            />
   688→          )}
   689→          {belowThresholdCount > 0 && (
   690→            <InsightCard
   691→              type="warning"
   692→              icon={Flame}
   693→              title={`${belowThresholdCount} subject${belowThresholdCount > 1 ? 's' : ''} below ${threshold}% attendance`}
   694→              description="Consider attending more classes to meet your threshold"
   695→            />
   696→          )}
   697→          {uniqueSubjects === 0 && weekMinutes === 0 && (
   698→            <InsightCard
   699→              type="positive"
   700→              icon={Activity}
   701→              title="Start studying this week"
   702→              description="Log study sessions to unlock detailed analytics"
   703→            />
   704→          )}
   705→        </motion.div>
   706→      )}
   707→    </div>
   708→  );
   709→}
